// Stagger / 交错入场
// Pack: motion (Motion graphics moves) · element 02 · Category: Entrances & text
// What happens: Chart bars rise one after another from the left, each slightly delayed.
// Motion spec:
//   - One pass per 2.35 s. Seven bars (width 44, gap 20, heights 52, 84, 68, 112, 92, 136, 158), coloured blue, coral, yellow, mint in rotation.
//   - Bar i (0-based) starts at 0.15 + 0.12·i s and grows from 0 to full height in 0.5 s with backOut (s = 2.6), anchored at the baseline, overshooting and settling.
//   - White dot (radius 6.5, 16 px above the bar top): starts 0.34 s after its bar with spring 1 − e^(−7t)·cos(16t).
//   - Exit: bars shrink with cubic ease-in over 0.26 s starting at 1.78 + 0.045·i s; dots vanish first, over 0.2 s from 1.70 + 0.045·i s.
// When to use it: Use it to bring in several items (list rows, chart bars, cards) in sequence and lead the eye across them.
// Tags: stagger, sequence, delay, cascade, bar-chart, list, entrance
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion -> https://motion-prompt.2077tech.com/api/v1/kit/motion.js
// Live demo: https://motion-prompt.2077tech.com/#002
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 02
// 02 スタガー（グラフの棒が、少しずつ時間をずらして順に立ち上がる）
CELLS[1] = (k, w, h, u) => {
  stage(k);
  const N = 7, BW = 44, G = 20, x0 = (w - (N * BW + (N - 1) * G)) / 2, base = h - 30;
  const HS = [52, 84, 68, 112, 92, 136, 158];
  k.fillStyle = C.navy3; rr(k, x0 - 18, base, N * BW + (N - 1) * G + 36, 5, 2.5); k.fill();
  const lu = mod(u, CYC / 2);
  for (let i = 0; i < N; i++) {
    const t0 = 0.15 + i * 0.12, g = backOut(P(lu, t0, t0 + 0.5), 2.6) * (1 - eIn(P(lu, 1.78 + i * 0.045, 2.04 + i * 0.045)));
    if (g <= 0.001) continue;
    const hh = HS[i] * g, x = x0 + i * (BW + G);
    k.fillStyle = PAL[i % 4]; rr(k, x, base - hh, BW, hh, [12, 12, 3, 3]); k.fill();
    // 棒の上の点（さらに遅れて出る）
    const d = spring(lu - t0 - 0.34, 16, 7) * (1 - eIn(P(lu, 1.7 + i * 0.045, 1.9 + i * 0.045)));
    if (d > 0.01) { k.fillStyle = '#fff'; k.beginPath(); k.arc(x + BW / 2, base - hh - 16, 6.5 * d, 0, 7); k.fill(); }
  }
};
