// Burst / 放射迸发
// Pack: motion (Motion graphics moves) · element 03 · Category: Entrances & text
// What happens: A like heart fills in, and a ring, lines and dots burst outward at the same moment.
// Motion spec:
//   - One burst per 2.35 s. A heart (size 46, 7 px outline) sits in the centre.
//   - Anticipation: 0.3–0.5 s, scale 1→0.7 (quadratic ease-in-out).
//   - Trigger at 0.5 s: the heart fills with coral and scale returns 0.7→1 with spring 1 − e^(−6.5t)·cos(16t).
//   - Ring: yellow, radius 44→90 while line width goes 16→0, over 0.6 s with quintic ease-out.
//   - 10 radial lines (yellow/mint alternating, width 9): outer end radius 64→130 (y scaled 0.82); length 36 shrinks to 0 between 0.22 and 0.7 s after the trigger.
//   - 10 dots (coral/blue alternating, radius 8→4.8): radius 58→150 on an ellipse (x ×1.45, y ×0.78), fading out 0.3–0.72 s after the trigger. The heart fill fades at 1.92–2.2 s.
// When to use it: Use it to mark the instant something is triggered or achieved, such as a like, a success or a correct answer.
// Tags: burst, explosion, radial, like, heart, celebrate, accent, sparkle
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion -> https://motion-prompt.2077tech.com/api/v1/kit/motion.js
// Live demo: https://motion-prompt.2077tech.com/#003
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 03
// 03 バースト（「いいね」のハートが点くと同時に、まわりに線と粒が弾ける）
CELLS[2] = (k, w, h, u) => {
  stage(k);
  const T = CYC / 2, lu = mod(u, T), X = w / 2, Y = h / 2 + 2, t0 = 0.5, tt = lu - t0;
  const ant = eIOs(P(lu, 0.3, t0)), off = P(lu, 1.92, 2.2);
  const sc = tt < 0 ? 1 - 0.3 * ant : 0.7 + 0.3 * spring(tt, 16, 6.5) + 0.0;
  if (tt > 0 && tt < 0.75) {
    const q = eOut5(tt / 0.6 > 1 ? 1 : tt / 0.6), fade = 1 - P(tt, 0.3, 0.72);
    // 輪
    k.strokeStyle = C.yel; k.lineWidth = Math.max(0.1, 16 * (1 - q)); k.beginPath(); k.arc(X, Y, 44 + 46 * q, 0, 7); if (q < 0.98) k.stroke();
    // 放射の線
    k.lineCap = 'round';
    for (let i = 0; i < 10; i++) {
      const a = i / 10 * Math.PI * 2 - Math.PI / 2, r1 = 64 + 66 * q, len = 36 * (1 - P(tt, 0.22, 0.7));
      if (len > 0.5) { k.strokeStyle = i % 2 ? C.mint : C.yel; k.lineWidth = 9; k.beginPath(); k.moveTo(X + Math.cos(a) * (r1 - len), Y + Math.sin(a) * (r1 - len) * 0.82); k.lineTo(X + Math.cos(a) * r1, Y + Math.sin(a) * r1 * 0.82); k.stroke(); }
      const a2 = a + Math.PI / 10, r2 = 58 + 92 * q;
      k.fillStyle = i % 2 ? C.coral : C.blue; k.globalAlpha = fade; k.beginPath(); k.arc(X + Math.cos(a2) * r2 * 1.45, Y + Math.sin(a2) * r2 * 0.78, 8 * (1 - q * 0.4), 0, 7); k.fill(); k.globalAlpha = 1;
    }
  }
  k.save(); k.translate(X, Y - 2); k.scale(sc, sc);
  heartPath(k, 46); k.lineJoin = 'round'; k.lineWidth = 7; k.strokeStyle = C.navy3; k.stroke();
  const fa = tt < 0 ? 0 : 1 - off;
  if (fa > 0) { k.globalAlpha = fa; k.fillStyle = C.coral; k.strokeStyle = C.coral; k.fill(); k.stroke(); k.globalAlpha = 1; }
  k.restore();
};
