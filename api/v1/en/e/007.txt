// Count-up / 数字递增
// Pack: motion (Motion graphics moves) · element 07 · Category: Entrances & text
// What happens: A number races upward and locks onto its final value.
// Motion spec:
//   - One pass per 4.7 s. Fades in at 0.1–0.4 s and out at 4.0–4.4 s.
//   - Number (bold 104 px): 0→2,480 over 0.45–2.9 s. Value = round(target × (1 − (1 − p)^4)): fast at first, slow at the end.
//   - Thousands separator included. Every digit has the same width and the number is right-aligned, so it does not jitter sideways.
//   - Lands at 2.9 s: colour switches white→yellow and scale does a small bounce, 1 + 0.07·e^(−7t)·cos(15t).
//   - Up-arrow chip on the left (mint circle, radius 34): appears from 0.35 s with spring 1 − e^(−7t)·cos(14t); while counting it shakes vertically by 6 px × (1 − p)^3 × sin(30u).
//   - Bar below (330 × 12, blue→mint gradient): fills in proportion to value ÷ target.
// When to use it: Use it to make a figure feel big or growing: revenue, user counts, completion rates.
// Tags: count-up, counter, number, odometer, stats, kpi, progress
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion -> https://motion-prompt.2077tech.com/api/v1/kit/motion.js
// Live demo: https://motion-prompt.2077tech.com/#007
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 07
// 07 カウントアップ（数字が勢いよく増えて、最後にピタッと決まる）
CELLS[6] = (k, w, h, u) => {
  stage(k);
  const TARGET = 2480, a = P(u, 0.1, 0.4) * (1 - P(u, 4.0, 4.4));
  const p = P(u, 0.45, 2.9), v = Math.round(TARGET * (1 - Math.pow(1 - p, 4)));
  const s = v.toLocaleString('en-US'), f = `800 104px ${DISP}`;
  k.font = f; const dw = k.measureText('0').width + 2, cw = k.measureText(',').width;
  const full = TARGET.toLocaleString('en-US'); let tot = 0; for (const ch of full) tot += ch === ',' ? cw : dw;
  const land = u - 2.9, pop = 1 + (land > 0 ? 0.07 * Math.exp(-7 * land) * Math.cos(15 * land) : 0);
  const X = w / 2 + 42, Y = 96;
  k.save(); k.globalAlpha = a; k.translate(X, Y); k.scale(pop, pop);
  let x = tot / 2;   // 右そろえ
  const speed = p < 1 ? Math.pow(1 - p, 3) : 0;
  for (const ch of [...s].reverse()) {
    const cwid = ch === ',' ? cw : dw; x -= cwid;
    txt(k, ch, x + cwid / 2, 0, f, land > 0 ? C.yel : '#fff', 'center', 'middle');
  }
  k.restore();
  // 上向きの矢印チップ
  const ar = spring(u - 0.35, 14, 7) * (1 - P(u, 4.0, 4.3)), bob = speed * 6 * Math.sin(u * 30);
  if (ar > 0.01) { k.save(); k.translate(X - tot / 2 - 58, Y - 2 + bob); k.scale(ar, ar); k.fillStyle = C.mint; k.beginPath(); k.arc(0, 0, 34, 0, 7); k.fill(); k.strokeStyle = C.navy; k.lineWidth = 8; k.lineCap = 'round'; k.lineJoin = 'round'; k.beginPath(); k.moveTo(0, 15); k.lineTo(0, -14); k.moveTo(-13, -1); k.lineTo(0, -15); k.lineTo(13, -1); k.stroke(); k.restore(); }
  // 下のバー
  const bw = 330, bx = w / 2 - bw / 2, by = 168;
  k.globalAlpha = a; k.fillStyle = C.navy3; rr(k, bx, by, bw, 12, 6); k.fill();
  const g = k.createLinearGradient(bx, 0, bx + bw, 0); g.addColorStop(0, C.blue); g.addColorStop(1, C.mint);
  k.fillStyle = g; rr(k, bx, by, Math.max(12, bw * (v / TARGET)), 12, 6); k.fill(); k.globalAlpha = 1;
};
