// Particles / 粒子效果
// Pack: motion (Motion graphics moves) · element 15 · Category: Motion polish
// What happens: Particles keep shooting up from an emitter, spread out and fall.
// Motion spec:
//   - 84 particles, emitted from the bottom centre (horizontal centre, 20 px above the bottom edge).
//   - Initial velocity: ±190 px/s horizontally, 250–400 px/s upward. Gravity 470 px/s² (parabolic paths).
//   - Each particle repeats with a period of 2.35 s or about 1.57 s, with a random start offset (fixed seed, so every run looks the same).
//   - Lifetime up to 1.5 s: fade in over the first 0.06 s, fade out over the last 30%. Scale 1.15→0.7.
//   - Shapes are circles, squares and triangles (size 5–11 px) in blue, coral, yellow and mint, spinning at up to ±4.5 rad/s.
// When to use it: Use it to add celebration or liveliness, in place of confetti or sparks.
// Tags: particles, confetti, emitter, fountain, gravity, celebration, sparks
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion -> https://motion-prompt.2077tech.com/api/v1/kit/motion.js
// Live demo: https://motion-prompt.2077tech.com/#015
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 15
// 15 パーティクル（粒が次々と噴き出して、飛び散って落ちる）
const PARTS = (() => { const R = rand(15), a = []; for (let i = 0; i < 84; i++) { const n = R() < 0.5 ? 2 : 3; a.push({ T: CYC / n, ph: R() * CYC, vx: (R() - 0.5) * 2 * 190, vy: -(250 + R() * 150), sz: 5 + R() * 6, col: PAL[i % 4], kind: i % 3, spin: (R() - 0.5) * 9 }); } return a; })();
CELLS[14] = (k, w, h, u) => {
  stage(k);
  const X = w / 2, Y = h - 20, G = 470;
  for (const p of PARTS) {
    const t = mod(u + p.ph, p.T), life = Math.min(p.T, 1.5); if (t > life) continue;
    const x = X + p.vx * t, y = Y + p.vy * t + 0.5 * G * t * t, a = Math.min(1, t / 0.06) * (1 - smooth(P(t, life * 0.7, life))), sc = lerp(1.15, 0.7, t / life);
    k.globalAlpha = a; k.fillStyle = p.col; k.save(); k.translate(x, y); k.rotate(t * p.spin); k.scale(sc, sc);
    if (p.kind === 0) { k.beginPath(); k.arc(0, 0, p.sz, 0, 7); k.fill(); }
    else if (p.kind === 1) { k.fillRect(-p.sz, -p.sz, p.sz * 2, p.sz * 2); }
    else { k.beginPath(); k.moveTo(0, -p.sz * 1.3); k.lineTo(p.sz * 1.2, p.sz); k.lineTo(-p.sz * 1.2, p.sz); k.closePath(); k.fill(); }
    k.restore();
  }
  k.globalAlpha = 1;
  k.fillStyle = C.navy3; rr(k, X - 70, h - 12, 140, 30, 10); k.fill();
  k.fillStyle = '#fff'; rr(k, X - 30, h - 22, 60, 30, 10); k.fill();
};
