// Liquid transition / 液态转场
// Pack: motion (Motion graphics moves) · element 19 · Category: Transitions & highlights
// What happens: Paint-like liquid drips down from the top and fills the frame, becoming the next scene.
// Motion spec:
//   - Day → night → day, 2.35 s each. Two liquid layers, both moving top to bottom with quadratic ease-in-out.
//   - Layer 1 (flat colour: blue for day→night, coral for night→day): 0.35–1.5 s. Layer 2 (the next scene itself, clipped to the liquid shape): 0.7–1.8 s.
//   - The liquid front moves from y = −110 to 130 px below the bottom edge. Its edge waves with two sine waves (amplitudes 13 px and 8 px), strongest midway.
//   - Seven drips (30–54 px wide, round tips) hang from the front; length = base 30–110 px × (0.25 + 0.9 × wave strength), greatest midway.
//   - Merged look: draw the shape in white → blur 7 px → threshold the alpha → fill with the colour (or the next scene).
// When to use it: Use it for a soft, organic scene change, for example in food, beauty or creative videos.
// Tags: liquid, transition, drip, paint, goo, fluid, wipe, scene-change
// Size: 539x208 px · Cycle: 4.7 s
// Needs: kit motion -> https://motion-prompt.2077tech.com/api/v1/kit/motion.js
// Live demo: https://motion-prompt.2077tech.com/#019
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 19
// 19 リキッドトランジション（ペンキのような液体が上から垂れて流れ込み、次の画面になる）
const DRIPS = [0, 1].map(s => { const R = rand(190 + s * 7), a = []; for (let i = 0; i < 7; i++) a.push({ x: 36 + i * 78 + (R() - 0.5) * 30, len: 30 + R() * 80, wd: 15 + R() * 12, ph: R() * 6 }); return a; });
function liquidShape(g, w, h, p, set, u) {
  const base = lerp(-110, h + 130, p), amp = Math.sin(Math.PI * clamp(p * 1.1));
  g.beginPath(); g.moveTo(-20, -40);
  for (let x = -20; x <= w + 20; x += 10) g.lineTo(x, base + amp * (13 * Math.sin(x * 0.021 + u * 3 + set) + 8 * Math.sin(x * 0.05 + u * 4.3)));
  g.lineTo(w + 20, -40); g.closePath(); g.fill();
  for (const d of DRIPS[set]) { const L = d.len * (0.25 + amp * 0.9), y0 = base - 20; g.beginPath(); g.roundRect(d.x - d.wd, y0, d.wd * 2, L + 20, d.wd); g.fill(); g.beginPath(); g.arc(d.x, y0 + L + 18, d.wd * 1.25, 0, 7); g.fill(); }
}
CELLS[18] = abCell((k, w, h, u, a, b) => {
  const p1 = eIOs(P(u, TA - 0.15, TB)), p2 = eIOs(P(u, TA + 0.2, TB + 0.3));
  SCR[a](k, w, h);
  if (p1 > 0 && p2 < 1) k.drawImage(goo(g => liquidShape(g, w, h, p1, 0, u), 7, g => { g.fillStyle = a === 0 ? C.blue : C.coral; g.fillRect(0, 0, w, h); }), 0, 0);
  if (p2 >= 1) SCR[b](k, w, h);
  else if (p2 > 0) k.drawImage(goo(g => liquidShape(g, w, h, p2, 1, u + 1), 7, g => SCR[b](g, w, h)), 0, 0);
});
