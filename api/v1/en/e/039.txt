// Time-lapse / 延时摄影
// Pack: camera (Camera moves & transitions) · element 15 · Category: Shooting Style & Time
// What happens: With a fixed camera, 06:00 to 18:00 is compressed into a few seconds; sun, clouds, people and shadows move in fast forward.
// Motion spec:
//   - Camera: completely fixed (height 1.6 m, pitch +0.08 rad, about 5° up)
//   - Timing: 12 hours pass in 3.7 s (0.4 s to 4.1 s), cubic ease-in-out (about cubic-bezier(0.65,0,0.35,1))
//   - Sun: travels left to right along an arc, angle a = -1.25 rad → +1.25 rad; position (sin a×120, cos a×60 - 8) m at 170 m depth
//   - Three clouds drift 140 m to the right in total, wrapping every 80 m; seven small figures each cross an 18 m width 6 times (odd and even ones in opposite directions)
//   - Building shadows fall away from the sun and swing as it moves (translucent grey)
//   - Return: cross-fade back to the start over the last 0.5 s; the top-right clock runs 06:00 → 18:00 in 10-minute steps
// When to use it: To show a long stretch of time, or how a city, sky or task changes, in a short clip.
// Tags: time-lapse, timelapse, fast forward, time compression, sun, clouds, day to night
// Size: 539x208 px · Cycle: 9.4 s
// Needs: kit camera -> https://motion-prompt.2077tech.com/api/v1/kit/camera.js
// Live demo: https://motion-prompt.2077tech.com/#039
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 15
// 15 タイムラプス: 太陽が動き、雲と人が早送り
CELLS[14] = shot(camCell((k, w, h, p, u) => {
  const a = lerp(-1.25, 1.25, p);
  const clouds = [[-20 + mod(p * 140 + 0, 80) - 10, 18, 90, 2.2], [-50 + mod(p * 140 + 40, 80), 24, 120, 3], [-10 + mod(p * 140 + 20, 80) - 30, 13, 70, 1.6]];
  const ex = [];
  for (let i = 0; i < 7; i++) { const ph = mod(p * 6 + i / 7, 1), dir = i % 2 ? 1 : -1; ex.push([dir * lerp(-9, 9, ph), 0, 10 + i * 1.3, (kk) => { kk.fillStyle = i % 3 ? C.g4 : C.blueM; kk.fillRect(-0.12, 0, 0.24, 0.9); kk.beginPath(); kk.arc(0, 1.05, 0.13, 0, 7); kk.fill(); }]); }
  world(k, w, h, cam({ y: 1.6, pitch: 0.08 }), {
    sun: [Math.sin(a) * 120, Math.cos(a) * 60 - 8, 170], sunCol: C.blueM, clouds, extra: ex,
    after: (kk, cc) => {   // 建物の影
      kk.fillStyle = 'rgba(160,160,165,.22)';
      BLD.slice(0, 6).forEach(([x, bw, bh]) => { const L = bh * 0.9; const q1 = proj(cc, x - bw / 2, 0, 48), q2 = proj(cc, x + bw / 2, 0, 48), q3 = proj(cc, x + bw / 2 - Math.sin(a) * L, 0, 48 - Math.cos(a) * L * 0.4 - 1), q4 = proj(cc, x - bw / 2 - Math.sin(a) * L, 0, 48 - Math.cos(a) * L * 0.4 - 1); kk.beginPath(); kk.moveTo(q1.x, q1.y); kk.lineTo(q2.x, q2.y); kk.lineTo(q3.x, q3.y); kk.lineTo(q4.x, q4.y); kk.fill(); });
    }
  });
}, p => { const m = Math.round(lerp(6 * 60, 18 * 60, p) / 10) * 10; return `${String(Math.floor(m / 60)).padStart(2, '0')}:${String(m % 60).padStart(2, '0')}`; }, 0.4, 4.1));
