// Freeze frame / 定格
// Pack: camera (Camera moves & transitions) · element 16 · Category: Shooting Style & Time
// What happens: The picture stops at the moment a jumping person reaches the top, then resumes from the same frame 1.4 s later.
// Motion spec:
//   - Camera: completely fixed (height 1.3 m); what stops is the footage time
//   - Freeze: stops at 1.6 s and resumes at 3.0 s (1.4 s); footage time runs normally for 0–1.6 s, stays at 1.6 s while frozen, then continues as u-1.4
//   - Subject: the jump of a person 7 m away is a parabola over footage time 0.6–2.6 s, peak 1.1 m; the freeze lands exactly on the peak (50% through)
//   - Nine falling leaves also stop during the freeze; they fall 1.5× as fast as in 13 Slow motion
//   - Freeze markers: a 5 px blue border and a pause icon at bottom centre (two blue bars on a 68×40 px white rounded box), fading in and out over 0.12 s
//   - Top-right readout shows "FREEZE" only while frozen; no cross-fade reset
// When to use it: To emphasise a decisive moment, or to pause so a name or caption can be placed over the frame.
// Tags: freeze frame, still frame, pause, hold, stop motion, highlight
// Size: 539x208 px · Cycle: 9.4 s
// Needs: kit camera -> https://motion-prompt.2077tech.com/api/v1/kit/camera.js
// Live demo: https://motion-prompt.2077tech.com/#040
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 16
// 16 フリーズフレーム
CELLS[15] = shot((k, w, h, u) => {
  // 跳躍の頂点で 1.4 秒止める
  const tj = u < 1.6 ? u : u < 3.0 ? 1.6 : u - 1.4;   // 実時間→映像時間
  const q = P(tj, 0.6, 2.6);
  const frozen = u >= 1.6 && u < 3.0;
  world(k, w, h, cam({ y: 1.3 }), { person: { z: 7, y: jumpY(q), armUp: q > 0 && q < 1 }, extra: leaves(tj, 1.5) });
  if (frozen) {
    const a = Math.min(1, (u - 1.6) / 0.12, (3.0 - u) / 0.12);
    k.save(); k.globalAlpha = a; k.strokeStyle = C.blue; k.lineWidth = 5; k.strokeRect(2.5, 2.5, w - 5, h - 5);
    k.fillStyle = 'rgba(255,255,255,.88)'; rr(k, w / 2 - 34, h - 58, 68, 40, 8); k.fill();
    k.fillStyle = C.blue; k.fillRect(w / 2 - 12, h - 49, 8, 22); k.fillRect(w / 2 + 4, h - 49, 8, 22); k.restore();
  }
  return frozen ? 'FREEZE' : '';
});
