// Smash cut / 猛切
// Pack: camera (Camera moves & transitions) · element 22 · Category: Transitions
// What happens: A quiet scene of a person asleep cuts with no warning straight to an intense one; the volume meter jumps at the same moment.
// Motion spec:
//   - Join: a straight cut at 2.4 s (0 frames); no transition effect, the contrast does the work
//   - Outgoing shot (0–2.4 s): fixed camera (height 1.2 m, focal length 330 px); a person lying down 7 m away, three "z" letters float up and fade; volume meter at about 0–1 of 8 bars (0.08±0.04)
//   - Incoming shot (2.4–4.2 s): focal length 2.6× (858 px) for a tight framing, rolled -0.12 rad (about -7°); the person has both arms raised, with 18 radial blue lines
//   - Shake in the incoming shot: roll ±0.03 rad (about 6 cycles per second) and yaw ±0.006 rad (about 8 per second); volume meter at 0.85±0.15, nearly full
//   - Return: cross-fade back to the quiet shot over the last 0.5 s (quadratic ease-in-out)
//   - Top-right readout shows "♪ 静か" (quiet) → "♪ 大音量" (loud)
// When to use it: For comedy or shock, showing in an instant the gap between calm and chaos or between expectation and reality.
// Tags: smash cut, hard cut, abrupt, contrast, quiet to loud, comedy, transition
// Size: 539x208 px · Cycle: 9.4 s
// Needs: kit camera -> https://motion-prompt.2077tech.com/api/v1/kit/camera.js
// Live demo: https://motion-prompt.2077tech.com/#046
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 22
// 22 スマッシュカット（静かな場面 → 急に激しい場面）
function meter(k, w, h, lv) {   // 音量メーター（左下）
  const x = 24, y = h - 22, n = 8;
  for (let i = 0; i < n; i++) { k.fillStyle = (i + 0.5) / n < lv ? C.blue : C.g2; k.fillRect(x + i * 9, y - 6 - i * 2.2, 6, 6 + i * 2.2); }
}
CELLS[21] = shot((k, w, h, u) => {
  const cut = 2.4, r = P(u, CYC - RESET, CYC);
  const quiet = (kk, uu) => {
    world(kk, w, h, cam({ y: 1.2 }), {
      buildings: false, sun: [24, 26, 170], sunCol: C.blueXL, extra: [[0.4, 0, 7, (k2) => { k2.fillStyle = C.blue; rr(k2, -0.8, 0, 1.6, 0.34, 0.15); k2.fill(); k2.beginPath(); k2.arc(-0.95, 0.24, 0.17, 0, 7); k2.fill(); }]],
      after: (k2, cc) => { const p0 = proj(cc, -0.6, 1.0, 7); for (let i = 0; i < 3; i++) { const ph = mod(uu * 0.35 + i / 3, 1); k2.globalAlpha = Math.sin(ph * Math.PI) * 0.8; txt(k2, 'z', p0.x - 10 - ph * 20 - i * 4, p0.y - ph * 50, `600 ${16 + i * 4}px ${EN}`, C.sub, 'center', 'middle'); } k2.globalAlpha = 1; }
    });
    meter(kk, w, h, 0.08 + 0.04 * Math.sin(uu * 5));
  };
  const loud = (kk, uu) => {
    const T = uu * 40;
    const c = cam({ y: 1.25, f: F0 * 2.6, roll: -0.12 + Math.sin(T) * 0.03, yaw: Math.sin(T * 1.3) * 0.006 });
    world(kk, w, h, c, {
      buildings: false, trees: false, mountains: false, grid: true, sun: null,
      person: { x: 0, z: 7, armUp: true, walk: 0.6 },
      after: (k2) => { k2.strokeStyle = C.blue; k2.lineWidth = 3.5; for (let i = 0; i < 18; i++) { const a = i / 18 * Math.PI * 2 + 0.1; const r0 = 92 + (i % 3) * 14; k2.beginPath(); k2.moveTo(Math.cos(a) * r0 * 1.7, Math.sin(a) * r0); k2.lineTo(Math.cos(a) * 500, Math.sin(a) * 300); k2.stroke(); } }
    });
    meter(kk, w, h, 0.85 + 0.15 * Math.sin(uu * 31));
  };
  if (u < cut) quiet(k, u);
  else blend(k, kk => loud(kk, u), kk => quiet(kk, 0), eIOs(r));
  return u < cut || r > .5 ? '♪ 静か' : '♪ 大音量';
});
