// Drawer (Off-canvas) / 抽屉面板
// Pack: ui (UI component names) · element 02 · Category: Opening & overlays
// What happens: Clicking "Menu" slides a tall panel in from the left; clicking outside slides it back.
// Motion spec:
//   - 4.5 s cycle on a 542×191 card. The cursor clicks the "Menu" button (110×34, top left) at 1.0 s.
//   - Open amount a: 0→1 over 1.0–1.38 s with easeOutCubic. Full-card scrim rgba(18,20,28) at opacity 0.36·a.
//   - Panel: white, 210 wide, full card height, x = −210 + 210·a so it slides in from the left edge. Shadow blur 22, alpha 0.25.
//   - Inside: an avatar (radius 14) and name at the top, then four menu rows 32 px apart; the first row is selected with a light-blue band.
//   - Close: the cursor clicks outside at (420, 120) at 3.3 s; a is multiplied by (1 − easeInOutCubic over 3.3–3.65 s) and the panel slides back.
// When to use it: Use it to keep navigation or filters off screen until needed, especially on mobile.
// Tags: drawer, off-canvas, offcanvas, side-menu, navigation-drawer, sheet, sidebar, slide-in
// Size: 542x191 px · Cycle: 4.5 s
// Needs: kit ui -> https://motion-prompt.2077tech.com/api/v1/kit/ui.js
// Live demo: https://motion-prompt.2077tech.com/#050
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 02
// 02 ドロワー（Drawer (Off-canvas)）
CELLS[1] = ui((w, h, t) => {
  strokeRR(24, 14, 110, 34, 8, C.line2, 1.5); text('メニュー', 79, 31, 15, '#555', 600, 'center');
  bar(160, 26, 140, C.line2, 11); pageLines(24, 70, 494); fillRR(24, 138, 494, 42, 8, C.block);
  const a = eo(seg(t, 1.0, 1.38)) * (1 - eio(seg(t, 3.3, 3.65)));
  if (a > 0) {
    X.fillStyle = `rgba(18,20,28,${.36 * a})`; X.fillRect(0, 0, w, h);
    const pw = 210, px = -pw + pw * a;
    shadowed(() => { X.fillStyle = '#fff'; X.fillRect(px, 0, pw, h); }, 22, 0, .25);
    circle(px + 34, 32, 14, C.mb); bar(px + 58, 27, 100, C.line2, 10);
    for (let i = 0; i < 4; i++) { const y = 62 + i * 32; if (i === 0) fillRR(px + 12, y - 2, pw - 24, 28, 6, C.lb); fillRR(px + 22, y + 5, 14, 14, 3, i === 0 ? C.blue : C.line2); bar(px + 46, y + 7, 90 + (i % 2) * 30, i === 0 ? C.sb : C.line); }
  }
  pointer([[.3, w + 30, h + 30], [.95, 70, 34], [1.6, 70, 34], [3.0, 420, 120], [3.6, 420, 120], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
