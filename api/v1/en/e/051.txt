// Bottom sheet / 底部面板
// Pack: ui (UI component names) · element 03 · Category: Opening & overlays
// What happens: A finger taps "View details" and a panel rises from the bottom; dragging its handle down closes it.
// Motion spec:
//   - 4.5 s cycle on a 542×191 card. A finger (semi-transparent circle, radius 17, white rim) taps the "View details" pill (170×36, bottom centre) at 1.0 s.
//   - Open amount: 0→1 over 1.05–1.45 s with easeOutCubic. Full-card scrim rgba(18,20,28) at opacity 0.36·a.
//   - Sheet: white, 136 tall, top corners radius 18, top edge at y = 191 − 136·a. Shadow blur 22, offset y −4. A 52×6 grab handle at top centre.
//   - Inside: a bold title bar and two rows with 26×26 icons.
//   - Close: the finger lands on the handle (271, 68) at 2.75 s and drags down over 3.0–3.6 s; a is multiplied by (1 − easeInOutCubic over 3.0–3.6 s), so the sheet follows the finger off the bottom.
// When to use it: Use it on mobile to show details or choices while keeping the current screen in view.
// Tags: bottom-sheet, sheet, modal-sheet, mobile, slide-up, drag-to-dismiss, panel
// Size: 542x191 px · Cycle: 4.5 s
// Needs: kit ui -> https://motion-prompt.2077tech.com/api/v1/kit/ui.js
// Live demo: https://motion-prompt.2077tech.com/#051
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 03
// 03 ボトムシート（Bottom sheet）
CELLS[2] = ui((w, h, t) => {
  pageLines(24, 16, 494); fillRR(24, 76, 239, 54, 8, C.block); fillRR(279, 76, 239, 54, 8, C.block);
  fillRR(186, 144, 170, 36, 18, C.blue); text('查看详情', 271, 162, 15, '#fff', 600, 'center');
  const open = eo(seg(t, 1.05, 1.45));
  const drag = eio(seg(t, 3.0, 3.6));
  const a = open * (1 - drag);
  if (open > 0 && a > 0.001) {
    X.fillStyle = `rgba(18,20,28,${.36 * a})`; X.fillRect(0, 0, w, h);
    const sh = 136, sy = h - sh * a;
    shadowed(() => { rr(0, sy, w, sh + 20, 18); X.fillStyle = '#fff'; X.fill(); }, 22, -4, .2);
    fillRR(w / 2 - 26, sy + 10, 52, 6, 3, C.line2);
    bar(28, sy + 30, 160, C.ink, 11);
    for (let i = 0; i < 2; i++) { const y = sy + 56 + i * 36; fillRR(28, y, 26, 26, 6, i ? C.mb : C.blue); bar(66, y + 4, 220, C.line); bar(66, y + 17, 140, C.line, 7); }
  }
  const sy = h - 136 * a;
  finger([[.3, 271, 162], [1.6, 271, 162], [2.75, w / 2, h - 136 + 13], [3.0, w / 2, h - 136 + 13], [3.6, w / 2, h + 10]], [1.0, 3.0], t, (t > .6 && t < 1.6) || (t > 2.75 && t < 3.6));
});
