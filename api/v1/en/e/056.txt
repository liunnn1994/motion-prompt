// Toast notification / 轻提示
// Pack: ui (UI component names) · element 08 · Category: Opening & overlays
// What happens: Clicking "Save" slides a dark notice into the bottom-right corner; when its time bar runs out it slides away.
// Motion spec:
//   - 4.5 s cycle on a 542×191 card. The cursor clicks the blue "Save" button (110×36, bottom left) at 1.0 s and leaves to the bottom left.
//   - Toast: #26282c, 214×52, radius 10, 18 px from the right and 16 px from the bottom; slides in 240 px from the right over 1.05–1.4 s with easeOutCubic.
//   - Inside: a white check in a blue circle and white 15 px text "Saved".
//   - A time-left line along the bottom (3 px tall) shrinks linearly from 194 px to 0 over 1.4–3.2 s.
//   - Exit: slides 240 px out to the right over 3.2–3.55 s with easeInOutCubic.
// When to use it: Use it to confirm a result such as a save or send without interrupting the user.
// Tags: toast, snackbar, notification, alert, message, auto-dismiss
// Size: 542x191 px · Cycle: 4.5 s
// Needs: kit ui -> https://motion-prompt.2077tech.com/api/v1/kit/ui.js
// Live demo: https://motion-prompt.2077tech.com/#056
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 08
// 08 トースト（Toast notification）
CELLS[7] = ui((w, h, t) => {
  bar(24, 16, 90, C.line2, 10); strokeRR(24, 34, 300, 34, 7, C.line2, 1.5); bar(24, 82, 90, C.line2, 10); strokeRR(24, 100, 300, 34, 7, C.line2, 1.5);
  fillRR(24, 146, 110, 36, 8, C.blue); text('保存', 79, 164, 16, '#fff', 600, 'center');
  const tin = eo(seg(t, 1.05, 1.4)), tout = eio(seg(t, 3.2, 3.55));
  if (tin > 0 && tout < 1) {
    const tw0 = 214, th = 52, tx = w - 18 - tw0 + (1 - tin) * 240 + tout * 240, ty = h - 16 - th;
    shadowed(() => fillRR(tx, ty, tw0, th, 10, C.dark), 18, 6, .25);
    circle(tx + 26, ty + 24, 11, C.blue); check(tx + 26, ty + 24, 5, '#fff', 2.4);
    text('保存しました', tx + 46, ty + 24, 15, '#fff', 600);
    const life = 1 - seg(t, 1.4, 3.2); fillRR(tx + 10, ty + th - 7, (tw0 - 20) * life, 3, 1.5, C.sb);
  }
  pointer([[.3, -30, h + 30], [.95, 84, 168], [1.5, 84, 168], [2.2, -40, h + 40]], [1.0], t);
});
