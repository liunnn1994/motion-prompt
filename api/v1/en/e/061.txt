// Filter chips / 筛选标签
// Pack: ui (UI component names) · element 13 · Category: Choosing & switching
// What happens: Clicking the "New" and "Free" chips adds a check mark and turns them blue, and cards that do not match fade out.
// Motion spec:
//   - 4.5 s cycle on a 542×191 card. Four chips ("All, New, Popular, Free" in Chinese): 34 tall, radius 17, width = text + 32, 10 px apart from the left.
//   - Clicks: New on at 0.95 s, Free on at 1.75 s, New off at 3.0 s, Free off at 3.6 s.
//   - Each change takes 0.15 s with easeOutCubic: the chip widens 18 px to make room for a check mark; fill light blue, border and text blue.
//   - List below: four cards (242×54, two columns, two rows); cards that do not match drop from opacity 1 to 0.12.
// When to use it: Use it to let people narrow a list or search results by combining a few quick filters.
// Tags: chips, filter-chips, tags, filter, facets, pills
// Size: 542x191 px · Cycle: 4.5 s
// Needs: kit ui -> https://motion-prompt.2077tech.com/api/v1/kit/ui.js
// Live demo: https://motion-prompt.2077tech.com/#061
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 13
// 13 チップ（Filter chips）
CELLS[12] = ui((w, h, t) => {
  const chips = ['全部', '最新', '热门', '免费']; let x = 24; const pos = [];
  const on = [0, eo(seg(t, 0.95, 1.1)) * (1 - eo(seg(t, 3.0, 3.15))), 0, eo(seg(t, 1.75, 1.9)) * (1 - eo(seg(t, 3.6, 3.75)))];
  chips.forEach((s, i) => {
    const k = on[i], cw0 = tw(s, 15, 600) + 32 + (k > 0 ? 18 * k : 0);
    fillRR(x, 12, cw0, 34, 17, k > .5 ? C.lb : '#fff'); strokeRR(x, 12, cw0, 34, 17, k > .5 ? C.blue : C.line2, 1.5);
    if (k > 0) { X.globalAlpha = k; check(x + 18, 29, 5, C.blue, 2.4); X.globalAlpha = 1; }
    text(s, x + 16 + 18 * k, 30, 15, k > .5 ? C.blue : '#555', 600);
    pos.push(x + cw0 / 2); x += cw0 + 10;
  });
  // 一覧：条件に合わないカードが消える
  const vis = [1, 1 - Math.max(on[1], on[3]), 1 - on[3], 1];
  for (let i = 0; i < 4; i++) {
    const cx = 24 + (i % 2) * 252, cy = 60 + Math.floor(i / 2) * 64, a = clamp(.12 + .88 * vis[i]);
    X.save(); X.globalAlpha = a; strokeRR(cx, cy, 242, 54, 8, C.line2, 1.5); fillRR(cx + 10, cy + 10, 34, 34, 6, i === 0 ? C.mb : C.line); bar(cx + 54, cy + 14, 120, C.line2, 10); bar(cx + 54, cy + 32, 150, C.line, 7); X.restore();
  }
  const p1 = pos[1] || 120, p3 = pos[3] || 300;
  pointer([[.3, w + 30, h + 30], [.9, 130, 32], [1.3, 130, 32], [1.7, 300, 32], [2.5, 300, 32], [2.95, 130, 32], [3.2, 130, 32], [3.55, 296, 32], [3.85, 296, 32], [4.3, w + 30, h + 30]], [0.95, 1.75, 3.0, 3.6], t);
});
