// Breadcrumbs / 面包屑导航
// Pack: ui (UI component names) · element 18 · Category: Navigation & status
// What happens: Clicking "Courses" in the breadcrumbs goes up to the list page; clicking a card there returns to the detail page.
// Motion spec:
//   - 4.5 s cycle on a 542×191 card. At the top "Home › Courses › Web design" (in Japanese, 17 px): links blue, current page black and bold; a divider at y 46.
//   - The cursor hovers "Courses" (underline 0.9–1.2 s) and clicks at 1.0 s.
//   - Up amount: 0→1 over 1.0–1.25 s with easeInOutCubic. "› Web design" fades out and "Courses" becomes the current page (black, bold).
//   - Content cross-fades from a detail page (title and text) to a list of three cards (158×118, with a landscape picture).
//   - Back: the cursor moves onto the first card (blue outline 2.45–2.8 s) and clicks at 2.7 s; the detail page returns over 2.7–2.95 s.
// When to use it: Use it on sites with deep hierarchies to show where the page is and give a way back up.
// Tags: breadcrumbs, breadcrumb, trail, hierarchy, navigation, path
// Size: 542x191 px · Cycle: 4.5 s
// Needs: kit ui -> https://motion-prompt.2077tech.com/api/v1/kit/ui.js
// Live demo: https://motion-prompt.2077tech.com/#066
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 18
// 18 パンくずリスト（Breadcrumbs）
CELLS[17] = ui((w, h, t) => {
  const up = eio(seg(t, 1.0, 1.25)) * (1 - eio(seg(t, 2.7, 2.95))); // 1=「講座」の階層にいる
  let x = 24; const y = 24;
  text('ホーム', x, y, 17, C.blue, 600); x += tw('ホーム', 17, 600) + 10; text('›', x, y, 17, '#aaa', 500); x += 18;
  const kx = x, kw = tw('講座', 17, 700);
  const hoverK = t > .9 && t < 1.2;
  text('講座', x, y, 17, up > .5 ? C.ink : C.blue, up > .5 ? 700 : 600); if (hoverK) { X.fillStyle = C.blue; X.fillRect(x, y + 12, kw, 2); }
  x += kw + 10;
  X.globalAlpha = 1 - up; text('›', x, y, 17, '#aaa', 500); text('Web制作', x + 18, y, 17, C.ink, 700); X.globalAlpha = 1;
  X.fillStyle = C.line; X.fillRect(24, 46, 494, 1.5);
  // 中身：詳細ページ ⇄ 一覧ページ
  if (up < 1) { X.save(); X.globalAlpha = 1 - up; bar(24, 64, 200, C.ink, 13); bar(24, 92, 494); bar(24, 110, 420); fillRR(24, 132, 494, 48, 8, C.block); X.restore(); }
  if (up > 0) { X.save(); X.globalAlpha = up; for (let k = 0; k < 3; k++) { const cx = 24 + k * 168, hov = k === 0 && t > 2.45 && t < 2.8; strokeRR(cx, 62, 158, 118, 8, hov ? C.blue : C.line2, hov ? 2 : 1.5); mountains(cx + 10, 72, 138, 58, k % 2); bar(cx + 10, 142, 90, k === 0 ? C.sb : C.line2, 10); bar(cx + 10, 160, 120, C.line, 7); } X.restore(); }
  pointer([[.3, w + 30, h + 30], [.95, kx + kw / 2, 30], [1.6, kx + kw / 2, 30], [2.4, 104, 120], [3.1, 104, 120], [3.9, w + 30, h + 30]], [1.0, 2.7], t);
});
