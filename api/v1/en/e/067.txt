// Mega menu / 大型下拉菜单
// Pack: ui (UI component names) · element 19 · Category: Navigation & status
// What happens: Hovering over "Products" in the header drops a full-width panel with three columns of links and an image; it closes when the cursor leaves.
// Motion spec:
//   - 4.5 s cycle on a 542×191 card. A 42-tall header with "Products, Pricing, Support" (in Chinese, 16 px, each with a down chevron).
//   - The cursor only hovers: it reaches "Products" at 1.0 s. Open amount a: 0→1 over 1.05–1.3 s with easeOutCubic; "Products" turns blue with an underline.
//   - Panel: white, full width, 146 tall, dropping 12 px into place under the header while fading in with a. Shadow blur 18, offset y 8.
//   - Inside: three columns 130 px apart (a blue heading and three links each) and a 104×112 picture at the right. The cursor moves down and the second link of the second column is highlighted light blue from 1.75 to 2.9 s.
//   - Close: the cursor leaves the panel from 2.9 s; it fades over 3.15–3.4 s (1 − easeInOutCubic).
// When to use it: Use it on sites with many products or sections so visitors can see the whole range at once.
// Tags: mega-menu, megamenu, dropdown-navigation, navigation, header-menu, multi-column
// Size: 542x191 px · Cycle: 4.5 s
// Needs: kit ui -> https://motion-prompt.2077tech.com/api/v1/kit/ui.js
// Live demo: https://motion-prompt.2077tech.com/#067
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 19
// 19 メガメニュー（Mega menu）
CELLS[18] = ui((w, h, t) => {
  const a = eo(seg(t, 1.05, 1.3)) * (1 - eio(seg(t, 3.15, 3.4)));
  pageLines(24, 62, 494); fillRR(24, 130, 494, 50, 8, C.block);
  X.fillStyle = '#fff'; X.fillRect(0, 0, w, 42); X.fillStyle = C.line; X.fillRect(0, 42, w, 1.5);
  fillRR(24, 14, 70, 16, 4, C.mb);
  const nav = [['产品', 190], ['价格', 262], ['支持', 334]];
  nav.forEach(([s, x], i) => { const on = i === 0 && a > .3; text(s, x, 22, 16, on ? C.blue : '#555', on ? 700 : 600); if (on) fillRR(x, 36, tw(s, 16, 700), 3, 1.5, C.blue); chevron(x + tw(s, 16, 700) + 10, 22, 3.5, 0, on ? C.blue : '#999', 1.8); });
  if (a > 0) {
    X.save(); X.globalAlpha = a; X.beginPath(); X.rect(0, 43, w, h); X.clip(); const py = 43 - 12 * (1 - a);
    shadowed(() => { X.fillStyle = '#fff'; X.fillRect(0, py, w, 146); }, 18, 8, .16);
    for (let c = 0; c < 3; c++) { const cx = 24 + c * 130; bar(cx, py + 16, 80, C.blue, 10); for (let k = 0; k < 3; k++) { const hov = c === 1 && k === 1 && t > 1.75 && t < 2.9; if (hov) fillRR(cx - 6, py + 34 + k * 26, 112, 22, 5, C.lb); bar(cx, py + 40 + k * 26, [90, 70, 100][(k + c) % 3], hov ? C.sb : C.line2, 9); } }
    mountains(414, py + 14, 104, 112);
    X.restore();
  }
  // 乗せるだけで開く（クリックしない）
  pointer([[.3, w + 30, h + 30], [1.0, 205, 26], [1.4, 205, 26], [1.8, 196, 98], [2.9, 196, 98], [3.2, 300, 200], [3.9, w + 30, h + 30]], [], t);
});
