// Skeleton screen / 骨架屏
// Pack: ui (UI component names) · element 21 · Category: Navigation & status
// What happens: Clicking reload turns the list into grey blocks of the same shape with a sweeping shine; the content returns when loading ends.
// Motion spec:
//   - 4.5 s cycle on a 542×191 card. A two-row list (a 96×56 picture and three text lines each). The cursor clicks the reload icon at top right at 1.0 s.
//   - Over 1.05–1.2 s (linear) the content switches to grey #e9e7e5 rounded blocks of the same size and position.
//   - Shine: a white band 160 wide (transparent at the edges, opacity 0.75 in the middle) sweeps from x −200 to x 742 every 0.9 s, drawn only over the blocks.
//   - The content returns over 3.0–3.3 s (linear).
// When to use it: Use it for lists or cards that take a moment to load, so the layout is visible while people wait.
// Tags: skeleton, skeleton-screen, placeholder, shimmer, loading, ghost-content
// Size: 542x191 px · Cycle: 4.5 s
// Needs: kit ui -> https://motion-prompt.2077tech.com/api/v1/kit/ui.js
// Live demo: https://motion-prompt.2077tech.com/#069
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 21
// 21 スケルトン（Skeleton screen）
CELLS[20] = ui((w, h, t) => {
  const sk = seg(t, 1.05, 1.2) * (1 - seg(t, 3.0, 3.3));
  ring(w - 40, 26, 14, C.line2, 1.8); X.beginPath(); X.arc(w - 40, 26, 7, -Math.PI * .1, Math.PI * 1.4); X.strokeStyle = '#666'; X.lineWidth = 2.2; X.stroke();
  bar(24, 20, 160, C.line2, 12);
  for (let k = 0; k < 2; k++) {
    const y = 56 + k * 66;
    X.save(); X.globalAlpha = 1 - sk; mountains(24, y, 96, 56, k); bar(136, y + 6, 200, C.ink, 11); bar(136, y + 26, 340); bar(136, y + 42, 260); X.restore();
    if (sk > 0) {
      X.save(); X.globalAlpha = sk; const g = '#e9e7e5';
      fillRR(24, y, 96, 56, 8, g); fillRR(136, y + 4, 200, 14, 7, g); fillRR(136, y + 26, 340, 10, 5, g); fillRR(136, y + 42, 260, 10, 5, g);
      // 光が横に流れる
      const sx = -200 + ((t - 1.1) % 0.9) / 0.9 * (w + 400);
      X.globalCompositeOperation = 'source-atop';
      const gr = X.createLinearGradient(sx - 80, 0, sx + 80, 0); gr.addColorStop(0, 'rgba(255,255,255,0)'); gr.addColorStop(.5, 'rgba(255,255,255,.75)'); gr.addColorStop(1, 'rgba(255,255,255,0)');
      X.fillStyle = gr; X.fillRect(0, y - 2, w, 64);
      X.restore();
    }
  }
  pointer([[.3, w + 30, h + 30], [.95, w - 38, 30], [1.4, w - 38, 30], [2.1, w + 30, h + 30]], [1.0], t);
});
