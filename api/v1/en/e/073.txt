// Parallax scrolling / 视差滚动
// Pack: lp (Landing page & web effects) · element 01 · Category: Scroll effects
// What happens: As the page scrolls down, the headline moves up normally while the sky and sun, far hills and near hills move more slowly.
// Motion spec:
//   - A browser window with a 511 × 168 viewport. Scroll: 0→120 px at 0.3–1.9 s, 120→0 at 2.9–4.3 s (cubic ease-in-out).
//   - Each layer moves by scroll × factor: sky and sun 0.15, far hills 0.4, near hills 0.7.
//   - Headline (28 px), button and the next section move at 1.0 (normal scrolling).
//   - The next section (white) starts at the bottom of the hero (168 px) and slides over the slower background.
//   - The scrollbar thumb on the right moves with the scroll.
// When to use it: Use it to give the hero or a section photo a sense of depth.
// Tags: parallax, parallax-scrolling, scroll, depth, layers, hero, background
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// Live demo: https://motion-prompt.2077tech.com/#073
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 01
// 01 パララックススクロール（スクロールすると、空・山・見出しが違う速さで動く）
CELLS[0] = (k, w, h, u) => {
  frame(k);
  const s = steps([[0.3, 0], [1.9, 1], [2.9, 1], [4.3, 0]], u), y = s * 120;
  view(k, () => {
    const g = k.createLinearGradient(0, 0, 0, 200); g.addColorStop(0, '#9DB6FB'); g.addColorStop(1, '#E6ECFF');
    k.save(); k.translate(0, -y * 0.15);                         // 空と太陽: 0.15倍
    k.fillStyle = g; k.fillRect(0, 0, VW, 260); k.fillStyle = C.acc; k.beginPath(); k.arc(392, 54, 25, 0, 7); k.fill(); k.restore();
    k.save(); k.translate(0, -y * 0.4); mount(k, VW, 190, M1, '#7D9AF4', 0); k.restore();   // 遠い山: 0.4倍
    k.save(); k.translate(0, -y * 0.7); mount(k, VW, 205, M2, '#4B6BDB', 1); k.restore();   // 近い山: 0.7倍
    k.save(); k.translate(0, -y);                                // 見出し・ボタン・次のセクション: 1倍
    txt(k, '山へ、行こう。', 34, 64, `800 28px ${JP}`, '#fff', 'left', 'alphabetic', '1px');
    bars(k, 34, 78, 160, 2, 'rgba(255,255,255,.85)');
    btn(k, 34, 106, 92, 26, '予約する');
    k.fillStyle = '#fff'; k.fillRect(0, VH, VW, 200);
    txt(k, '人気のコース', 34, VH + 34, `800 16px ${JP}`, C.ink);
    for (let i = 0; i < 3; i++) { pic(k, 34 + i * 152, VH + 50, 136, 58, i, 6); bars(k, 34 + i * 152, VH + 116, 110, 1, C.faint); }
    k.restore();
    sbar(k, s * 0.35, 0.4);
  });
};
