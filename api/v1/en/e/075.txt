// Sticky header / 吸顶导航
// Pack: lp (Landing page & web effects) · element 03 · Category: Scroll effects
// What happens: The announcement strip scrolls away, but the header below it stops at the top edge and the content scrolls underneath.
// Motion spec:
//   - Scroll: 0→110 px at 0.4–1.3 s, 110→250 px at 1.9–2.8 s, 250→0 at 3.4–4.4 s (cubic ease-in-out).
//   - At the very top, a 22 px announcement strip (navy); below it a 34 px header (logo, three menu items, button).
//   - Header position = max(22 − scroll, 0): once the strip is gone, it stops at the top edge (position: sticky; top: 0).
//   - Over the next 24 px of scroll, a 12 px shadow under the header fades from 0 to 0.16 opacity.
//   - The content (photo, heading, three cards) scrolls normally under the header.
// When to use it: Use it to keep the menu and the call-to-action within reach on a long page.
// Tags: sticky, sticky-header, fixed-header, position-sticky, navigation, header, scroll
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// Live demo: https://motion-prompt.2077tech.com/#075
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 03
// 03 追従ヘッダー（スクロールすると、ヘッダーが画面の上で止まってついてくる）
CELLS[2] = (k, w, h, u) => {
  frame(k);
  const y = steps([[0.4, 0], [1.3, 110], [1.9, 110], [2.8, 250], [3.4, 250], [4.4, 0]], u);
  const STRIP = 22;
  view(k, () => {
    k.save(); k.translate(0, -y);
    pic(k, 24, 70, 214, 112, 0, 8);
    txt(k, '开启全新一天，', 262, 104, `800 22px ${JP}`, C.ink); bars(k, 262, 120, 210, 3); btn(k, 262, 160, 100, 26, '查看详情');
    k.fillStyle = C.soft; k.fillRect(0, 212, VW, 240);
    txt(k, '特色', VW / 2, 246, `800 18px ${JP}`, C.ink, 'center');
    for (let i = 0; i < 3; i++) { const x = 34 + i * 152; k.fillStyle = '#fff'; rr(k, x, 262, 136, 120, 8); k.fill(); pic(k, x + 8, 270, 120, 58, i, 5); bars(k, x + 10, 340, 110, 3, C.faint, 6, 11); }
    k.fillStyle = C.ink; k.fillRect(0, 0, VW, STRIP);
    txt(k, '限时优惠进行中', VW / 2, STRIP / 2 + 0.5, `600 11px ${JP}`, '#fff', 'center', 'middle');
    k.restore();
    const hy = Math.max(STRIP - y, 0), sh = clamp((y - STRIP) / 24);
    if (sh > 0) { const g = k.createLinearGradient(0, hy + 34, 0, hy + 46); g.addColorStop(0, `rgba(28,35,51,${0.16 * sh})`); g.addColorStop(1, 'rgba(28,35,51,0)'); k.fillStyle = g; k.fillRect(0, hy + 34, VW, 12); }
    header(k, hy);
    sbar(k, y / 250, 0.42);
  });
};
