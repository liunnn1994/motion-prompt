// Scroll snap / 滚动吸附
// Pack: lp (Landing page & web effects) · element 06 · Category: Scroll effects
// What happens: After a small scroll, the page carries itself the rest of the way until the next section's top lines up with the top of the view.
// Motion spec:
//   - Each section is as tall as the viewport (168 px). The first is pale blue, the second navy.
//   - One snap per 2.4 s. At 0.45–0.85 s the user scrolls only 34 px (cubic ease-out).
//   - At 0.9–1.5 s the page moves the remaining 134 px by itself and lines up the next section's top (cubic ease-out).
//   - The second 2.4 s does the same upward, back to the first section (the behaviour of scroll-snap-type: y mandatory).
// When to use it: Use it for a page that should read one full screen at a time, like slides.
// Tags: scroll-snap, snap, full-page, sections, css, mandatory, scroll
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// Live demo: https://motion-prompt.2077tech.com/#078
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 06
// 06 スクロールスナップ（少しスクロールすると、次のセクションの頭までピタッと送られる）
CELLS[5] = (k, w, h, u) => {
  frame(k);
  const T = CYC / 2, lu = mod(u, T), dir = u < T ? 1 : -1, from = u < T ? 0 : VH, DRIFT = 34;
  const y = from + dir * (DRIFT * eOut(P(lu, 0.45, 0.85)) + (VH - DRIFT) * eOut(P(lu, 0.9, 1.5)));
  view(k, () => {
    k.save(); k.translate(0, -y);
    for (let i = 0; i < 2; i++) {
      const y0 = i * VH, dark = i === 1;
      k.fillStyle = dark ? C.ink : C.blueL; k.fillRect(0, y0, VW, VH);
      txt(k, '0' + (i + 1), 34, y0 + 70, `800 46px ${EN}`, C.acc);
      txt(k, dark ? '使ってみる' : '知ってもらう', 34, y0 + 102, `800 18px ${JP}`, dark ? '#fff' : C.ink);
      bars(k, 34, y0 + 118, 170, 2, dark ? 'rgba(255,255,255,.3)' : '#C3CDEA');
      pic(k, 262, y0 + 24, 220, 120, dark ? 2 : 0, 8);
    }
    k.restore();
    sbar(k, y / VH, 0.5);
  });
};
