// Scroll progress bar / 滚动进度条
// Pack: lp (Landing page & web effects) · element 07 · Category: Scroll effects
// What happens: As the article is read a bit at a time, the line along the top grows from the left and reaches the far end at the bottom of the page.
// Motion spec:
//   - The article scrolls 0→430 px in four 110 px steps, each moving for 0.45 s then pausing 0.4 s (cubic ease-in-out).
//   - It returns 430→0 at 3.75–4.55 s.
//   - A 6 px bar along the top of the viewport: light grey track, orange fill.
//   - Fill width = 511 px × scroll / 430, the scroll fraction itself (the same as CSS animation-timeline: scroll()).
// When to use it: Use it on long articles to show how much is left to read.
// Tags: progress-bar, reading-progress, scroll-progress, indicator, scroll-driven, article
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// Live demo: https://motion-prompt.2077tech.com/#079
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 07
// 07 スクロールプログレスバー（読み進めた分だけ、画面上の線が伸びる）
CELLS[6] = (k, w, h, u) => {
  frame(k);
  const MAX = 430;
  const y = steps([[0.3, 0], [0.75, 110], [1.15, 110], [1.6, 220], [2.0, 220], [2.45, 330], [2.85, 330], [3.3, MAX], [3.75, MAX], [4.55, 0]], u);
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, 'はじめてのWebデザイン', 70, 44, `800 20px ${JP}`, C.ink); bars(k, 70, 56, 120, 1, C.faint, 5);
    pic(k, 70, 74, 371, 112, 0, 8);
    for (let b = 0; b < 4; b++) { const y0 = 204 + b * 82; k.fillStyle = C.ink; rr(k, 70, y0, 140, 9, 4.5); k.fill(); bars(k, 70, y0 + 22, 371, 4, C.faint, 6, 12); }
    k.restore();
    k.fillStyle = C.faint; k.fillRect(0, 0, VW, 6);
    k.fillStyle = C.acc; k.fillRect(0, 0, VW * (y / MAX), 6);
    sbar(k, y / MAX, 0.3);
  });
};
