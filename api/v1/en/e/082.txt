// Image zoom on hover / 悬停图片缩放
// Pack: lp (Landing page & web effects) · element 10 · Category: Hover & click interactions
// What happens: The photo under the cursor slowly enlarges while its frame stays the same size, and shrinks back when the cursor leaves.
// Motion spec:
//   - Two photo frames, 222 × 118 (radius 8). The cursor rests on the left one, then the right one, for about 1.5 s each.
//   - On hover the content scales 1→1.15 from the centre over 0.5 s (cubic ease-out); on leave it returns over 0.5 s.
//   - Anything outside the frame is clipped (overflow: hidden). The frame and the caption do not move.
// When to use it: Use it on photo grids (work, products, articles) to show which item is about to be chosen.
// Tags: hover, zoom, image, scale, overflow-hidden, thumbnail, gallery
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// Live demo: https://motion-prompt.2077tech.com/#082
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 10
// 10 ホバーで画像ズーム（枠はそのままで、中の写真だけがゆっくり拡大する）
CELLS[9] = (k, w, h, u) => {
  frame(k);
  const FX = [22, 267], FY = 16, FW = 222, FH = 118, ZOOM = 0.15;
  const cur = t => path([[0, 60, 160], [0.5, 120, 70], [1.8, 140, 84], [2.2, 370, 70], [3.4, 390, 84], [3.9, 470, 160], [4.8, 60, 160]], t);
  view(k, () => {
    for (let i = 0; i < 2; i++) {
      const inside = t => { const [x, y] = cur(t); return x > FX[i] && x < FX[i] + FW && y > FY && y < FY + FH; };
      const q = trans(inside, u, 0.5);
      pic(k, FX[i], FY, FW, FH, i, 8, 1 + ZOOM * q);
      txt(k, ['夏の山小屋', '海辺のカフェ'][i], FX[i], FY + FH + 20, `800 13px ${JP}`, C.ink); bars(k, FX[i], FY + FH + 28, 150, 1);
    }
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
