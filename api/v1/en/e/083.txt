// Cursor follower / 光标跟随
// Pack: lp (Landing page & web effects) · element 11 · Category: Hover & click interactions
// What happens: As the cursor moves around the page, an orange circle follows it a little behind.
// Motion spec:
//   - Cursor path: x = 255 + 175·sin(2πt/4.8), y = 86 + 46·sin(4πt/4.8) (a figure eight lying on its side, one lap per 4.8 s).
//   - Circle: radius 17, orange, opacity 0.88, drawn under the cursor.
//   - Follow rule: every frame (60 fps) it covers 12% of the remaining distance (x += (cursor x − x) × 0.12).
//   - The arrow cursor stays visible.
// When to use it: Use it on brand sites and portfolios to add a playful, hands-on feel.
// Tags: cursor, custom-cursor, cursor-follower, mouse-follow, lerp, trail, interaction
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// Live demo: https://motion-prompt.2077tech.com/#083
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 11
// 11 マウスストーカー（カーソルのあとを、丸が少し遅れて追いかける）
CELLS[10] = (k, w, h, u) => {
  frame(k);
  const cur = t => [255 + 175 * Math.sin(2 * Math.PI * t / CYC), 86 + 46 * Math.sin(4 * Math.PI * t / CYC)];
  // 毎フレーム（60fps）残りの距離の 12% ずつ近づく
  let t = u - 2, [fx, fy] = cur(t);
  for (let i = 0; i < 120; i++) { t += 1 / 60; const [tx, ty] = cur(t); fx += (tx - fx) * 0.12; fy += (ty - fy) * 0.12; }
  view(k, () => {
    txt(k, 'CREATIVE STUDIO', VW / 2, 74, `800 34px ${EN}`, C.ink, 'center', 'alphabetic', '3px');
    bars(k, VW / 2 - 120, 96, 240, 2, C.faint, 6, 12, 0.7);
    btn(k, VW / 2 - 50, 128, 100, 24, '查看作品', C.ink);
    k.fillStyle = C.acc; k.globalAlpha = 0.88; k.beginPath(); k.arc(fx, fy, 17, 0, 7); k.fill(); k.globalAlpha = 1;
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
