// Exit-intent popup / 离站提示弹窗
// Pack: lp (Landing page & web effects) · element 16 · Category: Hover & click interactions
// What happens: When the cursor heads out through the top of the page (toward the tabs and close button), the page dims and a "Wait!" popup opens. The × closes it.
// Motion spec:
//   - The cursor rises from the page toward the browser's address bar at 0.7–1.25 s (cubic ease-in-out).
//   - It fires when the cursor crosses the top edge of the viewport (y = 0, at about 1.07 s), the same test as mouseleave on document with clientY ≤ 0.
//   - Opens 0.05–0.45 s after firing (cubic ease-out): backdrop navy 0→0.5 opacity; popup (260 × 132, radius 12) opacity 0→1 and scale 0.9→1.
//   - Inside: a heading, a discount line (orange), a coupon button and an × at the top right.
//   - Click on the × at 2.9 s; it disappears 0.03–0.25 s later (linear).
// When to use it: Use it to make one last offer to visitors who are about to leave without converting.
// Tags: exit-intent, popup, mouseleave, coupon, conversion, retention, lp
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// Live demo: https://motion-prompt.2077tech.com/#088
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 16
// 16 離脱防止ポップアップ（カーソルがタブや閉じるボタンの方へ出ていくと、引き止める小窓が出る）
CELLS[15] = (k, w, h, u) => {
  frame(k);
  const OX = WIN.x, OY = WIN.y + WIN.bar;               // カーソルは窓の外（タブ・アドレスバー）まで出るので、カードの座標で描く
  const [cx, cy] = path([[0, 440, 150], [0.7, 300, 120], [1.25, 360, -18], [1.9, 372, -20], [2.6, 366, 38], [3.3, 366, 38], [4.4, 440, 150], [4.8, 440, 150]], u);
  const CLOSE = 2.9;
  let TRIG = 0.7; while (TRIG < 1.3 && path([[0.7, 300, 120], [1.25, 360, -18]], TRIG)[1] >= 0) TRIG += 1 / 240;   // 表示域の上端（y = 0）を越えた時刻（約1.07秒）
  const m = eOut(P(u, TRIG + 0.05, TRIG + 0.45)) * (1 - P(u, CLOSE + 0.03, CLOSE + 0.25));
  const D = { x: 125.5, y: 18, w: 260, h: 132 };
  view(k, () => {
    header(k, 0);
    pic(k, 24, 46, 222, 110, 0, 8); txt(k, 'はじめての方へ', 264, 82, `800 19px ${JP}`, C.ink); bars(k, 264, 96, 210, 3); btn(k, 264, 132, 100, 24, '申し込む');
    if (m > 0) {
      k.fillStyle = `rgba(28,35,51,${0.5 * m})`; k.fillRect(0, 0, VW, VH);
      k.save(); k.globalAlpha = m; k.translate(D.x + D.w / 2, D.y + D.h / 2); const s = 0.9 + 0.1 * m; k.scale(s, s); k.translate(-D.w / 2, -D.h / 2);
      k.fillStyle = '#fff'; rr(k, 0, 0, D.w, D.h, 12); k.fill();
      txt(k, 'ちょっと待って！', D.w / 2, 34, `800 17px ${JP}`, C.ink, 'center');
      txt(k, '今なら初回 10%OFF', D.w / 2, 60, `800 15px ${JP}`, C.acc, 'center');
      btn(k, D.w / 2 - 75, 80, 150, 30, 'クーポンを受け取る', C.acc, '#fff', 12);
      k.strokeStyle = C.sub; k.lineWidth = 2; k.lineCap = 'round'; k.beginPath(); k.moveTo(D.w - 26, 14); k.lineTo(D.w - 14, 26); k.moveTo(D.w - 14, 14); k.lineTo(D.w - 26, 26); k.stroke();
      k.restore();
    }
  });
  // カーソルの × のクリック（窓の中）と、窓の外への移動
  k.save(); k.beginPath(); k.rect(0, 0, CW, CH); k.clip(); k.translate(OX, OY); click(k, cx, cy, u, [CLOSE]); k.restore();
};
