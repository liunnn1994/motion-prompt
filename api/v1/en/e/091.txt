// Infinite scroll / 无限滚动
// Pack: lp (Landing page & web effects) · element 19 · Category: Loading, navigation & ambient loops
// What happens: Scrolling down to the end of the list brings up a spinner, then three new items are added so reading can go on.
// Motion spec:
//   - A list of items (331 × 44, radius 8, 52 apart) with a spinner (radius 10, one turn per 0.8 s) under the last item.
//   - Scroll: 38→116 px at 0.25–0.85 s, 116→194 px at 1.15–1.75 s (cubic ease-in-out); it stops with the end of the list in view.
//   - At 2.6 s three items are added: each fades 0→1 and rises from 10 px below over 0.35 s (cubic ease-out). The spinner moves to the new end.
//   - Each cycle advances by three items (156 px). The content repeats every three items, so the end of the cycle looks the same as the start.
// When to use it: Use it for feeds and news lists that people should keep reading without paging.
// Tags: infinite-scroll, endless-scroll, load-more, feed, pagination, spinner, list
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// Live demo: https://motion-prompt.2077tech.com/#091
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 19
// 19 無限スクロール（いちばん下まで来ると読み込み表示が出て、続きが足される）
CELLS[18] = (k, w, h, u) => {
  frame(k);
  // 1周期で 3件（156px）ぶん進む。周期の終わりは始まりと同じ見た目（内容は3件ごとにくり返す）
  const PITCH = 52, BATCH = 3, L0 = 6, TL = 2.6;
  const y = steps([[0.25, 38], [0.85, 116], [1.15, 116], [1.75, 194], [4.8, 194]], u);
  const n = u < TL ? L0 : L0 + BATCH, end = 10 + n * PITCH;
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    k.save(); k.translate(0, -y);
    for (let i = 0; i < n; i++) {
      const top = 10 + i * PITCH, v = i % 3, fresh = i >= L0 ? eOut(P(u, TL, TL + 0.35)) : 1;
      if (top - y > VH || top + 44 - y < 0) continue;
      k.save(); k.globalAlpha = fresh; k.translate(0, 10 * (1 - fresh));
      k.fillStyle = '#fff'; rr(k, 90, top, 331, 44, 8); k.fill(); pic(k, 98, top + 6, 48, 32, v, 5);
      bars(k, 158, top + 12, [180, 140, 200][v], 1, C.ink, 7); bars(k, 158, top + 26, [120, 150, 100][v], 1, C.faint, 6);
      k.restore();
    }
    spinner(k, VW / 2, end + 18, 10, u);
    k.restore();
  });
};
