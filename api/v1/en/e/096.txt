// Floating animation / 悬浮动画
// Pack: lp (Landing page & web effects) · element 24 · Category: Loading, navigation & ambient loops
// What happens: Next to the headline, a phone, a heart badge and a rating chip each bob gently out of step, and the shadow under the phone grows and shrinks.
// Motion spec:
//   - Phone (92 × 132): y = −8·sin(2πt/2.4), one bob per 2.4 s; a sine slows at both ends.
//   - Heart badge (radius 20): y = −6·sin(2πt/1.6 + 1.2). Rating chip (76 × 30): y = −6·sin(2πt/2.4 + 2.4).
//   - Shadow under the phone (54 × 6 ellipse): size and opacity scale by (1 + y/40), so it shrinks and fades as the phone rises.
//   - Both periods (2.4 s and 1.6 s) divide 4.8 s, so the loop has no seam.
// When to use it: Use it to add quiet motion to product images, app mockups or decorative icons.
// Tags: floating, float, bob, hover-animation, keyframes, translatey, decoration, loop
// Size: 539x208 px · Cycle: 4.8 s
// Needs: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// Live demo: https://motion-prompt.2077tech.com/#096
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 24
// 24 ふわふわアニメーション（画面の飾りが、ゆっくり上下に浮き沈みし続ける）
CELLS[23] = (k, w, h, u) => {
  frame(k);
  const bob = (T, A, ph) => -A * Math.sin(2 * Math.PI * u / T + ph);   // 上下（easeInOut に近いサイン波）
  const y1 = bob(2.4, 8, 0), y2 = bob(1.6, 6, 1.2), y3 = bob(2.4, 6, 2.4);
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    txt(k, '用上应用，', 34, 62, `800 24px ${JP}`, C.ink); txt(k, '生活更轻松。', 34, 94, `800 24px ${JP}`, C.ink);
    bars(k, 34, 108, 170, 2); btn(k, 34, 134, 112, 24, '立即下载');
    // 影（上がるほど小さく薄く）
    const sh = 1 + y1 / 40;
    k.fillStyle = `rgba(28,35,51,${0.16 * sh})`; k.beginPath(); k.ellipse(352, 160, 54 * sh, 6 * sh, 0, 0, 7); k.fill();
    // スマホ
    k.save(); k.translate(0, y1);
    k.fillStyle = C.ink; rr(k, 306, 18, 92, 132, 16); k.fill(); k.fillStyle = '#fff'; rr(k, 312, 24, 80, 120, 11); k.fill();
    pic(k, 318, 32, 68, 52, 0, 6); bars(k, 318, 92, 68, 3, C.faint, 5, 10);
    k.restore();
    // 飾り（ハートの丸と星のチップ）
    k.save(); k.translate(0, y2); k.shadowColor = 'rgba(28,35,51,.18)'; k.shadowBlur = 10; k.shadowOffsetY = 4;
    k.fillStyle = C.acc; k.beginPath(); k.arc(268, 54, 20, 0, 7); k.fill(); k.shadowColor = 'transparent';
    k.fillStyle = '#fff'; k.beginPath(); k.moveTo(268, 62); k.bezierCurveTo(254, 53, 260, 41, 268, 48); k.bezierCurveTo(276, 41, 282, 53, 268, 62); k.fill(); k.restore();
    k.save(); k.translate(0, y3); k.shadowColor = 'rgba(28,35,51,.16)'; k.shadowBlur = 10; k.shadowOffsetY = 4;
    k.fillStyle = '#fff'; rr(k, 410, 92, 76, 30, 15); k.fill(); k.shadowColor = 'transparent';
    txt(k, '★ 4.8', 448, 108, `800 13px ${EN}`, C.acc, 'center', 'middle'); k.restore();
  });
};
