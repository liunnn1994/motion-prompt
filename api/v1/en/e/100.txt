// Split lighting / 分割光
// Pack: photo (Photo looks, lighting & styles) · element 04 · Category: Lighting & time of day
// What happens: The light swings from near the front (20°) to directly at the side (90°); the shadow on the head grows from a crescent to exactly half, splitting the face down the middle.
// Motion spec:
//   - Scene 1: a head-and-shoulders figure in front of a horizon at y 150 (head centre x 270, y 78, radius 29 px, no face or hair). the background drops to a darker scene (sky #E7E5E1)
//   - Light: from the left of the frame, angle 20° → 90° (plan at the bottom right, top-right readout "90° · ↑0°")
//   - Head: right of x = 270 + cos(angle)·√(29² − dy²) is shadow #33405A, left is light #FFC75A; at 90° the edge is the straight centre line
//   - Body (half-width 56 px): right of x = 270 + cos(angle)·56 is shadow, a strip on the left edge is lit
//   - Timing: the effect comes in from 0.4 s to 1.4 s, holds until 4.0 s and is gone again by 4.8 s (one pass = 5 s, cubic ease-in-out (about cubic-bezier(0.65,0,0.35,1)))
// When to use it: For dramatic, tense or moody portraits.
// Tags: split lighting, side light, portrait lighting, half face, dramatic, chiaroscuro
// Size: 539x208 px · Cycle: 5 s
// Needs: kit photo -> https://motion-prompt.2077tech.com/api/v1/kit/photo.js
// Live demo: https://motion-prompt.2077tech.com/#100
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 04
// スプリットライティング: 光が真横（90°）へ回り、顔の右半分が影に沈む
CELLS[3] = cell((k, u) => {
  const a = env(u), az = lerp(0.35, Math.PI / 2, a);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a), W = torsoW(PR);
  bust(k, PX, PY, PR, BASE.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, PR, 0, TAU); kk.fill();
    headShade(kk, PX, PY, PR, az, 1, sh);
    torso(kk, PX, PY, PR, PX - W, PX - Math.cos(az) * 0.4 * W - 0.4 * W, lit);
    torso(kk, PX, PY, PR, PX + Math.cos(az) * W, PX + W, sh);
  });
  plan(k, [[az, a]]);
  hud(k, `${Math.round(az * 180 / Math.PI)}° · ↑0°`);
});
