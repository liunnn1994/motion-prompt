// Rembrandt lighting / 伦勃朗光
// Pack: photo (Photo looks, lighting & styles) · element 05 · Category: Lighting & time of day
// What happens: A light 45° to the side and 45° up comes on; the right side of the face falls into shadow except for a small inverted triangle of light on the shadow-side cheek.
// Motion spec:
//   - Scene 1: a head-and-shoulders figure in front of a horizon at y 150 (head centre x 270, y 78, radius 29 px, no face or hair). the background drops to a darker scene (sky #E7E5E1)
//   - Head: right of x = 270 + 0.04·29 is shadow #33405A, left is light #FFC75A
//   - Light triangle: (x 270+0.24R, y 78−0.12R), (x 270+0.7R, y 78−0.12R), (x 270+0.24R, y 78+0.5R) with R = 29 (about 13 px wide, 18 px tall), separated from the lit side by a shadow strip about 6 px wide
//   - Body: lit left of centre −0.45×56 px, shadow right of +0.3×56 px; plan shows the light at 45°, top-right readout "45° · ↑45°"
//   - Timing: the effect comes in from 0.4 s to 1.4 s, holds until 4.0 s and is gone again by 4.8 s (one pass = 5 s, cubic ease-in-out (about cubic-bezier(0.65,0,0.35,1)))
// When to use it: For portraits with a painterly, sculpted, classic look.
// Tags: rembrandt lighting, portrait lighting, triangle, 45 degree light, chiaroscuro, painterly
// Size: 539x208 px · Cycle: 5 s
// Needs: kit photo -> https://motion-prompt.2077tech.com/api/v1/kit/photo.js
// Live demo: https://motion-prompt.2077tech.com/#101
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 05
// レンブラントライティング: 斜め45°・上からの光。影側の頬に三角形の光が残る
CELLS[4] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a);
  bust(k, PX, PY, R, BASE.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
    kk.fillStyle = sh; kk.fillRect(PX + 0.04 * R, PY - R - 2, R * 2, R * 2 + 4);       // 鼻筋から右は影
    kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.clip();
    kk.fillStyle = lit; kk.beginPath();                                                   // 影側の頬に残る光の三角形
    kk.moveTo(PX + 0.24 * R, PY - 0.12 * R); kk.lineTo(PX + 0.7 * R, PY - 0.12 * R); kk.lineTo(PX + 0.24 * R, PY + 0.5 * R); kk.closePath(); kk.fill();
  });
  bust(k, PX, PY, R, 'rgba(0,0,0,0)', kk => {
    kk.fillStyle = 'rgba(0,0,0,0)';
    torso(kk, PX, PY, R, PX - W, PX - 0.45 * W, lit);
    torso(kk, PX, PY, R, PX + 0.3 * W, PX + W, sh);
  });
  plan(k, [[Math.PI / 4, a]]);
  hud(k, '45° · ↑45°');
});
