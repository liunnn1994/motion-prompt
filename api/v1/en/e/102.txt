// Butterfly lighting / 蝴蝶光
// Pack: photo (Photo looks, lighting & styles) · element 06 · Category: Lighting & time of day
// What happens: A light in front of the face and high above comes on; the face lights evenly on both sides and only a small symmetrical shadow falls under the chin.
// Motion spec:
//   - Scene 1: a head-and-shoulders figure in front of a horizon at y 150 (head centre x 270, y 78, radius 29 px, no face or hair). the background drops to a darker scene (sky #E7E5E1)
//   - Head: entirely lit #FFC75A (no shadow side)
//   - Shadow under the chin: an ellipse centred at y 78+1.06R, 0.72R wide and 0.38R tall (R = 29, about 21×11 px), colour #33405A
//   - Body: lit within ±0.8×56 px of the centre, thin outer strips keep the base colour; plan shows the light on the camera side (0°), top-right readout "0° · ↑45°"
//   - Timing: the effect comes in from 0.4 s to 1.4 s, holds until 4.0 s and is gone again by 4.8 s (one pass = 5 s, cubic ease-in-out (about cubic-bezier(0.65,0,0.35,1)))
// When to use it: For beauty and fashion portraits that flatter the face.
// Tags: butterfly lighting, paramount lighting, beauty lighting, portrait lighting, overhead light, glamour
// Size: 539x208 px · Cycle: 5 s
// Needs: kit photo -> https://motion-prompt.2077tech.com/api/v1/kit/photo.js
// Live demo: https://motion-prompt.2077tech.com/#102
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 06
// バタフライライティング: 顔の正面・高い位置からの光。顔は左右対称に明るく、あごの下にだけ小さな影
CELLS[5] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a);
  bust(k, PX, PY, R, BASE.person, kk => {
    torso(kk, PX, PY, R, PX - 0.8 * W, PX + 0.8 * W, lit);
    kk.fillStyle = sh; kk.beginPath(); kk.ellipse(PX, PY + R * 1.06, R * 0.72, R * 0.38, 0, 0, TAU); kk.fill();    // あごの下の影（左右対称）
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
  });
  plan(k, [[0, a]]);
  hud(k, '0° · ↑45°');
});
