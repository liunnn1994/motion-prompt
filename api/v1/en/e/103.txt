// Golden hour / 黄金时刻
// Pack: photo (Photo looks, lighting & styles) · element 07 · Category: Lighting & time of day
// What happens: The sun drops to just above the horizon; sky and ground turn orange, the left side of the person lights up and the shadows of the lamp and tree stretch far to the right.
// Motion spec:
//   - Scene 1: a head-and-shoulders figure in front of a horizon at y 150 (head centre x 270, y 78, radius 29 px, no face or hair). 
//   - Sun: radius 18 px, from (x 110, y −30) to (x 96, y 136), half hidden by the hills
//   - Colour: sky #FAF9F7 → #FFE8BC at the top and #FFCF80 near the horizon, ground #F1F0EE → #F6DDB0; light on the figure #FFB547, shadow #3E3650
//   - Head: light from the left at about 66°, shadow right of the edge; body lit on the left, shadow on the right
//   - Shadows: from the foot of the lamp and tree to the right, length 0 → 150 px (#DDAA6E); top-right time 12:00 → 17:40
//   - Timing: the effect comes in from 0.4 s to 1.4 s, holds until 4.0 s and is gone again by 4.8 s (one pass = 5 s, cubic ease-in-out (about cubic-bezier(0.65,0,0.35,1)))
// When to use it: For warm, nostalgic, gentle-feeling images.
// Tags: golden hour, magic hour, sunset light, warm light, long shadows, low sun
// Size: 539x208 px · Cycle: 5 s
// Needs: kit photo -> https://motion-prompt.2077tech.com/api/v1/kit/photo.js
// Live demo: https://motion-prompt.2077tech.com/#103
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 07
// ゴールデンアワー: 太陽が地平線近くまで下り、全体が橙に染まって影が長く伸びる
CELLS[6] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  const GOLD = {
    sky: '#FFE8BC', sky2: '#FFCF80', far: '#F2C58C', mid: '#E8B27A', ground: '#F6DDB0', ground2: '#ECC892', mark: '#F6DDB0', cast: '#DDAA6E',
    bld: '#F7DCAE', edge: '#D9AE76', win: '#ECC892', winOff: '#ECC892', tree: '#E9C08A', trunk: '#C9965E', post: '#B98A58', bulb: '#C9965E',
    person: '#5C5470', shade: '#3E3650', lit: '#FFB547', car: '#A08E8E', glass: '#F6DDB0', tire: '#3E3650', sun: '#F59A24', glow: '#FFD48A'
  };
  const p = pal(BASE, GOLD, a);
  stagePortrait(k, p, { sun: [lerp(110, 96, a), lerp(-30, 136, a), 18, 1], cast: lerp(0, 150, a) });
  const lit = mixc(p.person, GOLD.lit, a), sh = mixc(p.person, GOLD.shade, a);
  bust(k, PX, PY, R, p.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
    headShade(kk, PX, PY, R, 1.15, 1, sh);
    torso(kk, PX, PY, R, PX - W, PX - 0.5 * W, lit);
    torso(kk, PX, PY, R, PX + 0.4 * W, PX + W, sh);
  });
  hud(k, clock(lerp(720, 1060, a)));
});
