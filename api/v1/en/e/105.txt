// Bokeh / 散景
// Pack: photo (Photo looks, lighting & styles) · element 09 · Category: Lens & camera effects
// What happens: As the aperture opens, the in-focus person stays sharp, the buildings behind soften and the street lamps and distant lights spread into large, pale discs.
// Motion spec:
//   - Scene 2: a street corner; buildings, trees and street lamps (x 114, 292, 494) behind a road at y 132–162; the person stands front left (feet at x 176, y 200, 124 px tall). the background drops to a darker scene (sky #E7E5E1)
//   - Background blur (everything but the person): 0 → 6 px
//   - Lights: the three lamps (size 1.25) and 12 more spaced 45 px apart (y 44–114, size 0.6–1.2, fixed seed); disc radius = (2.4 + 15 × effect) × size
//   - Discs: opacity 100% → 55%, lamps amber #FFC75A, distant lights #FFE3A3, with a 1.2 px amber rim
//   - The person stays sharp; top-right aperture f/16 → f/1.4 (8 stops)
//   - Timing: the effect comes in from 0.4 s to 1.4 s, holds until 4.0 s and is gone again by 4.8 s (one pass = 5 s, cubic ease-in-out (about cubic-bezier(0.65,0,0.35,1)))
// When to use it: To lift a person or product off the background and add a soft, festive night mood.
// Tags: bokeh, shallow depth of field, background blur, wide aperture, f/1.4, out of focus, lights
// Size: 539x208 px · Cycle: 5 s
// Needs: kit photo -> https://motion-prompt.2077tech.com/api/v1/kit/photo.js
// Live demo: https://motion-prompt.2077tech.com/#105
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 09
// ボケ: 絞りを開けると、人物はくっきりのまま、背景の灯りが大きな丸になってにじむ
CELLS[8] = cell((k, u) => {
  const a = env(u), p = pal(BASE, DIM, a);
  const [B, kb] = layer('bokeh'); stageStreet(kb, p, { person: false });
  const [Bb, kbb] = layer('bokeh-b'); kbb.filter = `blur(${(6 * a).toFixed(2)}px)`; kbb.drawImage(B, 0, 0); kbb.filter = 'none';
  k.drawImage(B, 0, 0); k.drawImage(Bb, 0, 0);
  // 灯り: 街灯3つと、窓や遠くの灯り12個（横に45pxおき・高さと大きさは固定シード）。絞りを開くほど丸が大きく薄くなる
  const pts = once('bokeh-pts', () => {
    const o = LAMPS.map(x => [x + 9, ROAD - 53, 1.25, 0]), r = rng(9);
    for (let i = 0; i < 12; i++) o.push([22 + i * 45 + r() * 20, 44 + r() * 70, 0.6 + r() * 0.6, 1]);
    return o;
  });
  k.save();
  for (const [x, y, s, w] of pts) {
    const r = (2.4 + 15 * a) * s;
    k.globalAlpha = lerp(1, 0.55, a); k.fillStyle = w ? BASE.glow : BASE.lit; k.beginPath(); k.arc(x, y, r, 0, TAU); k.fill();
    k.globalAlpha = 0.7 * a; k.strokeStyle = BASE.lit; k.lineWidth = 1.2; k.stroke();
  }
  k.restore();
  figure(pen(k), FIG[0], FIG[1], FIG[2], p.person);
  const stops = ['f/16', 'f/11', 'f/8', 'f/5.6', 'f/4', 'f/2.8', 'f/2', 'f/1.4'];
  hud(k, stops[Math.round(a * 7)]);
});
