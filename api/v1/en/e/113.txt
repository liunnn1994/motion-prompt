// High-key / 高调摄影
// Pack: photo (Photo looks, lighting & styles) · element 17 · Category: Tones & finishing looks
// What happens: The landscape and the person turn to light, near-white colours with very faint shadows, and the hump of the histogram at the bottom right moves to the right end.
// Motion spec:
//   - Scene 3: a bust on the left (head centre x 140, y 82, radius 28 px) and hills, a house, trees and the sun (x 432, y 54, radius 15 px) on the right. 
//   - Colour: sky #FAF9F7 → #FFFFFF, hills #EAEDF1 → #F5F6F8, figure #4D5A75 → #B9C0CE, shadow #33405A → #A6AEBF, sun #F5A524 → #FFD889
//   - Histogram (96×46 px, 20 bars): peak centre 0.56 → 0.85, width 0.2 → 0.09
//   - Timing: the effect comes in from 0.4 s to 1.4 s, holds until 4.0 s and is gone again by 4.8 s (one pass = 5 s, cubic ease-in-out (about cubic-bezier(0.65,0,0.35,1)))
// When to use it: For clean, light, gentle product shots and portraits.
// Tags: high-key, high key, bright, white background, airy, low contrast
// Size: 539x208 px · Cycle: 5 s
// Needs: kit photo -> https://motion-prompt.2077tech.com/api/v1/kit/photo.js
// Live demo: https://motion-prompt.2077tech.com/#113
//
// cells[i](k, w, h, u): k = CanvasRenderingContext2D (already cleared and wrapped in save/restore by the caller), w,h = size, u = seconds in [0, cycle). The picture depends only on u.
// Element code starts here. CELLS, constants and helpers live in the kit.

// @element 17
// ハイキー: 明るい色だけになり、影はごく薄く、明るさの山が右端へ寄る
CELLS[16] = cell((k, u) => {
  const a = env(u);
  const HIGH = {
    sky: '#FFFFFF', sky2: '#FFFFFF', far: '#F5F6F8', mid: '#EEF0F3', ground: '#FBFBFA', ground2: '#F3F3F1', mark: '#FFFFFF', cast: '#F3F3F1',
    bld: '#FBFBFA', edge: '#E9E9E7', win: '#F2F2F0', winOff: '#F2F2F0', tree: '#F0F0EE', trunk: '#E5E5E3', post: '#DADAD8', bulb: '#E5E5E3',
    person: '#B9C0CE', shade: '#A6AEBF', lit: '#FFE6AE', car: '#E4E7EC', glass: '#FFFFFF', tire: '#C5CAD3', sun: '#FFD889', glow: '#FFF3D8'
  };
  stageLand(k, pal(BASE, HIGH, a));
  histo(k, lerp(0.56, 0.85, a), lerp(0.2, 0.09, a));
});
