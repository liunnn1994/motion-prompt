// スタガー / Stagger
// パック: motion (モーショングラフィックスの動き) · element 02 · カテゴリ: 登場・文字の演出
// 何が起きるか: グラフの棒が、少しずつ時間をずらして左から順に立ち上がる。
// 動きの仕様:
//   - 2.35秒で1回。棒7本（幅44・間隔20・高さ 52, 84, 68, 112, 92, 136, 158）。色は青・コーラル・黄・ミントの順で繰り返す
//   - i本目（0始まり）は 0.15 + 0.12×i 秒に開始。0.5秒かけて高さ 0→100% を backOut（s=2.6）で伸ばす（底辺固定・行き過ぎてから戻る）
//   - 棒の上の白い点（半径6.5・棒の上端から16px上）: 棒の開始から 0.34秒遅れて ばね 1 − e^(−7t)·cos(16t) で出す
//   - 退場: 棒は 1.78 + 0.045×i 秒から0.26秒で easeIn（3乗）で縮む。点は 1.70 + 0.045×i 秒から0.2秒で先に消える
// 使いどころ: リスト・グラフ・カードなど、複数の要素を順番に出して視線を流したいとき。
// タグ: stagger, sequence, delay, cascade, bar-chart, list, entrance
// サイズ: 539x208 px · 1周: 4.7 s
// Needs: kit motion -> https://motion-prompt.2077tech.com/api/v1/kit/motion.js
// 動く見本: https://motion-prompt.2077tech.com/#002
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 02
// 02 スタガー（グラフの棒が、少しずつ時間をずらして順に立ち上がる）
CELLS[1] = (k, w, h, u) => {
  stage(k);
  const N = 7, BW = 44, G = 20, x0 = (w - (N * BW + (N - 1) * G)) / 2, base = h - 30;
  const HS = [52, 84, 68, 112, 92, 136, 158];
  k.fillStyle = C.navy3; rr(k, x0 - 18, base, N * BW + (N - 1) * G + 36, 5, 2.5); k.fill();
  const lu = mod(u, CYC / 2);
  for (let i = 0; i < N; i++) {
    const t0 = 0.15 + i * 0.12, g = backOut(P(lu, t0, t0 + 0.5), 2.6) * (1 - eIn(P(lu, 1.78 + i * 0.045, 2.04 + i * 0.045)));
    if (g <= 0.001) continue;
    const hh = HS[i] * g, x = x0 + i * (BW + G);
    k.fillStyle = PAL[i % 4]; rr(k, x, base - hh, BW, hh, [12, 12, 3, 3]); k.fill();
    // 棒の上の点（さらに遅れて出る）
    const d = spring(lu - t0 - 0.34, 16, 7) * (1 - eIn(P(lu, 1.7 + i * 0.045, 1.9 + i * 0.045)));
    if (d > 0.01) { k.fillStyle = '#fff'; k.beginPath(); k.arc(x + BW / 2, base - hh - 16, 6.5 * d, 0, 7); k.fill(); }
  }
};
