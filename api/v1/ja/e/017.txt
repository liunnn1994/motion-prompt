// マスクトランジション / Mask transition
// パック: motion (モーショングラフィックスの動き) · element 17 · カテゴリ: 切り替え・見せ場
// 何が起きるか: 丸や星の形の「窓」が広がり、その中に次の画面が見える。
// 動きの仕様:
//   - 昼の画面 → 夜の画面 → 昼の画面を2.35秒ずつ。各区間の 0.45〜1.6秒（1.15秒）で切り替え、easeInOut（3乗）
//   - 今の画面の上に、形で切り抜いた次の画面を重ねる。形を広げて画面全体を覆ったら完了
//   - 昼→夜: 丸。太陽の中心（112, 104）から半径 0→470
//   - 夜→昼: 5角の星。月の中心（427, 104）から外側の半径 0→1150、同時に1.2rad（約69°）回す
//   - 広がっている間、形のふちに白い線（線幅7）を付ける
// 使いどころ: 画面の中の物（太陽・ロゴ・ボタン）を起点に、次の場面へ切り替えたいとき。
// タグ: mask, transition, iris, wipe, shape-wipe, reveal, scene-change
// サイズ: 539x208 px · 1周: 4.7 s
// Needs: kit motion -> https://motion-prompt.2077tech.com/api/v1/kit/motion.js
// 動く見本: https://motion-prompt.2077tech.com/#017
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 17
// 17 マスクトランジション（丸や星の形の「窓」が広がり、その中に次の画面が見える）
CELLS[16] = abCell((k, w, h, u, a, b) => {
  const p = eIO(P(u, TA - 0.05, TB + 0.1));
  SCR[a](k, w, h);
  if (p > 0) {
    const c = a === 0 ? SUN : MOON, shape = () => { if (a === 0) { k.beginPath(); k.arc(c[0], c[1], lerp(0, 470, p), 0, 7); } else star(k, c[0], c[1], lerp(0, 1150, p), p * 1.2, 0.5); };
    k.save(); shape(); k.clip(); SCR[b](k, w, h); k.restore();
    if (p < 1) { k.strokeStyle = '#fff'; k.lineWidth = 7; k.lineJoin = 'round'; shape(); k.stroke(); }
  }
});
