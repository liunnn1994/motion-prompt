// ラックフォーカス / Rack focus
// パック: camera (カメラワークと切り替え) · element 09 · カテゴリ: 撮り方・時間の演出
// 何が起きるか: カメラは動かさず、ピントを手前の花から奥の人物へ送る。ぼけている側が入れ替わる。
// 動きの仕様:
//   - カメラ: 位置・向き・焦点距離すべて固定（高さ1.1m）。変わるのはぼかしの量だけ
//   - ぼかし: 手前の花（左0.75m・奥2m）を 0px → 7px、人物と背景（人物は右0.9m・奥9m）を 6px → 0px。地面の格子は背景の0.6倍
//   - 時間: 1.0秒から2.6秒まで1.6秒。3次のイーズインアウト（cubic-bezier(0.65,0,0.35,1) 相当）
//   - ピント枠: 青い四隅の枠が花から人物へ直線移動し、一辺の半分が 46px → 30px に縮む
//   - 被写体: 花も人物も動かない
//   - 戻り: 2.6〜4.2秒は止め、最後の0.5秒で開始状態へクロスフェード。右上にピント距離（◎ 2m → ◎ 9m）
// 使いどころ: カットを割らずに、見てほしい対象を手前から奥（または逆）へ移すとき。
// タグ: rack focus, focus pull, pull focus, depth of field, blur, bokeh, foreground
// サイズ: 539x208 px · 1周: 9.4 s
// Needs: kit camera -> https://motion-prompt.2077tech.com/api/v1/kit/camera.js
// 動く見本: https://motion-prompt.2077tech.com/#033
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 09
// 09 ラックフォーカス
CELLS[8] = shot(camCell((k, w, h, p) => {
  world(k, w, h, cam({ y: 1.1 }), {
    person: { x: 0.9, z: 9 }, blurFg: 7 * p, blurBg: 6 * (1 - p),
    extra: [[-0.75, 0, 2.0, (kk, s) => { kk.fillStyle = C.g3; kk.fillRect(-0.025, 0, 0.05, 0.75); kk.fillStyle = C.blueM; for (let i = 0; i < 6; i++) { kk.beginPath(); kk.ellipse(Math.cos(i) * 0.09, 0.8 + Math.sin(i) * 0.09, 0.08, 0.05, i, 0, 7); kk.fill(); } kk.fillStyle = C.blue; kk.beginPath(); kk.arc(0, 0.8, 0.05, 0, 7); kk.fill(); }, true]],
    after: (kk, cc) => {   // ピント位置の枠
      const a = proj(cc, -0.75, 0.8, 2.0), b = proj(cc, 0.9, 0.9, 9);
      const x = lerp(a.x, b.x, p), y = lerp(a.y, b.y, p), r = lerp(46, 30, p);
      kk.strokeStyle = C.blue; kk.lineWidth = 2; kk.beginPath();
      [[-1, -1], [1, -1], [-1, 1], [1, 1]].forEach(([sx, sy]) => { kk.moveTo(x + sx * r, y + sy * r - sy * 9); kk.lineTo(x + sx * r, y + sy * r); kk.lineTo(x + sx * r - sx * 9, y + sy * r); });
      kk.stroke();
    }
  });
}, p => `◎ ${lerp(2, 9, p).toFixed(0)}m`, 1.0, 2.6));
