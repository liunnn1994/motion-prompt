// マッチカット / Match cut
// パック: camera (カメラワークと切り替え) · element 18 · カテゴリ: 切り替え（トランジション）
// 何が起きるか: 投げ上げたボールが、次のカットでは同じ位置・同じ大きさの太陽になる。形をそろえて別の場面へつなぐ。
// 動きの仕様:
//   - つなぎ方: 2.3秒で直結（0フレーム）。前後で青い円の位置（中心から右120px・上46px）と半径17pxを一致させる
//   - 前の絵（0〜2.3秒）: 固定カメラ（高さ1.4m）。人物（左1.2m・奥7m）が0.5秒から両腕を上げてボールを投げる。3次のイーズアウト（cubic-bezier(0.33,1,0.68,1) 相当）
//   - ボール: 人物の頭上（高さ2.3m）から一致点へ移動。半径 5px → 17px、途中は最大30px上へふくらむ弧
//   - 後の絵（2.3〜4.7秒）: 別の場所（建物なし・高さ1.2m）。同じ位置の青い円が太陽になり、2.4秒かけて14px下がる
//   - 目印: 半径26pxの青い点線の円を、カットの前後0.7秒だけフェードで表示
//   - 右上に「SHOT 1」→「SHOT 2」。4.7秒でSHOT 1へ直結で戻る
// 使いどころ: 時間や場所が大きく離れた2つの場面を、形や動きの共通点で意味を持たせてつなぐとき。
// タグ: match cut, graphic match, shape match, cut, transition, edit
// サイズ: 539x208 px · 1周: 9.4 s
// Needs: kit camera -> https://motion-prompt.2077tech.com/api/v1/kit/camera.js
// 動く見本: https://motion-prompt.2077tech.com/#042
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 18
// 18 マッチカット（投げたボール → 同じ位置・大きさの太陽）
CELLS[17] = shot((k, w, h, u) => {
  const cut = 2.3;
  const BX = 120, BY = -46, BR = 17;
  if (u < cut) {
    const q = eOut(P(u, 0.5, cut));
    world(k, w, h, cam({ y: 1.4 }), {
      person: { x: -1.2, z: 7, armUp: q > 0.05 }, sun: null,
      after: (kk, cc) => { const s = proj(cc, -1.2, 2.3, 7); const x = lerp(s.x, BX, q), y = lerp(s.y, BY, q) - Math.sin(q * Math.PI) * 30, r = lerp(5, BR, q); kk.fillStyle = C.blue; kk.beginPath(); kk.arc(x, y, r, 0, 7); kk.fill(); }
    });
  } else {
    const q = P(u, cut, CYC);
    world(k, w, h, cam({ y: 1.2 }), {
      buildings: false, sun: null, trees: true,
      after: (kk) => { kk.fillStyle = C.blue; kk.beginPath(); kk.arc(BX, BY + q * 14, BR, 0, 7); kk.fill(); }
    });
  }
  // 形がそろっていることを示す点線の円
  const g = Math.max(0, 1 - Math.abs(u - cut) / 0.7);
  if (g > 0) { k.save(); k.translate(w / 2, h / 2); k.globalAlpha = g; k.strokeStyle = C.blue; k.lineWidth = 2; k.setLineDash([5, 4]); k.beginPath(); k.arc(BX, BY + (u > cut ? P(u, cut, CYC) * 14 : 0), BR + 9, 0, 7); k.stroke(); k.restore(); }
  return u < cut ? 'SHOT 1' : 'SHOT 2';
});
