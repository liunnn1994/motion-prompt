// ホイップパン / Whip pan
// パック: camera (カメラワークと切り替え) · element 19 · カテゴリ: 切り替え（トランジション）
// 何が起きるか: カメラを素早く横に振り、流れてぼけた映像の中で次の場面に入れ替える。
// 動きの仕様:
//   - つなぎ方: 1.7秒から2.5秒までの0.8秒。前半0.4秒は前の絵を振り出し、後半0.4秒は次の絵が振り込んで止まる
//   - 前半: 前の絵のヨーを 0 → +0.45rad（約26°）、2次のイーズインアウト。後半: 次の絵のヨーを -0.45rad → 0、同じイージング
//   - 入れ替え: 進み50%の瞬間に直結で差し替える。ぶれが最大なので切れ目は見えない
//   - ぶれ: 前後1.6フレームぶん（30fps）の範囲を9枚描いて平均する
//   - カメラ: 高さ1.4m。振っている間以外は固定。被写体（Aは人物、Bは車と街）は動かない
//   - 4.7秒ごとに A→B、B→A を交互に行う（往復で9.4秒）。右上に「A → B」「B → A」
// 使いどころ: 勢いを保ったまま場所や話題を切り替えるとき。
// タグ: whip pan, swish pan, motion blur, fast pan, transition, energy
// サイズ: 539x208 px · 1周: 9.4 s
// Needs: kit camera -> https://motion-prompt.2077tech.com/api/v1/kit/camera.js
// 動く見本: https://motion-prompt.2077tech.com/#043
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 19
// 19 ホイップパン（ブレで次の場面へ）
CELLS[18] = shot(abCell((k, w, h, u, A, B) => {
  const t0 = 1.7, t1 = 2.5;
  const at = (kk, s) => {   // s: 0..1 の進み。前半 A を右へ振り、後半 B が左から止まる
    if (s < .5) world(kk, w, h, cam({ y: 1.4, yaw: eIOs(s * 2) * 0.45 }), A === sceneA ? { person: { x: -0.6, z: 7 } } : SB);
    else world(kk, w, h, cam({ y: 1.4, yaw: -(1 - eIOs((s - .5) * 2)) * 0.45 }), B === sceneA ? { person: { x: -0.6, z: 7 } } : SB);
  };
  if (u < t0) return A(k, w, h);
  if (u > t1) return B(k, w, h);
  const s = P(u, t0, t1), ds = 1 / FPS / (t1 - t0);
  mblur(k, at, clamp(s - ds * 1.6), clamp(s + ds * 1.6), 9);
}));
