// モーダル / Modal dialog
// パック: ui (UIパーツの名前) · element 01 · カテゴリ: 開く・重なるパーツ
// 何が起きるか: 「開く」ボタンを押すと画面全体が暗くなり、中央に小窓が少し拡大しながら出る。×を押すと閉じる。
// 動きの仕様:
//   - 4.5秒周期・カード542×191。カーソルが0.95秒に左下の「開く」ボタン（120×36）へ着き、1.0秒にクリック
//   - 開く量 a: 1.0〜1.32秒に easeOutCubic で 0→1。全面に rgba(18,20,28) の幕を不透明度 0.42×a で敷く
//   - 窓: 白 280×140・角丸12を (271, 97) 中心に置き、影 blur 24・下へ8・濃さ0.25。拡大率 0.9+0.1×a、不透明度 a
//   - 窓の中: 見出しの太線・本文2行・右下に青いボタン 88×30・右上に×
//   - 閉じる: カーソルが3.0秒に×へ移り3.3秒にクリック。a に (1 − easeInOutCubic(3.3〜3.58秒)) を掛けて消す
// 使いどころ: 確認・入力など、今の作業を止めて1つのことに集中させたいとき。
// タグ: modal, modal-dialog, dialog, popup, overlay, backdrop, lightbox, confirm
// サイズ: 542x191 px · 1周: 4.5 s
// Needs: kit ui -> https://motion-prompt.2077tech.com/api/v1/kit/ui.js
// 動く見本: https://motion-prompt.2077tech.com/#049
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 01
// 01 モーダル（Modal dialog）
CELLS[0] = ui((w, h, t) => {
  pageLines(24, 18, 494);
  fillRR(24, 92, 120, 36, 8, C.blue); text('開く', 84, 110, 16, '#fff', 600, 'center');
  fillRR(170, 92, 348, 36, 8, C.block); fillRR(24, 142, 494, 38, 8, C.block);
  const a = eo(seg(t, 1.0, 1.32)) * (1 - eio(seg(t, 3.3, 3.58)));
  if (a > 0) {
    X.fillStyle = `rgba(18,20,28,${.42 * a})`; X.fillRect(0, 0, w, h);
    X.save(); X.globalAlpha = a; X.translate(271, 97); const s = .9 + .1 * a; X.scale(s, s);
    shadowed(() => fillRR(-140, -70, 280, 140, 12, '#fff'), 24, 8, .25);
    bar(-116, -48, 120, C.ink, 11); bar(-116, -20, 232); bar(-116, -2, 180);
    fillRR(28, 26, 88, 30, 7, C.blue);
    X.strokeStyle = '#9a9896'; X.lineWidth = 2.2; X.lineCap = 'round'; X.beginPath(); X.moveTo(110, -54); X.lineTo(120, -44); X.moveTo(120, -54); X.lineTo(110, -44); X.stroke();
    X.restore();
  }
  pointer([[.3, w + 30, h + 30], [.95, 86, 114], [1.5, 86, 114], [3.0, 389, 50], [3.5, 389, 50], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
