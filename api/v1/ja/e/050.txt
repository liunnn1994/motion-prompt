// 動きの仕様:
//   - 4.5秒周期・カード542×191。カーソルが左上の「メニュー」ボタン（110×34）を1.0秒にクリック
//   - 開く量 a: 1.0〜1.38秒に easeOutCubic で 0→1。全面に rgba(18,20,28) の幕を不透明度 0.36×a
//   - パネル: 白・幅210・カードの高さいっぱい。x = −210 + 210×a で左端から出る。影 blur 22・濃さ0.25
//   - パネルの中: 上に丸アイコン（半径14）と名前、下に32px間隔で4行のメニュー。1行目は薄い青の帯で選択中
//   - 閉じる: カーソルがパネルの外 (420, 120) を3.3秒にクリック。a に (1 − easeInOutCubic(3.3〜3.65秒)) を掛けて左へ戻す
// 使いどころ: スマホの全体メニューや、絞り込み条件など、場所を取る操作を横に隠しておきたいとき。
// タグ: drawer, off-canvas, offcanvas, side-menu, navigation-drawer, sheet, sidebar, slide-in
// サイズ: 542x191 px · 1周: 4.5 s
// Needs: kit ui -> https://motion-prompt.2077tech.com/api/v1/kit/ui.js
// 動く見本: https://motion-prompt.2077tech.com/#050
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 02
// 02 ドロワー（Drawer (Off-canvas)）
CELLS[1] = ui((w, h, t) => {
  strokeRR(24, 14, 110, 34, 8, C.line2, 1.5); text('菜单', 79, 31, 15, '#555', 600, 'center');
  bar(160, 26, 140, C.line2, 11); pageLines(24, 70, 494); fillRR(24, 138, 494, 42, 8, C.block);
  const a = eo(seg(t, 1.0, 1.38)) * (1 - eio(seg(t, 3.3, 3.65)));
  if (a > 0) {
    X.fillStyle = `rgba(18,20,28,${.36 * a})`; X.fillRect(0, 0, w, h);
    const pw = 210, px = -pw + pw * a;
    shadowed(() => { X.fillStyle = '#fff'; X.fillRect(px, 0, pw, h); }, 22, 0, .25);
    circle(px + 34, 32, 14, C.mb); bar(px + 58, 27, 100, C.line2, 10);
    for (let i = 0; i < 4; i++) { const y = 62 + i * 32; if (i === 0) fillRR(px + 12, y - 2, pw - 24, 28, 6, C.lb); fillRR(px + 22, y + 5, 14, 14, 3, i === 0 ? C.blue : C.line2); bar(px + 46, y + 7, 90 + (i % 2) * 30, i === 0 ? C.sb : C.line); }
  }
  pointer([[.3, w + 30, h + 30], [.95, 70, 34], [1.6, 70, 34], [3.0, 420, 120], [3.6, 420, 120], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
