// ボトムシート / Bottom sheet
// パック: ui (UIパーツの名前) · element 03 · カテゴリ: 開く・重なるパーツ
// 何が起きるか: 指で「詳細を見る」を押すと下からパネルがせり上がり、上端のつまみを下へ引くと閉じる。
// 動きの仕様:
//   - 4.5秒周期・カード542×191。指（半径17の半透明の丸＋白い縁）が下中央の「詳細を見る」（170×36の青い角丸）を1.0秒に押す
//   - 開く量: 1.05〜1.45秒に easeOutCubic で 0→1。全面に rgba(18,20,28) の幕を不透明度 0.36×a
//   - シート: 白・高さ136・上の角丸18。上端 y = 191 − 136×a。影 blur 22・上へ4。上端中央につまみ 52×6
//   - シートの中: 見出しの太線と、アイコン（26×26）付きの2行
//   - 閉じる: 指が2.75秒につまみ (271, 68) に乗り、3.0〜3.6秒に下へ引く。a に (1 − easeInOutCubic(3.0〜3.6秒)) を掛け、シートが指と一緒に下へ消える
// 使いどころ: スマホで、今の画面を残したまま詳細や選択肢を出したいとき。
// タグ: bottom-sheet, sheet, modal-sheet, mobile, slide-up, drag-to-dismiss, panel
// サイズ: 542x191 px · 1周: 4.5 s
// Needs: kit ui -> https://motion-prompt.2077tech.com/api/v1/kit/ui.js
// 動く見本: https://motion-prompt.2077tech.com/#051
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 03
// 03 ボトムシート（Bottom sheet）
CELLS[2] = ui((w, h, t) => {
  pageLines(24, 16, 494); fillRR(24, 76, 239, 54, 8, C.block); fillRR(279, 76, 239, 54, 8, C.block);
  fillRR(186, 144, 170, 36, 18, C.blue); text('詳細を見る', 271, 162, 15, '#fff', 600, 'center');
  const open = eo(seg(t, 1.05, 1.45));
  const drag = eio(seg(t, 3.0, 3.6));
  const a = open * (1 - drag);
  if (open > 0 && a > 0.001) {
    X.fillStyle = `rgba(18,20,28,${.36 * a})`; X.fillRect(0, 0, w, h);
    const sh = 136, sy = h - sh * a;
    shadowed(() => { rr(0, sy, w, sh + 20, 18); X.fillStyle = '#fff'; X.fill(); }, 22, -4, .2);
    fillRR(w / 2 - 26, sy + 10, 52, 6, 3, C.line2);
    bar(28, sy + 30, 160, C.ink, 11);
    for (let i = 0; i < 2; i++) { const y = sy + 56 + i * 36; fillRR(28, y, 26, 26, 6, i ? C.mb : C.blue); bar(66, y + 4, 220, C.line); bar(66, y + 17, 140, C.line, 7); }
  }
  const sy = h - 136 * a;
  finger([[.3, 271, 162], [1.6, 271, 162], [2.75, w / 2, h - 136 + 13], [3.0, w / 2, h - 136 + 13], [3.6, w / 2, h + 10]], [1.0, 3.0], t, (t > .6 && t < 1.6) || (t > 2.75 && t < 3.6));
});
