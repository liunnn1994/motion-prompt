// アコーディオン / Accordion
// パック: ui (UIパーツの名前) · element 04 · カテゴリ: 開く・重なるパーツ
// 何が起きるか: 3段の見出しのうち1段目を押すと下に本文が開き、下の段が押し下げられる。もう一度押すと閉じる。
// 動きの仕様:
//   - 4.5秒周期・カード542×191。3段の見出し（幅494・高さ34・角丸8・間隔6）。右端に下向きの矢印
//   - カーソルが0.95秒に1段目に着き、1.0秒と3.3秒にクリック
//   - 開く量 ex: 1.0〜1.35秒に easeInOutCubic で 0→1、3.3〜3.62秒に同じく 1→0
//   - 1段目の高さを 34 → 92（+58×ex）に伸ばし、下の段をその分押し下げる。本文2行は枠の中に切り抜いて不透明度 ex で出す
//   - 開いている間、矢印は 180°×ex 回り、枠は薄い青・見出しは青になる
// 使いどころ: FAQや詳細な説明など、見出しだけ並べて中身は必要な人にだけ見せたいとき。
// タグ: accordion, collapsible, expand, collapse, disclosure, faq, details
// サイズ: 542x191 px · 1周: 4.5 s
// Needs: kit ui -> https://motion-prompt.2077tech.com/api/v1/kit/ui.js
// 動く見本: https://motion-prompt.2077tech.com/#052
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 04
// 04 アコーディオン（Accordion）
CELLS[3] = ui((w, h, t) => {
  const ex = eio(seg(t, 1.0, 1.35)) * (1 - eio(seg(t, 3.3, 3.62)));
  let y = 12; const H0 = 34, EX = 58;
  for (let i = 0; i < 3; i++) {
    const extra = i === 0 ? EX * ex : 0;
    strokeRR(24, y, 494, H0 + extra, 8, i === 0 && ex > .05 ? C.sb : C.line2, 1.5);
    bar(42, y + 12, [180, 220, 150][i], i === 0 && ex > .05 ? C.blue : C.line2, 10);
    chevron(494, y + 17, 6, i === 0 ? Math.PI * ex : 0, i === 0 && ex > .05 ? C.blue : '#9a9896');
    if (i === 0 && ex > 0) { X.save(); rr(24, y, 494, H0 + extra, 8); X.clip(); X.globalAlpha = ex; bar(42, y + H0 + 6, 420); bar(42, y + H0 + 24, 320); X.restore(); }
    y += H0 + extra + 6;
  }
  pointer([[.3, w + 30, h + 30], [.95, 300, 31], [1.6, 300, 31], [3.0, 300, 31], [3.55, 300, 31], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
