// ドロップダウン / Dropdown (select)
// パック: ui (UIパーツの名前) · element 05 · カテゴリ: 開く・重なるパーツ
// 何が起きるか: 入力欄を押すと下に4つの選択肢が開き、「名古屋」を選ぶと閉じて欄に入る。
// 動きの仕様:
//   - 4.5秒周期・カード542×191。入力欄 260×38（角丸8）に「選択してください」と右端の矢印。カーソルが1.0秒にクリック
//   - 開く量: 1.0〜1.25秒に easeOutCubic で 0→1。一覧（白 260×112・影 blur 18）が不透明度とともに上から6px下りて出る。矢印は180°回り、枠は青2px
//   - 一覧は26px間隔の4行（北京・上海・深圳・广州）。カーソルが下りるにつれ1.55秒から上海、1.95秒から深圳の行を薄い青で強調
//   - 2.3秒に深圳をクリック: 一覧は2.3〜2.45秒に (1 − easeInOutCubic) で閉じ、プレースホルダーは2.2〜2.3秒に消え、「深圳」が2.32〜2.42秒に出る
//   - 戻し: 深圳は3.9〜4.08秒に消え、プレースホルダーが4.1〜4.3秒に戻る（文字を重ねない）
// 使いどころ: フォームで、決まった候補の中から1つを選ばせたいとき。
// タグ: dropdown, dropdown-menu, select, picker, combobox, options, form
// サイズ: 542x191 px · 1周: 4.5 s
// Needs: kit ui -> https://motion-prompt.2077tech.com/api/v1/kit/ui.js
// 動く見本: https://motion-prompt.2077tech.com/#053
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 05
// 05 ドロップダウン（Dropdown menu）
CELLS[4] = ui((w, h, t) => {
  bar(24, 14, 90, C.line2, 10);
  const open = eo(seg(t, 1.0, 1.25)) * (1 - eio(seg(t, 2.3, 2.45)));
  const chosen = seg(t, 2.32, 2.42) * (1 - seg(t, 3.9, 4.08));
  const ph = (1 - seg(t, 2.2, 2.3)) + seg(t, 4.1, 4.3); // プレースホルダーは先に消し、名前が消えてから戻す（重ねない）
  strokeRR(24, 34, 260, 38, 8, open > .1 || (chosen > 0 && t < 2.9) ? C.blue : C.line2, open > .1 ? 2 : 1.5);
  if (ph > 0) { X.globalAlpha = clamp(ph); text('请选择', 40, 53, 15, '#9a9896', 500); X.globalAlpha = 1; }
  if (chosen > 0) { X.globalAlpha = chosen; text('深圳', 40, 53, 16, C.ink, 600); X.globalAlpha = 1; }
  chevron(262, 52, 5, Math.PI * open, '#777');
  fillRR(310, 34, 208, 70, 8, C.block); bar(310, 120, 208); bar(310, 138, 160); bar(24, 150, 200); bar(24, 168, 150);
  if (open > 0) {
    X.save(); X.globalAlpha = open; const oy = 78 - 6 * (1 - open);
    shadowed(() => fillRR(24, oy, 260, 112, 8, '#fff'), 18, 6, .18); strokeRR(24, oy, 260, 112, 8, C.border, 1);
    const hi = seg(t, 1.55, 1.75) > 0 ? (t < 1.95 ? 1 : 2) : -1;
    ['北京', '上海', '深圳', '广州'].forEach((s, i) => { const yy = oy + 4 + i * 26; if (i === hi + (hi === 1 ? 0 : 0) && hi >= 0) fillRR(28, yy, 252, 26, 5, C.lb); text(s, 40, yy + 13, 15, i === hi ? C.blue : '#444', i === hi ? 600 : 500); });
    X.restore();
  }
  pointer([[.3, w + 30, h + 30], [.95, 200, 56], [1.25, 200, 56], [1.6, 170, 121], [2.0, 170, 147], [2.45, 170, 147], [3.2, 170, 147], [4.2, w + 30, h + 30]], [1.0, 2.3], t);
});
