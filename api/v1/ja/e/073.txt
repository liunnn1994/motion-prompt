// パララックススクロール / Parallax scrolling
// パック: lp (LP・Webサイトの演出) · element 01 · カテゴリ: スクロールで起きる演出
// 何が起きるか: ページを下へスクロールすると、見出しはふつうに上がり、空と太陽・遠い山・近い山はそれより遅く動く。
// 動きの仕様:
//   - 表示域 511×168 のブラウザの窓。スクロール量: 0.3〜1.9秒で 0→120px、2.9〜4.3秒で 120→0（easeInOut 3乗）
//   - 各層の移動量 = スクロール量 × 係数。空と太陽 0.15、遠い山 0.4、近い山 0.7
//   - 見出し（28px）・ボタン・次のセクションは 1.0（ふつうのスクロール）
//   - 次のセクション（白地）はファーストビューの下端（168px）から始まり、遅れて動く背景の上にかぶさる
//   - 右端のスクロールバーのつまみもスクロール量に合わせて動く
// 使いどころ: ファーストビューや区切りの写真に奥行きを出したいとき。
// タグ: parallax, parallax-scrolling, scroll, depth, layers, hero, background
// サイズ: 539x208 px · 1周: 4.8 s
// Needs: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// 動く見本: https://motion-prompt.2077tech.com/#073
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 01
// 01 パララックススクロール（スクロールすると、空・山・見出しが違う速さで動く）
CELLS[0] = (k, w, h, u) => {
  frame(k);
  const s = steps([[0.3, 0], [1.9, 1], [2.9, 1], [4.3, 0]], u), y = s * 120;
  view(k, () => {
    const g = k.createLinearGradient(0, 0, 0, 200); g.addColorStop(0, '#9DB6FB'); g.addColorStop(1, '#E6ECFF');
    k.save(); k.translate(0, -y * 0.15);                         // 空と太陽: 0.15倍
    k.fillStyle = g; k.fillRect(0, 0, VW, 260); k.fillStyle = C.acc; k.beginPath(); k.arc(392, 54, 25, 0, 7); k.fill(); k.restore();
    k.save(); k.translate(0, -y * 0.4); mount(k, VW, 190, M1, '#7D9AF4', 0); k.restore();   // 遠い山: 0.4倍
    k.save(); k.translate(0, -y * 0.7); mount(k, VW, 205, M2, '#4B6BDB', 1); k.restore();   // 近い山: 0.7倍
    k.save(); k.translate(0, -y);                                // 見出し・ボタン・次のセクション: 1倍
    txt(k, '一起去山里。', 34, 64, `800 28px ${JP}`, '#fff', 'left', 'alphabetic', '1px');
    bars(k, 34, 78, 160, 2, 'rgba(255,255,255,.85)');
    btn(k, 34, 106, 92, 26, '立即预订');
    k.fillStyle = '#fff'; k.fillRect(0, VH, VW, 200);
    txt(k, '热门路线', 34, VH + 34, `800 16px ${JP}`, C.ink);
    for (let i = 0; i < 3; i++) { pic(k, 34 + i * 152, VH + 50, 136, 58, i, 6); bars(k, 34 + i * 152, VH + 116, 110, 1, C.faint); }
    k.restore();
    sbar(k, s * 0.35, 0.4);
  });
};
