// 追従ヘッダー / Sticky header
// パック: lp (LP・Webサイトの演出) · element 03 · カテゴリ: スクロールで起きる演出
// 何が起きるか: 上のお知らせの帯はスクロールで消えるが、その下のヘッダーは画面の上端に来たところで止まり、本文がその下を通っていく。
// 動きの仕様:
//   - スクロール量: 0.4〜1.3秒で 0→110px、1.9〜2.8秒で 110→250px、3.4〜4.4秒で 250→0（easeInOut 3乗）
//   - ページの一番上に高さ22pxのお知らせの帯（濃紺）、その下に高さ34pxのヘッダー（ロゴ・メニュー3つ・ボタン）
//   - ヘッダーの位置 = max(22 − スクロール量, 0)。帯が見えなくなった所で上端に止まる（position: sticky; top: 0）
//   - 止まってからの24pxのスクロールで、ヘッダーの下に高さ12pxの影を 不透明度 0→0.16 で付ける
//   - 本文（写真・見出し・カード3枚）はヘッダーの下をふつうにスクロールする
// 使いどころ: 長いページでも、メニューや申し込みボタンにいつでも手が届くようにしたいとき。
// タグ: sticky, sticky-header, fixed-header, position-sticky, navigation, header, scroll
// サイズ: 539x208 px · 1周: 4.8 s
// Needs: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// 動く見本: https://motion-prompt.2077tech.com/#075
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 03
// 03 追従ヘッダー（スクロールすると、ヘッダーが画面の上で止まってついてくる）
CELLS[2] = (k, w, h, u) => {
  frame(k);
  const y = steps([[0.4, 0], [1.3, 110], [1.9, 110], [2.8, 250], [3.4, 250], [4.4, 0]], u);
  const STRIP = 22;
  view(k, () => {
    k.save(); k.translate(0, -y);
    pic(k, 24, 70, 214, 112, 0, 8);
    txt(k, '开启全新一天，', 262, 104, `800 22px ${JP}`, C.ink); bars(k, 262, 120, 210, 3); btn(k, 262, 160, 100, 26, '查看详情');
    k.fillStyle = C.soft; k.fillRect(0, 212, VW, 240);
    txt(k, '特色', VW / 2, 246, `800 18px ${JP}`, C.ink, 'center');
    for (let i = 0; i < 3; i++) { const x = 34 + i * 152; k.fillStyle = '#fff'; rr(k, x, 262, 136, 120, 8); k.fill(); pic(k, x + 8, 270, 120, 58, i, 5); bars(k, x + 10, 340, 110, 3, C.faint, 6, 11); }
    k.fillStyle = C.ink; k.fillRect(0, 0, VW, STRIP);
    txt(k, '限时优惠进行中', VW / 2, STRIP / 2 + 0.5, `600 11px ${JP}`, '#fff', 'center', 'middle');
    k.restore();
    const hy = Math.max(STRIP - y, 0), sh = clamp((y - STRIP) / 24);
    if (sh > 0) { const g = k.createLinearGradient(0, hy + 34, 0, hy + 46); g.addColorStop(0, `rgba(28,35,51,${0.16 * sh})`); g.addColorStop(1, 'rgba(28,35,51,0)'); k.fillStyle = g; k.fillRect(0, hy + 34, VW, 12); }
    header(k, hy);
    sbar(k, y / 250, 0.42);
  });
};
