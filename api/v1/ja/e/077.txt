// 横スクロール / Horizontal scroll
// パック: lp (LP・Webサイトの演出) · element 05 · カテゴリ: スクロールで起きる演出
// 何が起きるか: 縦にスクロールすると、セクションが止まったまま、番号付きの作品カード4枚が左へ流れていく。
// 動きの仕様:
//   - スクロール量: 0.3〜3.0秒で 0→420px、3.5〜4.5秒で 420→0（easeInOut 3乗）
//   - 見出し「WORKS」は上へ流れ、セクションはスクロール量40pxで上端に着いて止まる
//   - カード: 200×112 を4枚、間隔20、左右の余白34。横に動く量 = 34×2 + 200×4 + 20×3 − 511 = 417px
//   - カードの横位置 = 元の位置 − p × 417。p = (スクロール量 − 40) / 380（スクロールに比例・イージングなし）
//   - スクロールバーのつまみは縦に動き、カードは横に動く
// 使いどころ: 作品・事例・商品を、縦長のページの中で横一列に見せたいとき。
// タグ: horizontal-scroll, sideways, scrolltrigger, pin, gallery, portfolio, scroll
// サイズ: 539x208 px · 1周: 4.8 s
// Needs: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// 動く見本: https://motion-prompt.2077tech.com/#077
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 05
// 05 横スクロール（縦にスクロールすると、カードの列が横へ流れる）
CELLS[4] = (k, w, h, u) => {
  frame(k);
  const y = steps([[0.3, 0], [3.0, 420], [3.5, 420], [4.5, 0]], u);
  const PIN0 = 40, PD = 380, N = 4, PWD = 200, GAP = 20;
  const top = y < PIN0 ? PIN0 - y : 0, p = clamp((y - PIN0) / PD);
  const travel = 34 * 2 + N * PWD + (N - 1) * GAP - VW;
  view(k, () => {
    txt(k, 'WORKS', 34, 30 - y, `800 20px ${EN}`, C.ink, 'left', 'alphabetic', '2px');
    k.save(); k.translate(0, top);
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH + 40);
    for (let i = 0; i < N; i++) {
      const x = 34 + i * (PWD + GAP) - p * travel;
      pic(k, x, 18, PWD, 112, i % 3, 8);
      txt(k, '0' + (i + 1), x + 14, 118, `800 22px ${EN}`, '#fff');
      bars(k, x, 140, 120, 1, '#CBD2DE');
    }
    k.restore();
    sbar(k, y / 420, 0.36);
  });
};
