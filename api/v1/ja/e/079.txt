// スクロールプログレスバー / Scroll progress bar
// パック: lp (LP・Webサイトの演出) · element 07 · カテゴリ: スクロールで起きる演出
// 何が起きるか: 記事を少しずつ読み進めるごとに、画面の上端の線が左から伸び、最後まで来ると端まで届く。
// 動きの仕様:
//   - 記事のページを 0→430px までスクロール。110pxずつ4回、それぞれ0.45秒で動いて0.4秒止まる（easeInOut 3乗）
//   - 3.75〜4.55秒で 430→0 に戻る
//   - 表示域の上端に高さ6pxの線。地は薄いグレー、伸びる部分は朱色
//   - 伸びる幅 = 511px × スクロール量 / 430（スクロールの割合そのまま。CSS の animation-timeline: scroll() と同じ）
// 使いどころ: 長い記事やコラムで、あとどれくらいで読み終わるかを伝えたいとき。
// タグ: progress-bar, reading-progress, scroll-progress, indicator, scroll-driven, article
// サイズ: 539x208 px · 1周: 4.8 s
// Needs: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// 動く見本: https://motion-prompt.2077tech.com/#079
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 07
// 07 スクロールプログレスバー（読み進めた分だけ、画面上の線が伸びる）
CELLS[6] = (k, w, h, u) => {
  frame(k);
  const MAX = 430;
  const y = steps([[0.3, 0], [0.75, 110], [1.15, 110], [1.6, 220], [2.0, 220], [2.45, 330], [2.85, 330], [3.3, MAX], [3.75, MAX], [4.55, 0]], u);
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, '网页设计入门', 70, 44, `800 20px ${JP}`, C.ink); bars(k, 70, 56, 120, 1, C.faint, 5);
    pic(k, 70, 74, 371, 112, 0, 8);
    for (let b = 0; b < 4; b++) { const y0 = 204 + b * 82; k.fillStyle = C.ink; rr(k, 70, y0, 140, 9, 4.5); k.fill(); bars(k, 70, y0 + 22, 371, 4, C.faint, 6, 12); }
    k.restore();
    k.fillStyle = C.faint; k.fillRect(0, 0, VW, 6);
    k.fillStyle = C.acc; k.fillRect(0, 0, VW * (y / MAX), 6);
    sbar(k, y / MAX, 0.3);
  });
};
