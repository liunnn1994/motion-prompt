// 離脱防止ポップアップ / Exit-intent popup
// パック: lp (LP・Webサイトの演出) · element 16 · カテゴリ: マウス・クリックへの反応
// 何が起きるか: カーソルがページの上の方（タブや閉じるボタンのある所）へ出ていくと、背景が暗くなって「ちょっと待って！」の小窓が開く。×で閉じる。
// 動きの仕様:
//   - カーソルは 0.7〜1.25秒にページの中からブラウザのアドレスバーの方へ上がる（easeInOut 3乗）
//   - 表示域の上端（y = 0）を越えた時刻（約1.07秒）に発火（document の mouseleave で clientY ≤ 0 と同じ判定）
//   - 発火の0.05〜0.45秒後に開く（easeOut 3乗）: 背景 濃紺 不透明度 0→0.5、小窓（260×132・角丸12）不透明度 0→1、拡大率 0.9→1
//   - 小窓の中は見出し・割引の一言（朱色）・クーポンのボタン・右上の×
//   - 2.9秒に×をクリック。0.03〜0.25秒後に消える（等速）
// 使いどころ: 申し込みや購入の直前で帰ってしまう人に、最後の一押しをしたいとき。
// タグ: exit-intent, popup, mouseleave, coupon, conversion, retention, lp
// サイズ: 539x208 px · 1周: 4.8 s
// Needs: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// 動く見本: https://motion-prompt.2077tech.com/#088
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 16
// 16 離脱防止ポップアップ（カーソルがタブや閉じるボタンの方へ出ていくと、引き止める小窓が出る）
CELLS[15] = (k, w, h, u) => {
  frame(k);
  const OX = WIN.x, OY = WIN.y + WIN.bar;               // カーソルは窓の外（タブ・アドレスバー）まで出るので、カードの座標で描く
  const [cx, cy] = path([[0, 440, 150], [0.7, 300, 120], [1.25, 360, -18], [1.9, 372, -20], [2.6, 366, 38], [3.3, 366, 38], [4.4, 440, 150], [4.8, 440, 150]], u);
  const CLOSE = 2.9;
  let TRIG = 0.7; while (TRIG < 1.3 && path([[0.7, 300, 120], [1.25, 360, -18]], TRIG)[1] >= 0) TRIG += 1 / 240;   // 表示域の上端（y = 0）を越えた時刻（約1.07秒）
  const m = eOut(P(u, TRIG + 0.05, TRIG + 0.45)) * (1 - P(u, CLOSE + 0.03, CLOSE + 0.25));
  const D = { x: 125.5, y: 18, w: 260, h: 132 };
  view(k, () => {
    header(k, 0);
    pic(k, 24, 46, 222, 110, 0, 8); txt(k, '新用户指南', 264, 82, `800 19px ${JP}`, C.ink); bars(k, 264, 96, 210, 3); btn(k, 264, 132, 100, 24, '立即报名');
    if (m > 0) {
      k.fillStyle = `rgba(28,35,51,${0.5 * m})`; k.fillRect(0, 0, VW, VH);
      k.save(); k.globalAlpha = m; k.translate(D.x + D.w / 2, D.y + D.h / 2); const s = 0.9 + 0.1 * m; k.scale(s, s); k.translate(-D.w / 2, -D.h / 2);
      k.fillStyle = '#fff'; rr(k, 0, 0, D.w, D.h, 12); k.fill();
      txt(k, '请稍等一下！', D.w / 2, 34, `800 17px ${JP}`, C.ink, 'center');
      txt(k, '首次体验享九折', D.w / 2, 60, `800 15px ${JP}`, C.acc, 'center');
      btn(k, D.w / 2 - 75, 80, 150, 30, '领取优惠券', C.acc, '#fff', 12);
      k.strokeStyle = C.sub; k.lineWidth = 2; k.lineCap = 'round'; k.beginPath(); k.moveTo(D.w - 26, 14); k.lineTo(D.w - 14, 26); k.moveTo(D.w - 14, 14); k.lineTo(D.w - 26, 26); k.stroke();
      k.restore();
    }
  });
  // カーソルの × のクリック（窓の中）と、窓の外への移動
  k.save(); k.beginPath(); k.rect(0, 0, CW, CH); k.clip(); k.translate(OX, OY); click(k, cx, cy, u, [CLOSE]); k.restore();
};
