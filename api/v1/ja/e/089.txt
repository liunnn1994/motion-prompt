// ローディング画面 / Preloader
// パック: lp (LP・Webサイトの演出) · element 17 · カテゴリ: 読み込み・移動・動き続ける演出
// 何が起きるか: 濃紺の画面でロゴの下の数字が0%から100%まで進み、終わると画面が上へ抜けて、下のページがふわっと現れる。
// 動きの仕様:
//   - 表示域いっぱいの濃紺の画面に、ロゴ・数字（28px）・線（180×4）
//   - 0.25〜2.1秒で 0→100%（easeOut 3乗・整数に丸める）。線も同じ割合で朱色に伸びる
//   - 2.3〜2.85秒に画面ごと上へ168px抜ける（easeInOut 3乗）
//   - ページの見出し・写真は 2.35〜2.95秒に 不透明度 0→1、14px下から上がる（easeOut 3乗）
//   - 4.25〜4.75秒に画面が上から下りてきて、0%の状態に戻る（次の読み込み）
// 使いどころ: 写真や動画の多いサイトで、読み込みの待ち時間をブランドの見せ場にしたいとき。
// タグ: preloader, loading-screen, loader, percentage, intro, splash, curtain
// サイズ: 539x208 px · 1周: 4.8 s
// Needs: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// 動く見本: https://motion-prompt.2077tech.com/#089
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 17
// 17 ローディング画面（数字が100%まで進むと、幕が上がってページが現れる）
CELLS[16] = (k, w, h, u) => {
  frame(k);
  const n = Math.round(100 * eOut(P(u, 0.25, 2.1)));
  const cy = u < 4.25 ? -VH * eIO(P(u, 2.3, 2.85)) : -VH * (1 - eIO(P(u, 4.25, 4.75)));
  const show = eOut(P(u, 2.35, 2.95));
  view(k, () => {
    header(k, 0);
    k.save(); k.globalAlpha = show; k.translate(0, 14 * (1 - show));
    txt(k, 'ようこそ。', 34, 82, `800 26px ${JP}`, C.ink); bars(k, 34, 98, 200, 3); btn(k, 34, 136, 96, 24, '始める');
    pic(k, 270, 46, 214, 112, 0, 8); k.restore();
    if (cy > -VH) {
      k.save(); k.translate(0, cy); k.fillStyle = C.ink; k.fillRect(0, 0, VW, VH);
      logo(k, VW / 2 - 40, 56, '#fff');
      txt(k, (u >= 4.25 ? 0 : n) + '%', VW / 2, 98, `800 28px ${EN}`, '#fff', 'center', 'middle');
      k.fillStyle = 'rgba(255,255,255,.15)'; rr(k, VW / 2 - 90, 122, 180, 4, 2); k.fill();
      k.fillStyle = C.acc; rr(k, VW / 2 - 90, 122, 180 * (u >= 4.25 ? 0 : n / 100), 4, 2); k.fill();
      k.restore();
    }
  });
};
