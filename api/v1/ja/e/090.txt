// 遅延読み込み / Lazy loading
// パック: lp (LP・Webサイトの演出) · element 18 · カテゴリ: 読み込み・移動・動き続ける演出
// 何が起きるか: スクロールで画面に入った灰色の枠に読み込み中の印が回り、画像が届くと順にフェードインする。上の段は最初から読み込み済み。
// 動きの仕様:
//   - 画像の枠 151×92 を3列×3段。1段目は最初から表示。スクロール量: 0.35〜1.1秒で 0→110px、1.7〜2.5秒で 110→225px、3.5〜4.5秒で 225→0（easeInOut 3乗）
//   - 枠の上端が表示域の下端（168px）より上に入った時刻に読み込み開始。列ごとに0.12秒ずつずらす
//   - 読み込み中の0.45秒は、枠の中央で回る印（半径11・0.8秒で1回転）
//   - そのあと0.35秒で画像の不透明度 0→1（easeOut 3乗）。読み込み前の枠には画像の印を描く
//   - 画面の下へ出た枠は、見えない所で読み込み前に戻る
// 使いどころ: 画像の多いページで、最初の表示を速くしたいとき。
// タグ: lazy-loading, lazy-load, images, placeholder, performance, spinner, gallery
// サイズ: 539x208 px · 1周: 4.8 s
// Needs: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// 動く見本: https://motion-prompt.2077tech.com/#090
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 18
// 18 遅延読み込み（画面に入った画像だけ、その時点で読み込まれて表示される）
CELLS[17] = (k, w, h, u) => {
  frame(k);
  const SY = t => steps([[0.35, 0], [1.1, 110], [1.7, 110], [2.5, 225], [3.5, 225], [4.5, 0]], t), y = SY(u);
  const COLX = [15, 180, 345], IW = 151, IH = 92, ROWS = [12, 168, 276], WAIT = 0.45, FADE = 0.35;
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, '作品集', 15, 140, `800 18px ${JP}`, C.ink); bars(k, 120, 128, 200, 1);
    ROWS.forEach((top, r) => COLX.forEach((x, c) => {
      const t0 = r === 0 ? -1e9 : enteredAt(SY, top, VH, u), age = t0 === null ? -1 : u - t0 - c * 0.12;
      k.fillStyle = C.skel; rr(k, x, top, IW, IH, 6); k.fill();
      if (age < WAIT + FADE) {   // まだの画像: 灰色の枠と画像の印
        k.save(); k.translate(x + IW / 2, top + IH / 2); k.strokeStyle = '#B9C1CF'; k.lineWidth = 2; k.lineJoin = 'round';
        if (age >= 0 && age < WAIT) spinner(k, 0, 0, 11, u, C.blue);
        else { rr(k, -15, -11, 30, 22, 3); k.stroke(); k.beginPath(); k.moveTo(-11, 7); k.lineTo(-3, -1); k.lineTo(3, 5); k.lineTo(7, 1); k.lineTo(11, 6); k.stroke(); }
        k.restore();
      }
      if (age >= WAIT) { k.save(); k.globalAlpha = eOut(P(age, WAIT, WAIT + FADE)); pic(k, x, top, IW, IH, (r + c) % 3, 6); k.restore(); }
    }));
    k.restore();
    sbar(k, y / 225, 0.45);
  });
};
