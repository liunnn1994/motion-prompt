// ページ遷移アニメーション / Page transition
// パック: lp (LP・Webサイトの演出) · element 20 · カテゴリ: 読み込み・移動・動き続ける演出
// 何が起きるか: 作品の一覧でカードを押すと朱色の幕が左から画面を覆い、URLが変わって幕が右へ抜けると詳しいページになっている。「一覧へ」で同じように戻る。
// 動きの仕様:
//   - 0.8秒にカードをクリック。0.9〜1.3秒に幕（朱色・表示域いっぱい・中央にロゴ）が左から入って全体を覆う（easeInOut 3乗）
//   - 1.3秒、覆われている間にページとアドレスバーのURL（/works → /works/02）を切り替える
//   - 1.4〜1.85秒に幕が右へ抜ける（easeInOut 3乗）
//   - 3.0秒に「← 一覧へ」をクリック。3.1〜3.5秒に入り、3.5秒に切り替え、3.6〜4.05秒に抜ける
//   - ヘッダーも含めて表示域全体を覆う
// 使いどころ: ブランドサイトやポートフォリオで、ページを移るときの間をつなぎたいとき。
// タグ: page-transition, transition, view-transitions, navigation, wipe, barba, spa
// サイズ: 539x208 px · 1周: 4.8 s
// Needs: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// 動く見本: https://motion-prompt.2077tech.com/#092
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 20
// 20 ページ遷移アニメーション（リンクを押すと、色の幕が横切って次のページに切り替わる）
CELLS[19] = (k, w, h, u) => {
  frame(k, u >= 1.3 && u < 3.5 ? 'example.com/works/02' : 'example.com/works');
  const CLK = [0.8, 3.0];
  const cover = t0 => eIO(P(u, t0, t0 + 0.4)), uncover = t0 => eIO(P(u, t0, t0 + 0.45));
  // 幕の左端: 入る −VW→0、出る 0→VW
  const xs = u < 1.4 ? -VW * (1 - cover(0.9)) : u < 3.1 ? VW * uncover(1.4) : u < 3.6 ? -VW * (1 - cover(3.1)) : VW * uncover(3.6);
  const pageB = u >= 1.3 && u < 3.5;
  const [cx, cy] = path([[0, 420, 150], [0.65, 255, 100], [2.6, 255, 100], [2.85, 46, 50], [3.3, 46, 50], [4.6, 420, 150], [4.8, 420, 150]], u);
  view(k, () => {
    header(k, 0);
    if (!pageB) {
      txt(k, 'WORKS', 34, 62, `800 18px ${EN}`, C.ink, 'left', 'alphabetic', '2px');
      for (let i = 0; i < 3; i++) { const x = 34 + i * 152; pic(k, x, 74, 136, 60, i, 6); bars(k, x, 142, 100, 1, C.ink, 7); }
    } else {
      txt(k, '← 一覧へ', 34, 55, `700 12px ${JP}`, C.sub, 'left', 'middle');
      pic(k, 34, 68, 240, 90, 1, 8);
      txt(k, 'Project 02', 296, 96, `800 22px ${EN}`, C.ink); bars(k, 296, 110, 180, 3);
    }
    if (xs > -VW && xs < VW) { k.fillStyle = C.acc; k.fillRect(xs, 0, VW, VH); logo(k, xs + VW / 2 - 40, VH / 2, '#fff'); }
    click(k, cx, cy, u, CLK);
  });
};
