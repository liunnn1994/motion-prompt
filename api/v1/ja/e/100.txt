// スプリットライティング / Split lighting
// パック: photo (写真の表現・光と画風) · element 04 · カテゴリ: 光の当て方・時間帯
// 何が起きるか: 光が正面寄り（20°）から真横（90°）へ回り、頭の影が三日月から半分へ広がって、顔が鼻筋の位置で縦に明暗2つに分かれる。
// 動きの仕様:
//   - 場面①: 地平線（y150）の前に胸から上の人物（頭の中心 x270, y78・半径29px、顔・髪なし）。背景は一段暗い場面（空 #E7E5E1）へ
//   - 光: 画面の左から。角度 20° → 90°（右下の配置図と右上の表示「90° · ↑0°」）
//   - 頭: 明暗の境 x = 270 + cos(角度)·√(29² − dy²) より右が影 #33405A、左が光 #FFC75A。90°で中心線の真っすぐな境
//   - 胴（半幅56px）: 境 x = 270 + cos(角度)·56 より右が影、左端の帯が光
//   - 時間: 0.4〜1.4秒で効果が入り、4.0秒まで保ち、4.8秒までに元へ戻る（1周5秒、3次のイーズインアウト（cubic-bezier(0.65,0,0.35,1) 相当））
// 使いどころ: 強さや緊張感のある人物写真にしたいとき。
// タグ: split lighting, side light, portrait lighting, half face, dramatic, chiaroscuro
// サイズ: 539x208 px · 1周: 5 s
// Needs: kit photo -> https://motion-prompt.2077tech.com/api/v1/kit/photo.js
// 動く見本: https://motion-prompt.2077tech.com/#100
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 04
// スプリットライティング: 光が真横（90°）へ回り、顔の右半分が影に沈む
CELLS[3] = cell((k, u) => {
  const a = env(u), az = lerp(0.35, Math.PI / 2, a);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a), W = torsoW(PR);
  bust(k, PX, PY, PR, BASE.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, PR, 0, TAU); kk.fill();
    headShade(kk, PX, PY, PR, az, 1, sh);
    torso(kk, PX, PY, PR, PX - W, PX - Math.cos(az) * 0.4 * W - 0.4 * W, lit);
    torso(kk, PX, PY, PR, PX + Math.cos(az) * W, PX + W, sh);
  });
  plan(k, [[az, a]]);
  hud(k, `${Math.round(az * 180 / Math.PI)}° · ↑0°`);
});
