// ボケ / Bokeh
// パック: photo (写真の表現・光と画風) · element 09 · カテゴリ: レンズ・カメラの効果
// 何が起きるか: 絞りを開けていくと、ピントの合った人物はそのまま、背景の建物はぼけ、街灯と遠くの灯りが大きく薄い光の丸に広がる。
// 動きの仕様:
//   - 場面②: 街角。奥に建物の列・木・街灯（x114, 292, 494）、道は y132〜162、人物は左手前（足元 x176, y200・身長124px）。背景は一段暗い場面（空 #E7E5E1）へ
//   - 背景（人物以外）のぼけ: 0 → 6px
//   - 灯り: 街灯3つ（大きさ1.25）と、横45pxおきの12個（高さ y44〜114・大きさ0.6〜1.2、固定シード）。丸の半径 = (2.4 + 15 × 効果の量) × 大きさ
//   - 丸: 不透明度 100% → 55%、街灯はアンバー #FFC75A・遠くの灯りは #FFE3A3、縁にアンバーの1.2pxの線
//   - 人物はくっきりのまま。右上の絞り f/16 → f/1.4（8段）
//   - 時間: 0.4〜1.4秒で効果が入り、4.0秒まで保ち、4.8秒までに元へ戻る（1周5秒、3次のイーズインアウト（cubic-bezier(0.65,0,0.35,1) 相当））
// 使いどころ: 人物や商品を背景から浮かせ、夜や祝祭のやわらかな雰囲気を出したいとき。
// タグ: bokeh, shallow depth of field, background blur, wide aperture, f/1.4, out of focus, lights
// サイズ: 539x208 px · 1周: 5 s
// Needs: kit photo -> https://motion-prompt.2077tech.com/api/v1/kit/photo.js
// 動く見本: https://motion-prompt.2077tech.com/#105
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 09
// ボケ: 絞りを開けると、人物はくっきりのまま、背景の灯りが大きな丸になってにじむ
CELLS[8] = cell((k, u) => {
  const a = env(u), p = pal(BASE, DIM, a);
  const [B, kb] = layer('bokeh'); stageStreet(kb, p, { person: false });
  const [Bb, kbb] = layer('bokeh-b'); kbb.filter = `blur(${(6 * a).toFixed(2)}px)`; kbb.drawImage(B, 0, 0); kbb.filter = 'none';
  k.drawImage(B, 0, 0); k.drawImage(Bb, 0, 0);
  // 灯り: 街灯3つと、窓や遠くの灯り12個（横に45pxおき・高さと大きさは固定シード）。絞りを開くほど丸が大きく薄くなる
  const pts = once('bokeh-pts', () => {
    const o = LAMPS.map(x => [x + 9, ROAD - 53, 1.25, 0]), r = rng(9);
    for (let i = 0; i < 12; i++) o.push([22 + i * 45 + r() * 20, 44 + r() * 70, 0.6 + r() * 0.6, 1]);
    return o;
  });
  k.save();
  for (const [x, y, s, w] of pts) {
    const r = (2.4 + 15 * a) * s;
    k.globalAlpha = lerp(1, 0.55, a); k.fillStyle = w ? BASE.glow : BASE.lit; k.beginPath(); k.arc(x, y, r, 0, TAU); k.fill();
    k.globalAlpha = 0.7 * a; k.strokeStyle = BASE.lit; k.lineWidth = 1.2; k.stroke();
  }
  k.restore();
  figure(pen(k), FIG[0], FIG[1], FIG[2], p.person);
  const stops = ['f/16', 'f/11', 'f/8', 'f/5.6', 'f/4', 'f/2.8', 'f/2', 'f/1.4'];
  hud(k, stops[Math.round(a * 7)]);
});
