// ティルトシフト / Tilt-shift
// パック: photo (写真の表現・光と画風) · element 10 · カテゴリ: レンズ・カメラの効果
// 何が起きるか: 街の上下の帯がぼけていき、道と車と小さな人たちの並ぶ中央の帯だけにピントが残って、模型の街のように見える。
// 動きの仕様:
//   - 街角（奥に建物の列・木・街灯、道は y132〜162）から手前の人物を外し、歩道に身長23〜26pxの小さな人4人（足元 y172）を置く。車は5秒で x−60 → 600 を走る
//   - ピントの帯: y66〜176 はくっきり
//   - ぼけ: y0〜42 と y190〜208 は 0 → 7px、y42〜66 と y176〜190 は 0 → 3px
//   - 帯の境に破線（6px線・5px間・#8E8E93・75%）
//   - 時間: 0.4〜1.4秒で効果が入り、4.0秒まで保ち、4.8秒までに元へ戻る（1周5秒、3次のイーズインアウト（cubic-bezier(0.65,0,0.35,1) 相当））
// 使いどころ: 街や行事を、かわいらしい模型の世界のように見せたいとき。
// タグ: tilt-shift, miniature, miniature effect, fake miniature, selective focus, toy town, focus band
// サイズ: 539x208 px · 1周: 5 s
// Needs: kit photo -> https://motion-prompt.2077tech.com/api/v1/kit/photo.js
// 動く見本: https://motion-prompt.2077tech.com/#106
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 10
// ティルトシフト: 上下の帯がぼけて中央の帯だけピントが合い、街と小さな人と車が模型のように見える
CELLS[9] = cell((k, u) => {
  const a = env(u), carX = lerp(-60, 600, u / CYC);
  const minis = D => { [[196, 26], [322, 26], [340, 23], [452, 25]].forEach(([x, h]) => figure(D, x, 172, h, BASE.person)); };
  const [S, ks] = layer('tilt'); stageStreet(ks, BASE, { car: carX, person: false, after: minis });
  const [B1, k1] = layer('tilt-1'); k1.filter = `blur(${(3 * a).toFixed(2)}px)`; k1.drawImage(S, 0, 0); k1.filter = 'none';
  const [B2, k2] = layer('tilt-2'); k2.filter = `blur(${(7 * a).toFixed(2)}px)`; k2.drawImage(S, 0, 0); k2.filter = 'none';
  k.drawImage(S, 0, 0);
  const band = (img, y0, y1) => { k.save(); k.beginPath(); k.rect(0, y0, CW, y1 - y0); k.clip(); k.drawImage(img, 0, 0); k.restore(); };
  band(B2, 0, 42); band(B1, 42, 66); band(B1, 176, 190); band(B2, 190, CH);
  // ピントの帯の目印（写真アプリの線）
  k.save(); k.globalAlpha = 0.75 * a; k.strokeStyle = '#8E8E93'; k.lineWidth = 1.4; k.setLineDash([6, 5]);
  k.beginPath(); k.moveTo(0, 66); k.lineTo(CW, 66); k.moveTo(0, 176); k.lineTo(CW, 176); k.stroke(); k.setLineDash([]);
  k.restore();
});
