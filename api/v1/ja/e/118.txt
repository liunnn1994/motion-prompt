// フィルムグレイン / Film grain
// パック: photo (写真の表現・光と画風) · element 22 · カテゴリ: 仕上げ・画風
// 何が起きるか: 同じ風景の全体に細かい粒が乗り、色が少し暖かいクリーム色に寄る。
// 動きの仕様:
//   - 場面③: 左に胸像（頭の中心 x140, y82・半径28px）、右に丘・家・木・太陽（x432, y54・半径15px）。
//   - 粒: 2600個（固定シード）、大きさ 0.8〜1.7px の四角。55%が暗い #3A3530、45%が白
//   - 粒の濃さ: 16〜38% × 効果の量。粒の位置は動かさない（ちらつかせない）
//   - 色: すべての色を #F2DDBE へ16%寄せる
//   - 時間: 0.4〜1.4秒で効果が入り、4.0秒まで保ち、4.8秒までに元へ戻る（1周5秒、3次のイーズインアウト（cubic-bezier(0.65,0,0.35,1) 相当））
// 使いどころ: デジタルのつるっとした感じを消して、フィルムの手触りを出したいとき。
// タグ: film grain, grain, noise, analog, 35mm film, texture, warm tone
// サイズ: 539x208 px · 1周: 5 s
// Needs: kit photo -> https://motion-prompt.2077tech.com/api/v1/kit/photo.js
// 動く見本: https://motion-prompt.2077tech.com/#118
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 22
// フィルムグレイン: 細かい粒子が全体に乗り、全体が少し暖かい色に寄る
CELLS[21] = cell((k, u) => {
  const a = env(u);
  const FILM = mapPal(BASE, c => mixc(c, '#F2DDBE', 0.16));
  const p = pal(BASE, FILM, a);
  const G = once('grain', () => { const r = rng(22), o = []; for (let i = 0; i < 2600; i++) o.push([r() * CW, r() * CH, 0.8 + r() * 0.9, r() < 0.55 ? 0 : 1, 0.16 + r() * 0.22]); return o; });
  const grain = kk => { kk.save(); for (const [x, y, s, d, al] of G) { kk.globalAlpha = al * a; kk.fillStyle = d ? '#FFFFFF' : '#3A3530'; kk.fillRect(x, y, s, s); } kk.restore(); };
  stageLand(k, p); grain(k);
});
