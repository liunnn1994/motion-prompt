// 二重露光（多重露光） / Double exposure
// パック: photo (写真の表現・光と画風) · element 24 · カテゴリ: 仕上げ・画風
// 何が起きるか: 背景が白く飛び、胸像の形の中にだけ、縮めた丘・家・木・太陽の風景が写り込む。
// 動きの仕様:
//   - 場面③: 左に胸像（頭の中心 x140, y82・半径28px）、右に丘・家・木・太陽（x432, y54・半径15px）。
//   - 背景: 胸像の外は白 #FFFFFF
//   - 胸像の中: 同じ風景を0.7倍にして、太陽が頭の右上、丘の稜線が首のあたりに来るように置く
//   - 中の色: 空 #D3DBE6、丘 #9AA8BD・#76869F、地面 #55647E、木 #3F4C66
//   - 切り替え: 元の写真の上に、二重露光の絵を効果の量で重ねる
//   - 時間: 0.4〜1.4秒で効果が入り、4.0秒まで保ち、4.8秒までに元へ戻る（1周5秒、3次のイーズインアウト（cubic-bezier(0.65,0,0.35,1) 相当））
// 使いどころ: 人物と場所・記憶・内面を1枚で結びつけたい、ポスターやジャケットで。
// タグ: double exposure, multiple exposure, blend, silhouette, surreal, composite
// サイズ: 539x208 px · 1周: 5 s
// Needs: kit photo -> https://motion-prompt.2077tech.com/api/v1/kit/photo.js
// 動く見本: https://motion-prompt.2077tech.com/#120
//
// cells[i](k, w, h, u) … k = CanvasRenderingContext2D（呼ぶ側でクリアと save/restore 済み）、w・h = サイズ、u = 秒（0 以上 1周 未満）。u だけで絵が決まる。
// ここから要素のコード。CELLS や定数・補助関数は kit にある。

// @element 24
// 二重露光: 背景が白く飛び、胸像の形の中にだけ丘と家と木と太陽が写り込む
CELLS[23] = cell((k, u) => {
  const a = env(u);
  const DX = Object.assign({}, BASE, { sky: '#D3DBE6', sky2: '#D3DBE6', far: '#9AA8BD', mid: '#76869F', ground: '#55647E', bld: '#E8ECF2', edge: '#55647E', win: '#9AA8BD', winOff: '#9AA8BD', tree: '#3F4C66', trunk: '#3F4C66' });
  const draw = kk => {
    kk.fillStyle = '#FFFFFF'; kk.fillRect(0, 0, CW, CH);
    kk.save(); bustPath(kk, QX, QY, QR); kk.clip();
    // 風景を0.7倍にして、太陽が頭の右上、丘の稜線が首のあたりに来るように置く
    kk.translate(QX, QY); kk.scale(0.7, 0.7); kk.translate(-415, -77);
    kk.fillStyle = DX.sky; kk.fillRect(0, -200, CW * 2, CH * 3);
    stageLand(kk, DX, { person: false });
    kk.fillStyle = DX.ground; kk.fillRect(-CW, 200, CW * 3, CH * 3);
    kk.restore();
  };
  if (a >= 0.999) { draw(k); return; }   // 効果が入りきったら直接描く（拡大表示でも輪郭がくっきり）
  stageLand(k, BASE);
  if (a < 0.002) return;
  const [L, kl] = layer('dx'); draw(kl);   // 入り・戻りの途中は、別の面に描いて重ねる
  k.save(); k.globalAlpha = a; k.drawImage(L, 0, 0); k.restore();
});
