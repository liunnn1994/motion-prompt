// 交错入场 / Stagger
// 素材包: motion (动态图形) · element 02 · 分类: 入场与文字动效
// 示例内容: 柱状图从左到右依次升起，每根柱子略有延迟。
// 动画规格:
//   - 每 2.35 秒一轮。7 根柱子，宽 44、间距 20，高度依次为 52、84、68、112、92、136、158，依次循环使用蓝、珊瑚、黄、薄荷绿。
//   - 第 i 根柱子（从 0 编号）在 0.15 + 0.12·i 秒开始，以底边为锚点，用 0.5 秒按 backOut（s = 2.6）从 0 长到完整高度，越过目标后回落。
//   - 白色圆点：半径 6.5，位于柱顶上方 16px，比对应柱子晚 0.34 秒出现，弹簧公式为 1 − e^(−7t)·cos(16t)。
//   - 退场：柱子从 1.78 + 0.045·i 秒起，用 0.26 秒三次缓入收缩；圆点先从 1.70 + 0.045·i 秒起，用 0.2 秒消失。
// 适用场景: 适合列表、图表和卡片的顺序呈现，引导视线移动。
// 标签: stagger, sequence, delay, cascade, bar-chart, list, entrance
// 尺寸: 539x208 px · 周期: 4.7 s
// 依赖: kit motion -> https://motion-prompt.2077tech.com/api/v1/kit/motion.js
// 动态示例: https://motion-prompt.2077tech.com/#002
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（由调用方清空画布并执行 save/restore），w、h = 尺寸，u = 秒数，范围为 [0, cycle)。画面仅由 u 决定。
// 以下为素材代码。CELLS、常量及辅助函数位于 kit 文件中。

// @element 02
// 02 スタガー（グラフの棒が、少しずつ時間をずらして順に立ち上がる）
CELLS[1] = (k, w, h, u) => {
  stage(k);
  const N = 7, BW = 44, G = 20, x0 = (w - (N * BW + (N - 1) * G)) / 2, base = h - 30;
  const HS = [52, 84, 68, 112, 92, 136, 158];
  k.fillStyle = C.navy3; rr(k, x0 - 18, base, N * BW + (N - 1) * G + 36, 5, 2.5); k.fill();
  const lu = mod(u, CYC / 2);
  for (let i = 0; i < N; i++) {
    const t0 = 0.15 + i * 0.12, g = backOut(P(lu, t0, t0 + 0.5), 2.6) * (1 - eIn(P(lu, 1.78 + i * 0.045, 2.04 + i * 0.045)));
    if (g <= 0.001) continue;
    const hh = HS[i] * g, x = x0 + i * (BW + G);
    k.fillStyle = PAL[i % 4]; rr(k, x, base - hh, BW, hh, [12, 12, 3, 3]); k.fill();
    // 棒の上の点（さらに遅れて出る）
    const d = spring(lu - t0 - 0.34, 16, 7) * (1 - eIn(P(lu, 1.7 + i * 0.045, 1.9 + i * 0.045)));
    if (d > 0.01) { k.fillStyle = '#fff'; k.beginPath(); k.arc(x + BW / 2, base - hh - 16, 6.5 * d, 0, 7); k.fill(); }
  }
};
