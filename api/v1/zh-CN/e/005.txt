// 素材包: motion (动态图形) · element 05 · 分类: 入场与文字动效
// 示例内容: 标题从一条细线标记的不可见边缘后方滑出。
// 动画规格:
//   - 每 2.35 秒一轮。分隔线为珊瑚色，380 × 6，y = 124；0.1–0.5 秒从中心展开，使用三次缓入缓出，2.05–2.3 秒收起。
//   - 上方文字：白色、粗体 98px，仅显示在线条上方 y = 8–119 的裁剪区内；0.35–1.0 秒从下方 120px 处上移到位，使用五次缓出。
//   - 下方文字：黄色、32px，仅显示在线条下方 y = 129–181 的裁剪区内；0.55–1.15 秒从上方 60px 处下移到位，使用五次缓出。
//   - 退场：上方文字在 1.7–2.05 秒上移 150px，使用 backIn（s = 1.2），先轻微反向移动；下方文字在 1.75–2.05 秒下移 70px，使用三次缓入。
//   - 不透明度始终不变；文字通过移出裁剪区域而隐藏。
// 适用场景: 适合干净利落的标题入场。
// 标签: mask, reveal, clip, wipe, title, text, slide-up, headline
// 尺寸: 539x208 px · 周期: 4.7 s
// 依赖: kit motion -> https://motion-prompt.2077tech.com/api/v1/kit/motion.js
// 动态示例: https://motion-prompt.2077tech.com/#005
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（由调用方清空画布并执行 save/restore），w、h = 尺寸，u = 秒数，范围为 [0, cycle)。画面仅由 u 决定。
// 以下为素材代码。CELLS、常量及辅助函数位于 kit 文件中。

// @element 05
// 05 マスクリビール（見えない枠＝線の向こうから、見出しがせり上がる）
CELLS[4] = (k, w, h, u) => {
  stage(k);
  const T = CYC / 2, lu = mod(u, T), X = w / 2, LY = 124;
  const lg = eIO(P(lu, 0.1, 0.5)) * (1 - eIO(P(lu, 2.05, 2.3)));
  // 上の段：線の下から上へ
  const up = (1 - eOut5(P(lu, 0.35, 1.0))) * 120 + backIn(P(lu, 1.7, 2.05), 1.2) * -150;
  k.save(); k.beginPath(); k.rect(0, 8, w, LY - 8 - 5); k.clip();
  txt(k, 'Hello!', X, LY - 14 + up, `800 98px ${DISP}`, '#fff', 'center', 'alphabetic', '1px');
  k.restore();
  // 下の段：線の上から下へ
  const dn = (1 - eOut5(P(lu, 0.55, 1.15))) * -60 + eIn(P(lu, 1.75, 2.05)) * 70;
  k.save(); k.beginPath(); k.rect(0, LY + 5, w, 52); k.clip();
  txt(k, 'はじめまして', X, LY + 44 + dn, `700 32px ${JP}`, C.yel, 'center', 'alphabetic', '6px');
  k.restore();
  if (lg > 0.005) { k.fillStyle = C.coral; rr(k, X - 190 * lg, LY - 3, 380 * lg, 6, 3); k.fill(); }
};
