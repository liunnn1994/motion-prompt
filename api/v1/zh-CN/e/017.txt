// 蒙版转场 / Mask transition
// 素材包: motion (动态图形) · element 17 · 分类: 转场与视觉强调
// 示例内容: 圆形或星形窗口展开，露出其中的下一场景。
// 动画规格:
//   - 白天→夜晚→白天，每段 2.35 秒；每半周期的 0.45–1.6 秒转场，时长 1.15 秒，使用三次缓入缓出。
//   - 下一场景绘制在当前场景上方，并由形状裁剪，形状扩大直至覆盖整幅画面。
//   - 白天转夜晚：圆形从太阳中心（112, 104）展开，半径 0→470。
//   - 夜晚转白天：五角星从月亮中心（427, 104）展开，外半径 0→1150，扩大的同时旋转 1.2rad，约 69°。
//   - 形状扩大时，边缘带 7px 宽的白色描边。
// 适用场景: 适合从太阳、标志或按钮等画面元素开始转场。
// 标签: mask, transition, iris, wipe, shape-wipe, reveal, scene-change
// 尺寸: 539x208 px · 周期: 4.7 s
// 依赖: kit motion -> https://motion-prompt.2077tech.com/api/v1/kit/motion.js
// 动态示例: https://motion-prompt.2077tech.com/#017
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（由调用方清空画布并执行 save/restore），w、h = 尺寸，u = 秒数，范围为 [0, cycle)。画面仅由 u 决定。
// 以下为素材代码。CELLS、常量及辅助函数位于 kit 文件中。

// @element 17
// 17 マスクトランジション（丸や星の形の「窓」が広がり、その中に次の画面が見える）
CELLS[16] = abCell((k, w, h, u, a, b) => {
  const p = eIO(P(u, TA - 0.05, TB + 0.1));
  SCR[a](k, w, h);
  if (p > 0) {
    const c = a === 0 ? SUN : MOON, shape = () => { if (a === 0) { k.beginPath(); k.arc(c[0], c[1], lerp(0, 470, p), 0, 7); } else star(k, c[0], c[1], lerp(0, 1150, p), p * 1.2, 0.5); };
    k.save(); shape(); k.clip(); SCR[b](k, w, h); k.restore();
    if (p < 1) { k.strokeStyle = '#fff'; k.lineWidth = 7; k.lineJoin = 'round'; shape(); k.stroke(); }
  }
});
