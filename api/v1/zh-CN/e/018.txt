// 图形转场 / Shape transition
// 素材包: motion (动态图形) · element 18 · 分类: 转场与视觉强调
// 示例内容: 彩色色带扫过并覆盖画面，离开后露出下一场景。
// 动画规格:
//   - 白天→夜晚→白天，每段 2.35 秒。4 条高 52 的水平色带，分别为蓝、珊瑚、黄、薄荷绿，端部斜切偏移 80px。
//   - 覆盖阶段为 0.38–1.0 秒：色带前缘从一侧移动到另一侧；每条依次延迟阶段时长的 14%，各用阶段时长的 58% 完成三次缓入缓出。
//   - 1.0 秒完全遮住画面时，将下方场景替换为下一场景。
//   - 揭示阶段为 1.12–1.75 秒：色带后缘按同样延迟和缓动跟进，露出下一场景。
//   - 白天转夜晚时从左向右，夜晚转白天时方向镜像。
// 适用场景: 适合使用品牌色进行章节或主题切换。
// 标签: shape-transition, transition, wipe, bands, stripes, cover, scene-change
// 尺寸: 539x208 px · 周期: 4.7 s
// 依赖: kit motion -> https://motion-prompt.2077tech.com/api/v1/kit/motion.js
// 动态示例: https://motion-prompt.2077tech.com/#018
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（由调用方清空画布并执行 save/restore），w、h = 尺寸，u = 秒数，范围为 [0, cycle)。画面仅由 u 决定。
// 以下为素材代码。CELLS、常量及辅助函数位于 kit 文件中。

// @element 18
// 18 シェイプトランジション（色のついた帯が画面を覆い隠し、抜けると次の画面になっている）
CELLS[17] = abCell((k, w, h, u, a, b) => {
  const p1 = P(u, TA - 0.12, TA + 0.5), p2 = P(u, TA + 0.62, TA + 1.25), N = 4, sk = 80;
  SCR[p1 >= 1 ? b : a](k, w, h);
  const cols = [C.blue, C.coral, C.yel, C.mint];
  for (let i = 0; i < N; i++) {
    const d = i * 0.14, e1 = eIO(clamp((p1 - d) / (1 - 0.42))), e2 = eIO(clamp((p2 - d) / (1 - 0.42)));
    if (e1 <= 0 || e2 >= 1) continue;
    const bh = h / N + 1, y = i * h / N, span = w + sk + 20, dir = a === 0 ? 1 : -1;
    let xa = -sk - 10 + e2 * span, xb = -sk - 10 + e1 * span;
    k.save(); if (dir < 0) { k.translate(w, 0); k.scale(-1, 1); }
    k.fillStyle = cols[i]; k.beginPath(); k.moveTo(xa + sk, y); k.lineTo(xb + sk, y); k.lineTo(xb, y + bh); k.lineTo(xa, y + bh); k.closePath(); k.fill(); k.restore();
  }
});
