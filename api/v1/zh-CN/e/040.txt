// 定格 / Freeze frame
// 素材包: camera (镜头运动与转场) · element 16 · 分类: 拍摄手法与时间控制
// 示例内容: 跳跃者到达最高点时画面暂停，1.4 秒后从同一帧继续。
// 动画规格:
//   - 相机完全固定，高度 1.3m；暂停的是素材时间。
//   - 定格从 1.6 秒持续至 3.0 秒，共 1.4 秒；素材时间在 0–1.6 秒正常前进，定格时停在 1.6 秒，之后按 u−1.4 继续。
//   - 主体：前方 7m 的人物在素材时间 0.6–2.6 秒沿抛物线跳跃，最高 1.1m；定格恰好落在进度 50% 的最高点。
//   - 9 片落叶也在定格时暂停，下落速度为第 13 项慢动作示例的 1.5 倍。
//   - 定格标记：5px 蓝色边框，底部中央的 68×40px 白色圆角框内有两条蓝色暂停竖线；淡入和淡出各用 0.12 秒。
//   - 仅在定格期间，右上角显示“FREEZE”；无交叉淡化重置。
// 适用场景: 适合强调决定性瞬间，或停下来添加姓名和标题。
// 标签: freeze frame, still frame, pause, hold, stop motion, highlight
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit camera -> https://motion-prompt.2077tech.com/api/v1/kit/camera.js
// 动态示例: https://motion-prompt.2077tech.com/#040
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（由调用方清空画布并执行 save/restore），w、h = 尺寸，u = 秒数，范围为 [0, cycle)。画面仅由 u 决定。
// 以下为素材代码。CELLS、常量及辅助函数位于 kit 文件中。

// @element 16
// 16 フリーズフレーム
CELLS[15] = shot((k, w, h, u) => {
  // 跳躍の頂点で 1.4 秒止める
  const tj = u < 1.6 ? u : u < 3.0 ? 1.6 : u - 1.4;   // 実時間→映像時間
  const q = P(tj, 0.6, 2.6);
  const frozen = u >= 1.6 && u < 3.0;
  world(k, w, h, cam({ y: 1.3 }), { person: { z: 7, y: jumpY(q), armUp: q > 0 && q < 1 }, extra: leaves(tj, 1.5) });
  if (frozen) {
    const a = Math.min(1, (u - 1.6) / 0.12, (3.0 - u) / 0.12);
    k.save(); k.globalAlpha = a; k.strokeStyle = C.blue; k.lineWidth = 5; k.strokeRect(2.5, 2.5, w - 5, h - 5);
    k.fillStyle = 'rgba(255,255,255,.88)'; rr(k, w / 2 - 34, h - 58, 68, 40, 8); k.fill();
    k.fillStyle = C.blue; k.fillRect(w / 2 - 12, h - 49, 8, 22); k.fillRect(w / 2 + 4, h - 49, 8, 22); k.restore();
  }
  return frozen ? 'FREEZE' : '';
});
