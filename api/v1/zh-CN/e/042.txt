// 匹配剪辑 / Match cut
// 素材包: camera (镜头运动与转场) · element 18 · 分类: 剪辑与转场
// 示例内容: 抛出的球在下一镜头变为同位置、同大小的太阳，用相似形状连接两个场景。
// 动画规格:
//   - 2.3 秒处硬切，过渡时长 0 帧；切换前后蓝色圆形位置和大小一致，位于中心右侧 120px、上方 46px，半径 17px。
//   - 前一镜头为 0–2.3 秒，固定机位、高度 1.4m；人物位于左侧 1.2m、前方 7m，从 0.5 秒起抛球，使用三次缓出，约为 cubic-bezier(0.33,1,0.68,1)。
//   - 球从人物头上方 2.3m 高处移向匹配点，半径 5px→17px，路径向上拱起最多 30px。
//   - 后一镜头为 2.3–4.7 秒，切到无建筑的另一地点，机位高 1.2m；同位置蓝圆变为太阳，并在 2.4 秒内下沉 14px。
//   - 匹配点以半径 26px 的蓝色虚线圆标记，在切点前后各 0.7 秒内淡入、淡出。
//   - 右上角显示“SHOT 1”→“SHOT 2”，4.7 秒时直接切回“SHOT 1”。
// 适用场景: 适合通过形状或动作连接时间、地点相距较远的场景。
// 标签: match cut, graphic match, shape match, cut, transition, edit
// 尺寸: 539x208 px · 周期: 9.4 s
// 依赖: kit camera -> https://motion-prompt.2077tech.com/api/v1/kit/camera.js
// 动态示例: https://motion-prompt.2077tech.com/#042
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（由调用方清空画布并执行 save/restore），w、h = 尺寸，u = 秒数，范围为 [0, cycle)。画面仅由 u 决定。
// 以下为素材代码。CELLS、常量及辅助函数位于 kit 文件中。

// @element 18
// 18 マッチカット（投げたボール → 同じ位置・大きさの太陽）
CELLS[17] = shot((k, w, h, u) => {
  const cut = 2.3;
  const BX = 120, BY = -46, BR = 17;
  if (u < cut) {
    const q = eOut(P(u, 0.5, cut));
    world(k, w, h, cam({ y: 1.4 }), {
      person: { x: -1.2, z: 7, armUp: q > 0.05 }, sun: null,
      after: (kk, cc) => { const s = proj(cc, -1.2, 2.3, 7); const x = lerp(s.x, BX, q), y = lerp(s.y, BY, q) - Math.sin(q * Math.PI) * 30, r = lerp(5, BR, q); kk.fillStyle = C.blue; kk.beginPath(); kk.arc(x, y, r, 0, 7); kk.fill(); }
    });
  } else {
    const q = P(u, cut, CYC);
    world(k, w, h, cam({ y: 1.2 }), {
      buildings: false, sun: null, trees: true,
      after: (kk) => { kk.fillStyle = C.blue; kk.beginPath(); kk.arc(BX, BY + q * 14, BR, 0, 7); kk.fill(); }
    });
  }
  // 形がそろっていることを示す点線の円
  const g = Math.max(0, 1 - Math.abs(u - cut) / 0.7);
  if (g > 0) { k.save(); k.translate(w / 2, h / 2); k.globalAlpha = g; k.strokeStyle = C.blue; k.lineWidth = 2; k.setLineDash([5, 4]); k.beginPath(); k.arc(BX, BY + (u > cut ? P(u, cut, CYC) * 14 : 0), BR + 9, 0, 7); k.stroke(); k.restore(); }
  return u < cut ? 'SHOT 1' : 'SHOT 2';
});
