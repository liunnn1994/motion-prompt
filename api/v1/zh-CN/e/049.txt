// 模态对话框 / Modal dialog
// 素材包: ui (UI 组件) · element 01 · 分类: 弹层与面板
// 示例内容: 点击“打开”后背景变暗，小窗口放大到中央；点击 × 关闭。
// 动画规格:
//   - 卡片尺寸 542×191，每轮 4.5 秒。光标于 0.95 秒到达左下角 120×36 的“打开”按钮，1.0 秒点击。
//   - 展开进度 a：1.0–1.32 秒按 easeOutCubic 从 0 增至 1。整张卡片覆盖 rgba(18,20,28) 遮罩，不透明度为 0.42·a。
//   - 对话框：白色，280×140，圆角半径 12，中心（271, 97）；阴影模糊 24、纵向偏移 8、alpha 0.25；缩放比例为 0.9 + 0.1·a，不透明度为 a。
//   - 内部包含粗体标题、两行文字、右下角蓝色 88×30 按钮和右上角关闭符号 ×。
//   - 关闭：光标在 3.0 秒前移到 ×，3.3 秒点击；3.3–3.58 秒将 a 乘以 (1 − easeInOutCubic)。
// 适用场景: 适合需要暂停当前操作处理的确认或简短表单。
// 标签: modal, modal-dialog, dialog, popup, overlay, backdrop, lightbox, confirm
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://motion-prompt.2077tech.com/api/v1/kit/ui.js
// 动态示例: https://motion-prompt.2077tech.com/#049
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（由调用方清空画布并执行 save/restore），w、h = 尺寸，u = 秒数，范围为 [0, cycle)。画面仅由 u 决定。
// 以下为素材代码。CELLS、常量及辅助函数位于 kit 文件中。

// @element 01
// 01 モーダル（Modal dialog）
CELLS[0] = ui((w, h, t) => {
  pageLines(24, 18, 494);
  fillRR(24, 92, 120, 36, 8, C.blue); text('打开', 84, 110, 16, '#fff', 600, 'center');
  fillRR(170, 92, 348, 36, 8, C.block); fillRR(24, 142, 494, 38, 8, C.block);
  const a = eo(seg(t, 1.0, 1.32)) * (1 - eio(seg(t, 3.3, 3.58)));
  if (a > 0) {
    X.fillStyle = `rgba(18,20,28,${.42 * a})`; X.fillRect(0, 0, w, h);
    X.save(); X.globalAlpha = a; X.translate(271, 97); const s = .9 + .1 * a; X.scale(s, s);
    shadowed(() => fillRR(-140, -70, 280, 140, 12, '#fff'), 24, 8, .25);
    bar(-116, -48, 120, C.ink, 11); bar(-116, -20, 232); bar(-116, -2, 180);
    fillRR(28, 26, 88, 30, 7, C.blue);
    X.strokeStyle = '#9a9896'; X.lineWidth = 2.2; X.lineCap = 'round'; X.beginPath(); X.moveTo(110, -54); X.lineTo(120, -44); X.moveTo(120, -54); X.lineTo(110, -44); X.stroke();
    X.restore();
  }
  pointer([[.3, w + 30, h + 30], [.95, 86, 114], [1.5, 86, 114], [3.0, 389, 50], [3.5, 389, 50], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
