// 抽屉面板 / Drawer (Off-canvas)
// 素材包: ui (UI 组件) · element 02 · 分类: 弹层与面板
// 示例内容: 点击“菜单”后高面板从左侧滑入，点击外部后滑回。
// 动画规格:
//   - 卡片尺寸 542×191，每轮 4.5 秒。1.0 秒点击左上角 110×34 的“菜单”按钮。
//   - 展开进度 a：1.0–1.38 秒按 easeOutCubic 从 0 增至 1。全卡片遮罩颜色 rgba(18,20,28)，不透明度 0.36·a。
//   - 抽屉：白色，宽 210、高度与卡片相同；x = −210 + 210·a，从左侧滑入；阴影模糊 22、alpha 0.25。
//   - 顶部显示半径 14 的头像和姓名，下方有 4 行菜单，行间距 32px，首行以浅蓝背景表示选中。
//   - 关闭：3.3 秒点击外部坐标（420, 120）；3.3–3.65 秒将 a 乘以 (1 − easeInOutCubic)，抽屉向左滑回。
// 适用场景: 适合在需要时展示导航或筛选，尤其适用于移动端。
// 标签: drawer, off-canvas, offcanvas, side-menu, navigation-drawer, sheet, sidebar, slide-in
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://motion-prompt.2077tech.com/api/v1/kit/ui.js
// 动态示例: https://motion-prompt.2077tech.com/#050
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（由调用方清空画布并执行 save/restore），w、h = 尺寸，u = 秒数，范围为 [0, cycle)。画面仅由 u 决定。
// 以下为素材代码。CELLS、常量及辅助函数位于 kit 文件中。

// @element 02
// 02 ドロワー（Drawer (Off-canvas)）
CELLS[1] = ui((w, h, t) => {
  strokeRR(24, 14, 110, 34, 8, C.line2, 1.5); text('菜单', 79, 31, 15, '#555', 600, 'center');
  bar(160, 26, 140, C.line2, 11); pageLines(24, 70, 494); fillRR(24, 138, 494, 42, 8, C.block);
  const a = eo(seg(t, 1.0, 1.38)) * (1 - eio(seg(t, 3.3, 3.65)));
  if (a > 0) {
    X.fillStyle = `rgba(18,20,28,${.36 * a})`; X.fillRect(0, 0, w, h);
    const pw = 210, px = -pw + pw * a;
    shadowed(() => { X.fillStyle = '#fff'; X.fillRect(px, 0, pw, h); }, 22, 0, .25);
    circle(px + 34, 32, 14, C.mb); bar(px + 58, 27, 100, C.line2, 10);
    for (let i = 0; i < 4; i++) { const y = 62 + i * 32; if (i === 0) fillRR(px + 12, y - 2, pw - 24, 28, 6, C.lb); fillRR(px + 22, y + 5, 14, 14, 3, i === 0 ? C.blue : C.line2); bar(px + 46, y + 7, 90 + (i % 2) * 30, i === 0 ? C.sb : C.line); }
  }
  pointer([[.3, w + 30, h + 30], [.95, 70, 34], [1.6, 70, 34], [3.0, 420, 120], [3.6, 420, 120], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
