// 底部面板 / Bottom sheet
// 素材包: ui (UI 组件) · element 03 · 分类: 弹层与面板
// 示例内容: 手指点击“查看详情”，面板从底部升起，下拖手柄关闭。
// 动画规格:
//   - 卡片尺寸 542×191，每轮 4.5 秒。手指由半径 17、白色边缘的半透明圆表示，1.0 秒点击底部中央 170×36 的“查看详情”按钮。
//   - 展开进度 a：1.05–1.45 秒按 easeOutCubic 从 0 增至 1；全卡片遮罩颜色 rgba(18,20,28)，不透明度 0.36·a。
//   - 底部面板：白色，高 136，上方圆角半径 18，顶边 y = 191 − 136·a；阴影模糊 22、纵向偏移 −4；顶部中央有 52×6 的拖动手柄。
//   - 内部包含粗体标题和两行内容，各带 26×26 图标。
//   - 关闭：2.75 秒手指落在手柄（271, 68），3.0–3.6 秒向下拖动；a 乘以该区间的 (1 − easeInOutCubic)，面板跟随手指移出底部。
// 适用场景: 适合移动端查看详情或选择，同时保留当前页面。
// 标签: bottom-sheet, sheet, modal-sheet, mobile, slide-up, drag-to-dismiss, panel
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://motion-prompt.2077tech.com/api/v1/kit/ui.js
// 动态示例: https://motion-prompt.2077tech.com/#051
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（由调用方清空画布并执行 save/restore），w、h = 尺寸，u = 秒数，范围为 [0, cycle)。画面仅由 u 决定。
// 以下为素材代码。CELLS、常量及辅助函数位于 kit 文件中。

// @element 03
// 03 ボトムシート（Bottom sheet）
CELLS[2] = ui((w, h, t) => {
  pageLines(24, 16, 494); fillRR(24, 76, 239, 54, 8, C.block); fillRR(279, 76, 239, 54, 8, C.block);
  fillRR(186, 144, 170, 36, 18, C.blue); text('查看详情', 271, 162, 15, '#fff', 600, 'center');
  const open = eo(seg(t, 1.05, 1.45));
  const drag = eio(seg(t, 3.0, 3.6));
  const a = open * (1 - drag);
  if (open > 0 && a > 0.001) {
    X.fillStyle = `rgba(18,20,28,${.36 * a})`; X.fillRect(0, 0, w, h);
    const sh = 136, sy = h - sh * a;
    shadowed(() => { rr(0, sy, w, sh + 20, 18); X.fillStyle = '#fff'; X.fill(); }, 22, -4, .2);
    fillRR(w / 2 - 26, sy + 10, 52, 6, 3, C.line2);
    bar(28, sy + 30, 160, C.ink, 11);
    for (let i = 0; i < 2; i++) { const y = sy + 56 + i * 36; fillRR(28, y, 26, 26, 6, i ? C.mb : C.blue); bar(66, y + 4, 220, C.line); bar(66, y + 17, 140, C.line, 7); }
  }
  const sy = h - 136 * a;
  finger([[.3, 271, 162], [1.6, 271, 162], [2.75, w / 2, h - 136 + 13], [3.0, w / 2, h - 136 + 13], [3.6, w / 2, h + 10]], [1.0, 3.0], t, (t > .6 && t < 1.6) || (t > 2.75 && t < 3.6));
});
