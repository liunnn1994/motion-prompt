// 折叠面板 / Accordion
// 素材包: ui (UI 组件) · element 04 · 分类: 弹层与面板
// 示例内容: 点击第一个标题展开内容，下方条目被推开；再次点击收起。
// 动画规格:
//   - 卡片尺寸 542×191，每轮 4.5 秒。3 个标题行，宽 494、高 34、圆角半径 8、间距 6，右侧有向下箭头。
//   - 光标在 0.95 秒到达首行，分别在 1.0 秒和 3.3 秒点击。
//   - 展开进度 ex：1.0–1.35 秒按 easeInOutCubic 从 0 增至 1，3.3–3.62 秒从 1 降至 0。
//   - 首行高度从 34 增至 92，即增加 58·ex，下方条目被推开；两行正文裁剪在首行区域内，不透明度为 ex。
//   - 展开时，箭头旋转 180°·ex，边框变浅蓝，标题变蓝。
// 适用场景: 适合常见问题或详细说明，让用户按需展开。
// 标签: accordion, collapsible, expand, collapse, disclosure, faq, details
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://motion-prompt.2077tech.com/api/v1/kit/ui.js
// 动态示例: https://motion-prompt.2077tech.com/#052
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（由调用方清空画布并执行 save/restore），w、h = 尺寸，u = 秒数，范围为 [0, cycle)。画面仅由 u 决定。
// 以下为素材代码。CELLS、常量及辅助函数位于 kit 文件中。

// @element 04
// 04 アコーディオン（Accordion）
CELLS[3] = ui((w, h, t) => {
  const ex = eio(seg(t, 1.0, 1.35)) * (1 - eio(seg(t, 3.3, 3.62)));
  let y = 12; const H0 = 34, EX = 58;
  for (let i = 0; i < 3; i++) {
    const extra = i === 0 ? EX * ex : 0;
    strokeRR(24, y, 494, H0 + extra, 8, i === 0 && ex > .05 ? C.sb : C.line2, 1.5);
    bar(42, y + 12, [180, 220, 150][i], i === 0 && ex > .05 ? C.blue : C.line2, 10);
    chevron(494, y + 17, 6, i === 0 ? Math.PI * ex : 0, i === 0 && ex > .05 ? C.blue : '#9a9896');
    if (i === 0 && ex > 0) { X.save(); rr(24, y, 494, H0 + extra, 8); X.clip(); X.globalAlpha = ex; bar(42, y + H0 + 6, 420); bar(42, y + H0 + 24, 320); X.restore(); }
    y += H0 + extra + 6;
  }
  pointer([[.3, w + 30, h + 30], [.95, 300, 31], [1.6, 300, 31], [3.0, 300, 31], [3.55, 300, 31], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
