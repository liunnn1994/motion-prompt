// 轻提示 / Toast notification
// 素材包: ui (UI 组件) · element 08 · 分类: 弹层与面板
// 示例内容: 点击“保存”后，黑色通知滑入右下角，稍后自行滑出。
// 动画规格:
//   - 卡片尺寸 542×191，每轮 4.5 秒。1.0 秒点击左下角 110×36 的蓝色“保存”按钮，随后向左下移开。
//   - 轻提示：#26282c，214×52，圆角半径 10，距右边 18px、底边 16px；1.05–1.4 秒按 easeOutCubic 从右侧 240px 处滑入。
//   - 内部包含蓝圆中的白色勾号和 15px 白色“已保存”文字。
//   - 底部高 3px 的剩余时间进度线，在 1.4–3.2 秒从 194px 线性缩短至 0。
//   - 退场：3.2–3.55 秒按 easeInOutCubic 向右滑出 240px。
// 适用场景: 适合反馈保存、发送等操作结果，不打断用户。
// 标签: toast, snackbar, notification, alert, message, auto-dismiss
// 尺寸: 542x191 px · 周期: 4.5 s
// 依赖: kit ui -> https://motion-prompt.2077tech.com/api/v1/kit/ui.js
// 动态示例: https://motion-prompt.2077tech.com/#056
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（由调用方清空画布并执行 save/restore），w、h = 尺寸，u = 秒数，范围为 [0, cycle)。画面仅由 u 决定。
// 以下为素材代码。CELLS、常量及辅助函数位于 kit 文件中。

// @element 08
// 08 トースト（Toast notification）
CELLS[7] = ui((w, h, t) => {
  bar(24, 16, 90, C.line2, 10); strokeRR(24, 34, 300, 34, 7, C.line2, 1.5); bar(24, 82, 90, C.line2, 10); strokeRR(24, 100, 300, 34, 7, C.line2, 1.5);
  fillRR(24, 146, 110, 36, 8, C.blue); text('保存', 79, 164, 16, '#fff', 600, 'center');
  const tin = eo(seg(t, 1.05, 1.4)), tout = eio(seg(t, 3.2, 3.55));
  if (tin > 0 && tout < 1) {
    const tw0 = 214, th = 52, tx = w - 18 - tw0 + (1 - tin) * 240 + tout * 240, ty = h - 16 - th;
    shadowed(() => fillRR(tx, ty, tw0, th, 10, C.dark), 18, 6, .25);
    circle(tx + 26, ty + 24, 11, C.blue); check(tx + 26, ty + 24, 5, '#fff', 2.4);
    text('已保存', tx + 46, ty + 24, 15, '#fff', 600);
    const life = 1 - seg(t, 1.4, 3.2); fillRR(tx + 10, ty + th - 7, (tw0 - 20) * life, 3, 1.5, C.sb);
  }
  pointer([[.3, -30, h + 30], [.95, 84, 168], [1.5, 84, 168], [2.2, -40, h + 40]], [1.0], t);
});
