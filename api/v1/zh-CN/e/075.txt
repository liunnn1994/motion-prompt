// 吸顶导航 / Sticky header
// 素材包: lp (落地页与网页效果) · element 03 · 分类: 滚动效果
// 示例内容: 公告条滚出画面，下方导航停在顶端，正文在其下继续滚动。
// 动画规格:
//   - 滚动位置：0.4–1.3 秒为 0→110px，1.9–2.8 秒为 110→250px，3.4–4.4 秒返回 250→0，使用三次缓入缓出。
//   - 顶部有高 22px 的深蓝公告条，下方是高 34px 的页头，包含标志、三个菜单项和按钮。
//   - 页头位置 = max(22 − scroll, 0)；公告条滚出后页头停在顶部，相当于 position: sticky; top: 0。
//   - 随后 24px 的滚动距离内，页头下方模糊半径 12px 的阴影，不透明度从 0 增至 0.16。
//   - 照片、标题和三张卡片在页头下方正常滚动。
// 适用场景: 适合长页面中随时可用的导航和操作入口。
// 标签: sticky, sticky-header, fixed-header, position-sticky, navigation, header, scroll
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// 动态示例: https://motion-prompt.2077tech.com/#075
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（由调用方清空画布并执行 save/restore），w、h = 尺寸，u = 秒数，范围为 [0, cycle)。画面仅由 u 决定。
// 以下为素材代码。CELLS、常量及辅助函数位于 kit 文件中。

// @element 03
// 03 追従ヘッダー（スクロールすると、ヘッダーが画面の上で止まってついてくる）
CELLS[2] = (k, w, h, u) => {
  frame(k);
  const y = steps([[0.4, 0], [1.3, 110], [1.9, 110], [2.8, 250], [3.4, 250], [4.4, 0]], u);
  const STRIP = 22;
  view(k, () => {
    k.save(); k.translate(0, -y);
    pic(k, 24, 70, 214, 112, 0, 8);
    txt(k, '开启全新一天，', 262, 104, `800 22px ${JP}`, C.ink); bars(k, 262, 120, 210, 3); btn(k, 262, 160, 100, 26, '查看详情');
    k.fillStyle = C.soft; k.fillRect(0, 212, VW, 240);
    txt(k, '特色', VW / 2, 246, `800 18px ${JP}`, C.ink, 'center');
    for (let i = 0; i < 3; i++) { const x = 34 + i * 152; k.fillStyle = '#fff'; rr(k, x, 262, 136, 120, 8); k.fill(); pic(k, x + 8, 270, 120, 58, i, 5); bars(k, x + 10, 340, 110, 3, C.faint, 6, 11); }
    k.fillStyle = C.ink; k.fillRect(0, 0, VW, STRIP);
    txt(k, '限时优惠进行中', VW / 2, STRIP / 2 + 0.5, `600 11px ${JP}`, '#fff', 'center', 'middle');
    k.restore();
    const hy = Math.max(STRIP - y, 0), sh = clamp((y - STRIP) / 24);
    if (sh > 0) { const g = k.createLinearGradient(0, hy + 34, 0, hy + 46); g.addColorStop(0, `rgba(28,35,51,${0.16 * sh})`); g.addColorStop(1, 'rgba(28,35,51,0)'); k.fillStyle = g; k.fillRect(0, hy + 34, VW, 12); }
    header(k, hy);
    sbar(k, y / 250, 0.42);
  });
};
