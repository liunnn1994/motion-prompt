// 横向滚动 / Horizontal scroll
// 素材包: lp (落地页与网页效果) · element 05 · 分类: 滚动效果
// 示例内容: 页面向下滚动时区块保持固定，四张编号作品卡片向左移动。
// 动画规格:
//   - 滚动位置：0.3–3.0 秒为 0→420px，3.5–4.5 秒返回 420→0，使用三次缓入缓出。
//   - “WORKS”标题先滚出画面，滚动 40px 时作品区块到达顶部并固定。
//   - 4 张卡片各为 200 × 112，间距 20，两侧留白 34；总横移距离 = 34·2 + 200·4 + 20·3 − 511 = 417px。
//   - 卡片 x = 初始 x − p × 417，其中 p = (scroll − 40) / 380；位移与滚动线性对应，不加缓动。
//   - 卡片横向移动时，滚动条滑块继续向下。
// 适用场景: 适合在长页面中横向展示作品、案例或产品。
// 标签: horizontal-scroll, sideways, scrolltrigger, pin, gallery, portfolio, scroll
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// 动态示例: https://motion-prompt.2077tech.com/#077
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（由调用方清空画布并执行 save/restore），w、h = 尺寸，u = 秒数，范围为 [0, cycle)。画面仅由 u 决定。
// 以下为素材代码。CELLS、常量及辅助函数位于 kit 文件中。

// @element 05
// 05 横スクロール（縦にスクロールすると、カードの列が横へ流れる）
CELLS[4] = (k, w, h, u) => {
  frame(k);
  const y = steps([[0.3, 0], [3.0, 420], [3.5, 420], [4.5, 0]], u);
  const PIN0 = 40, PD = 380, N = 4, PWD = 200, GAP = 20;
  const top = y < PIN0 ? PIN0 - y : 0, p = clamp((y - PIN0) / PD);
  const travel = 34 * 2 + N * PWD + (N - 1) * GAP - VW;
  view(k, () => {
    txt(k, 'WORKS', 34, 30 - y, `800 20px ${EN}`, C.ink, 'left', 'alphabetic', '2px');
    k.save(); k.translate(0, top);
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH + 40);
    for (let i = 0; i < N; i++) {
      const x = 34 + i * (PWD + GAP) - p * travel;
      pic(k, x, 18, PWD, 112, i % 3, 8);
      txt(k, '0' + (i + 1), x + 14, 118, `800 22px ${EN}`, '#fff');
      bars(k, x, 140, 120, 1, '#CBD2DE');
    }
    k.restore();
    sbar(k, y / 420, 0.36);
  });
};
