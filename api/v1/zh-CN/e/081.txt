// 悬停抬升 / Hover lift
// 素材包: lp (落地页与网页效果) · element 09 · 分类: 悬停与点击交互
// 示例内容: 光标下的卡片微微抬起、阴影扩散，移开后复原。
// 动画规格:
//   - 3 张卡片各为 146 × 128，圆角半径 10；光标从左到右依次悬停，每张约 1 秒。
//   - 悬停时卡片用 0.3 秒上移 10px，移开时用 0.3 秒复原，均为三次缓出，类似 CSS transition。
//   - 阴影同步变化：模糊 6→22px，纵向偏移 2→12px，不透明度 0.08→0.22。
//   - 浅灰页面搭配白色卡片，便于看清阴影。
// 适用场景: 适合让卡片或按钮显得可点击。
// 标签: hover, lift, float, card, shadow, translatey, interaction
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// 动态示例: https://motion-prompt.2077tech.com/#081
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（由调用方清空画布并执行 save/restore），w、h = 尺寸，u = 秒数，范围为 [0, cycle)。画面仅由 u 决定。
// 以下为素材代码。CELLS、常量及辅助函数位于 kit 文件中。

// @element 09
// 09 ホバーで浮き上がる（カーソルが乗ったカードが、影をつけて少し持ち上がる）
CELLS[8] = (k, w, h, u) => {
  frame(k);
  const CX = [26, 192, 358], CY = 18, CWD = 146, CHT = 128, LIFT = 10;
  const cur = t => path([[0, 70, 160], [0.45, 96, 78], [1.4, 110, 86], [1.62, 262, 80], [2.5, 276, 88], [2.72, 428, 80], [3.6, 440, 88], [4.0, 470, 160], [4.8, 70, 160]], t);
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    for (let i = 0; i < 3; i++) {
      const inside = t => { const [x, y] = cur(t); return x > CX[i] && x < CX[i] + CWD && y > CY && y < CY + CHT; };
      const q = trans(inside, u, 0.3), x = CX[i], y = CY - LIFT * q;
      k.save(); k.shadowColor = `rgba(28,35,51,${0.08 + 0.14 * q})`; k.shadowBlur = 6 + 16 * q; k.shadowOffsetY = 2 + 10 * q;
      k.fillStyle = '#fff'; rr(k, x, y, CWD, CHT, 10); k.fill(); k.restore();
      pic(k, x + 8, y + 8, CWD - 16, 66, i, 6);
      txt(k, ['山间旅馆', '海边旅馆', '星空旅馆'][i], x + 12, y + 96, `800 13px ${JP}`, C.ink); bars(k, x + 12, y + 106, 100, 1);
    }
    const [cx, cy] = cur(u); cursor(k, cx, cy);
  });
};
