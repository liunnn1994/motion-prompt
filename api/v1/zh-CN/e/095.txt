// 动态渐变 / Animated gradient
// 素材包: lp (落地页与网页效果) · element 23 · 分类: 加载、导航与循环动效
// 示例内容: 标题后方柔和的橙、浅橙、蓝、浅蓝色块绕圈漂移，相互融合。
// 动画规格:
//   - 底色 #FFF1EB，上方叠加 4 个径向渐变，从中心颜色过渡到透明边缘，各层不透明度 0.85。
//   - 四层分别为橙色半径 230、浅橙 #FFB08F 半径 210、蓝色半径 240、浅蓝 #9DB6FB 半径 200。
//   - 各渐变中心每 4.8 秒沿椭圆移动一圈；横向半径为 70、60、80、50，纵向半径为 30、40、30、35，起始角度为 0、2.1、4.2、1.0rad。
//   - 上方白色标题为 34px，另有按钮；仅渐变背景运动。
// 适用场景: 适合无需照片也能为首屏或标题增加色彩和动感。
// 标签: gradient, animated-gradient, mesh-gradient, background, hero, ambient, loop
// 尺寸: 539x208 px · 周期: 4.8 s
// 依赖: kit lp -> https://motion-prompt.2077tech.com/api/v1/kit/lp.js
// 动态示例: https://motion-prompt.2077tech.com/#095
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（由调用方清空画布并执行 save/restore），w、h = 尺寸，u = 秒数，范围为 [0, cycle)。画面仅由 u 决定。
// 以下为素材代码。CELLS、常量及辅助函数位于 kit 文件中。

// @element 23
// 23 グラデーションアニメーション（背景のグラデーションの色が、ゆっくり流れるように動き続ける）
CELLS[22] = (k, w, h, u) => {
  frame(k);
  const a = 2 * Math.PI * u / CYC;                        // 4.8秒で1周
  const BLOBS = [[C.acc, 150, 60, 230, 70, 30, 0], ['#FFB08F', 380, 40, 210, 60, 40, 2.1], [C.blue, 330, 150, 240, 80, 30, 4.2], ['#9DB6FB', 90, 170, 200, 50, 35, 1.0]];
  view(k, () => {
    k.fillStyle = '#FFF1EB'; k.fillRect(0, 0, VW, VH);
    for (const [col, x0, y0, r, ax, ay, ph] of BLOBS) {
      const x = x0 + ax * Math.cos(a + ph), y = y0 + ay * Math.sin(a + ph);
      const g = k.createRadialGradient(x, y, 0, x, y, r); g.addColorStop(0, col); g.addColorStop(1, 'rgba(255,255,255,0)');
      k.globalAlpha = 0.85; k.fillStyle = g; k.fillRect(0, 0, VW, VH); k.globalAlpha = 1;
    }
    txt(k, 'Hello, Future.', 40, 82, `800 34px ${EN}`, '#fff', 'left', 'alphabetic', '1px');
    bars(k, 40, 100, 200, 2, 'rgba(255,255,255,.8)', 6, 12, 0.6);
    btn(k, 40, 128, 104, 26, '开始使用', '#fff', C.ink);
  });
};
