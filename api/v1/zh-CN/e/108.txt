// 追随拍摄 / Panning shot
// 素材包: photo (摄影效果、布光与风格) · element 12 · 分类: 镜头与相机效果
// 示例内容: 相机跟随汽车，车身保持在中央且清晰，背景建筑、树木和灯光横向模糊。
// 动画规格:
//   - 街角场景：道路 y = 132–162，后方为建筑、树和路灯，无前景大人物。
//   - 无效果时，汽车用 5 秒从 x = −80 行驶至 620；效果完全呈现时，汽车保持在 x = 270 ±10px。
//   - 背景沿水平方向分布渲染 12 次并取平均，采样范围由 0 增至 46px，形成拖影。
//   - 车轮辐条每 5 秒旋转 12 圈。
//   - 右上角快门读数 1/1000→1/15。
//   - 时序：0.4–1.4 秒效果逐渐增强，保持至 4.0 秒，4.8 秒前恢复原状；每轮 5 秒，使用三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合在静态照片中表现汽车、自行车或跑者的速度。
// 标签: panning, panning shot, motion blur, speed, sports photography, slow shutter
// 尺寸: 539x208 px · 周期: 5 s
// 依赖: kit photo -> https://motion-prompt.2077tech.com/api/v1/kit/photo.js
// 动态示例: https://motion-prompt.2077tech.com/#108
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（由调用方清空画布并执行 save/restore），w、h = 尺寸，u = 秒数，范围为 [0, cycle)。画面仅由 u 决定。
// 以下为素材代码。CELLS、常量及辅助函数位于 kit 文件中。

// @element 12
// 流し撮り: 走る車を追ってカメラを振ると、車はくっきり、背景だけ横に流れる
CELLS[11] = cell((k, u) => {
  const a = env(u);
  const drive = lerp(-80, 620, u / CYC), x = lerp(drive, 270 + 10 * Math.sin(u / CYC * TAU), a);
  const S = 46 * a, n = S < 0.5 ? 1 : 12;
  for (let i = 0; i < n; i++) {   // 背景を横にずらして重ね、平均をとる（流れ）
    k.save(); k.globalAlpha = 1 / (i + 1); k.translate(n > 1 ? lerp(-S / 2, S / 2, i / (n - 1)) : 0, 0); stageStreet(k, BASE, { car: false, person: false }); k.restore();
  }
  const D = pen(k); car(D, x, BASE, { head: BASE.lit });
  // 車輪の回転（1周に12回転＝継ぎ目なし）
  k.save(); k.strokeStyle = BASE.tire; k.lineWidth = 1.6;
  for (const wx of [x - 21, x + 21]) { const t = u / CYC * TAU * 12; k.beginPath(); k.moveTo(wx - Math.cos(t) * 5, 156 - Math.sin(t) * 5); k.lineTo(wx + Math.cos(t) * 5, 156 + Math.sin(t) * 5); k.stroke(); }
  k.restore();
  hud(k, a > 0.5 ? '1/15' : '1/1000');
});
