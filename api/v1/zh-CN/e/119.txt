// 漏光 / Light leak
// 素材包: photo (摄影效果、布光与风格) · element 23 · 分类: 影调与后期风格
// 示例内容: 画面右边缘渗入缓慢晃动的光带，内部淡黄、边缘偏红。
// 动画规格:
//   - 场景 3：左侧为半身像，头部圆心（140, 82）、半径 28px；右侧为山丘、房屋、树和太阳，太阳圆心（432, 54）、半径 15px。
//   - 4 个竖向椭圆中心位于右边缘外 30px，周期内横摆 ±6px，纵向半径为画面高度的 1.1 倍。
//   - 各椭圆横向半径从 0 增至目标值：250px、#FFD089、35%；170px、#FF9A3D、40%；95px、#FF5A3A、45%；45px、#FF3B30、50%。百分比为不透明度。
//   - 时序：0.4–1.4 秒效果逐渐增强，保持至 4.0 秒，4.8 秒前恢复原状；每轮 5 秒，使用三次缓入缓出，约为 cubic-bezier(0.65,0,0.35,1)。
// 适用场景: 适合怀旧、夏日和偶然记录般的回忆。
// 标签: light leak, film leak, vintage, analog, lomo, warm glow
// 尺寸: 539x208 px · 周期: 5 s
// 依赖: kit photo -> https://motion-prompt.2077tech.com/api/v1/kit/photo.js
// 动态示例: https://motion-prompt.2077tech.com/#119
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（由调用方清空画布并执行 save/restore），w、h = 尺寸，u = 秒数，范围为 [0, cycle)。画面仅由 u 决定。
// 以下为素材代码。CELLS、常量及辅助函数位于 kit 文件中。

// @element 23
// ライトリーク: 右端から、フィルムに光がもれたような赤と橙のにじみが帯になって入る
CELLS[22] = cell((k, u) => {
  const a = env(u), w = Math.sin(u / CYC * TAU);
  stageLand(k, BASE);
  k.save();
  // 縦長の楕円を右端の外に中心を置いて重ねる（外ほど淡い黄、端ほど赤）
  const band = (rx, col, al) => { k.globalAlpha = al * a; k.fillStyle = col; k.beginPath(); k.ellipse(CW + 30 + 6 * w, CH / 2, rx * a, CH * 1.1, 0, 0, TAU); k.fill(); };
  band(250, '#FFD089', 0.35); band(170, '#FF9A3D', 0.4); band(95, '#FF5A3A', 0.45); band(45, '#FF3B30', 0.5);
  k.restore();
});
